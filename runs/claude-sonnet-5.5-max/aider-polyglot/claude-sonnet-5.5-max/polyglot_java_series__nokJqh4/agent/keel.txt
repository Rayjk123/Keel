import java.util.ArrayList;
import java.util.List;

class Series {
    private final String string;

    Series(String string) {
        if (string == null || string.isEmpty()) {
            throw new IllegalArgumentException("Series cannot be empty.");
        }
        this.string = string;
    }

    List<String> slices(int num) {
        if (num <= 0) {
            throw new IllegalArgumentException("Slice size is too small.");
        }
        if (num > string.length()) {
            throw new IllegalArgumentException("Slice size is too big.");
        }

        List<String> slices = new ArrayList<>();
        for (int start = 0; start <= string.length() - num; start++) {
            slices.add(string.substring(start, start + num));
        }
        return slices;
    }
}

import java.util.List;

public class Check {
    static int failures = 0;

    static void expectEquals(String name, Object expected, Object actual) {
        if (expected.equals(actual)) {
            System.out.println("ok   " + name);
        } else {
            failures++;
            System.out.println("FAIL " + name + ": expected " + expected + " but got " + actual);
        }

    static void expectIae(String name, String message, Runnable action) {
        try {
            action.run();
            failures++;
            System.out.println("FAIL " + name + ": no exception thrown");
        } catch (IllegalArgumentException e) {
            if (message.equals(e.getMessage())) {
                System.out.println("ok   " + name);
            } else {
                failures++;
                System.out.println("FAIL " + name + ": wrong message: " + e.getMessage());
            }
        }
    }

    public static void main(String[] args) {
        expectEquals("one from one", List.of("1"), new Series("1").slices(1));
        expectEquals("one from two", List.of("1", "2"), new Series("12").slices(1));
        expectEquals("two from two", List.of("35"), new Series("35").slices(2));

        expectEquals("example 3-digit", List.of("491", "914", "142"), new Series("49142").slices(3));
        expectEquals("example 4-digit", List.of("4914", "9142"), new Series("49142").slices(4));
        expectEquals("whole string", List.of("49142"), new Series("49142").slices(5));

        expectEquals("repeat calls on same instance", reused.slices(2), reused.slices(2));

        expectIae("too big", "Slice size is too big.", () -> new Series("12345").slices(6));
        expectIae("zero", "Slice size is too small.", () -> new Series("12345").slices(0));
        expectIae("negative", "Slice size is too small.", () -> new Series("123").slices(-1));
        expectIae("empty series", "Series cannot be empty.", () -> new Series(""));

        System.out.println(failures == 0 ? "ALL PASSED" : failures + " FAILURE(S)");
        System.exit(failures == 0 ? 0 : 1);
    }
}
