import re

VOWELS = "aeiou"


def _split_index(word):
    """Return the index where the leading consonant cluster of ``word`` ends."""
    lower = word.lower()

    # Rule 1: a leading vowel, "xr" or "yt" sounds like a vowel -> nothing to move.
    if lower[0] in VOWELS or lower.startswith(("xr", "yt")):
        return 0

    index = 0
    while index < len(lower):
        char = lower[index]
        if char in VOWELS:
            # Rule 3: the "u" in "qu" travels together with the "q".
            if char == "u" and index > 0 and lower[index - 1] == "q":
                index += 1
                continue
            break
        # Rule 4: a "y" following one or more consonants acts like a vowel.
        if char == "y" and index > 0:
            break
        index += 1
    return index


def _translate_word(word):
    index = _split_index(word)
    return word[index:] + word[:index] + "ay"


def translate(text):
    # Translate every whitespace-separated word, keeping the original spacing.
    return re.sub(r"\S+", lambda match: _translate_word(match.group()), text)
