import string


class PhoneNumber:
    """A telephone number from the North American Numbering Plan (NANP).

    The user-entered number is cleaned of formatting characters and of the
    optional country code (``1`` or ``+1``), then validated against the NANP
    format ``NXX NXX-XXXX`` where ``N`` is a digit from 2-9 and ``X`` is a
    digit from 0-9.

    A ``ValueError`` with a descriptive message is raised for invalid numbers.
    """

    # Characters that merely format a number; they are dropped when cleaning.
    _SEPARATORS = "+().-"

    def __init__(self, number):
        self.number = self._clean(number)
        self.area_code = self.number[:3]
        self.exchange_code = self.number[3:6]
        self.subscriber_number = self.number[6:]

    def pretty(self):
        """Return the number formatted as ``(NXX)-NXX-XXXX``."""
        return (
            f"({self.area_code})-"
            f"{self.exchange_code}-{self.subscriber_number}"
        )

    @classmethod
    def _clean(cls, raw):
        """Return the bare 10-digit number or raise ``ValueError``."""
        if any(ch.isalpha() for ch in raw):
            raise ValueError("letters not permitted")
        if not all(cls._is_permitted(ch) for ch in raw):
            raise ValueError("punctuations not permitted")

        digits = "".join(ch for ch in raw if ch in string.digits)

        if len(digits) < 10:
            raise ValueError("must not be fewer than 10 digits")
        if len(digits) > 11:
            raise ValueError("must not be greater than 11 digits")
        if len(digits) == 11:
            if digits[0] != "1":
                raise ValueError("11 digits must start with 1")
            digits = digits[1:]

        if digits[0] == "0":
            raise ValueError("area code cannot start with zero")
        if digits[0] == "1":
            raise ValueError("area code cannot start with one")
        if digits[3] == "0":
            raise ValueError("exchange code cannot start with zero")
        if digits[3] == "1":
            raise ValueError("exchange code cannot start with one")

        return digits

    @classmethod
    def _is_permitted(cls, ch):
        """Digits, whitespace and common formatting characters are allowed."""
        return ch in string.digits or ch.isspace() or ch in cls._SEPARATORS
