/**
 * Calculates the energy points awarded to a player for completing a level.
 *
 * For every magical item that was found, all multiples of its base value that
 * are strictly less than the level number are collected. Numbers that turn up
 * for more than one item are only counted once, and the energy points are the
 * sum of everything that is left.
 *
 * Example: sum([3, 5], 20) === 78
 *
 * @param {number[]} baseValues - base values of the magical items that were found
 * @param {number} level - the level that was completed
 * @returns {number} the energy points to award
 */
export const sum = (baseValues, level) => {
  // The task description doesn't pin down the argument order, so accept both
  // sum(baseValues, level) and sum(level, baseValues).
  const [values, limit] =
    typeof baseValues === 'number' ? [level, baseValues] : [baseValues, level];

  const multiples = new Set();

  for (const value of values) {
    // A base value of 0 (or a negative / fractional one) has no meaningful
    // multiples below the level, so such an item contributes nothing.
    if (Number.isInteger(value) && value > 0) {
      for (let multiple = value; multiple < limit; multiple += value) {
        multiples.add(multiple);
      }
    }
  }

  let total = 0;
  for (const multiple of multiples) {
    total += multiple;
  }

  return total;
};
EOF

// Literal brute-force version of the rules, used as an independent reference.
const brute = (values, level) => {
  let total = 0;

- **How it works:** for each base value, it collects every multiple strictly below the level into a `Set`, which removes duplicates. It then adds up what is left.
- **Zero, negative and fractional base values:** these contribute nothing. This also stops a base value of `0` from looping forever.
- **Argument order:** the task text doesn't say which comes first, so the function accepts both `sum(baseValues, level)` (the usual convention) and `sum(level, baseValues)`.
