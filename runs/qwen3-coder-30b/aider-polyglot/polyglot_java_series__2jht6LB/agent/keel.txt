public class Series {
    private final String digits;

    public Series(String input) {
        if (input == null) {
            throw new IllegalArgumentException("String to search must be non-null.");
        }
        
        for (int i = 0; i < input.length(); i++) {
            if (!Character.isDigit(input.charAt(i))) {
                throw new IllegalArgumentException("String to search may only contain digits.");
            }
        }
        
        this.digits = input;
    }

    public int[] getDigits() {
        int[] result = new int[digits.length()];
        for (int i = 0; i < digits.length(); i++) {
            result[i] = Character.getNumericValue(digits.charAt(i));
        }
        return result;
    }

    public int[] getSpan(int length) {
        if (length < 0) {
            throw new IllegalArgumentException("span must be greater than zero.");
        }
        
        if (length > digits.length()) {
            throw new IllegalArgumentException("span must be smaller than string length.");
        }
        
        int[] result = new int[length];
        for (int i = 0; i < length; i++) {
            result[i] = Character.getNumericValue(digits.charAt(i));
        }
        return result;
    }

    public Iterable<int[]> getSpans(int length) {
        if (length < 0) {
            throw new IllegalArgumentException("span must be greater than zero.");
        }
        
        if (length > digits.length()) {
            throw new IllegalArgumentException("span must be smaller than string length.");
        }
        
        // This would return an iterable of all spans, but since we need to implement
        // the specific method that returns contiguous substrings of length n, 
        // I'll create a proper implementation
        return new java.util.AbstractList<int[]>() {
            @Override
            public int size() {
                return digits.length() - length + 1;
            }
            
            @Override
            public int[] get(int index) {
                if (index < 0 || index >= size()) {
                    throw new IndexOutOfBoundsException();
                }
                
                int[] result = new int[length];
                for (int i = 0; i < length; i++) {
                    result[i] = Character.getNumericValue(digits.charAt(index + i));
                }
                return result;
            }
        };
    }
}
