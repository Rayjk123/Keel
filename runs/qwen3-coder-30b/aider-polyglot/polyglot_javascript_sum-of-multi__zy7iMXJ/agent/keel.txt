export const to = (limit, factors) => {
  // Handle edge cases
  if (limit <= 0 || !factors || factors.length === 0) {
    return 0;
  }
  
  // Use a Set to automatically handle duplicates
  const multiples = new Set();
  
  // For each factor, find all multiples less than limit
  for (const factor of factors) {
    // Skip zero as it would lead to infinite multiples
    if (factor === 0) continue;
    
    // Find all multiples of this factor less than limit
    for (let multiple = factor; multiple < limit; multiple += factor) {
      multiples.add(multiple);
    }
  }
  
  // Calculate sum of all unique multiples
  let sum = 0;
  for (const multiple of multiples) {
    sum += multiple;
  }
  
  return sum;
};
EOF
