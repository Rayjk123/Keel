package hexadecimal

import (
	"errors"
	"strings"
)

func ParseHex(hex string) (int, error) {
	if hex == "" {
		return 0, errors.New("invalid hexadecimal string")
	}

	hex = strings.ToLower(hex)
	result := 0

	for _, char := range hex {
		var digitValue int
		switch char {
		case '0':
			digitValue = 0
		case '1':
			digitValue = 1
		case '2':
			digitValue = 2
		case '3':
			digitValue = 3
		case '4':
			digitValue = 4
		case '5':
			digitValue = 5
		case '6':
			digitValue = 6
		case '7':
			digitValue = 7
		case '8':
			digitValue = 8
		case '9':
			digitValue = 9
		case 'a':
			digitValue = 10
		case 'b':
			digitValue = 11
		case 'c':
			digitValue = 12
		case 'd':
			digitValue = 13
		case 'e':
			digitValue = 14
		case 'f':
			digitValue = 15
		default:
			return 0, errors.New("invalid hexadecimal character")
		}
		
		result = result*16 + digitValue
	}

	return result, nil
}
