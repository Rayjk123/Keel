$ echo "export class Triangle {
  constructor(a, b, c) {
    if (a <= 0 || b <= 0 || c <= 0) {
      throw new Error('Not a triangle');
    }
    if (a + b < c || b + c < a || a + c < b) {
      throw new Error('Not a triangle');
    }
    this.a = a;
    this.b = b;
    this.c = c;
  }

  get isEquilateral() {
    return this.a === this.b && this.b === this.c;
  }

  get isIsosceles() {
    return this.a === this.b || this.b === this.c || this.a === this.c;
  }

  get isScalene() {
    return this.a !== this.b && this.b !== this.c && this.a !== this.c;
  }
