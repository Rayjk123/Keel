package trinary

import (
	"math"
	"strconv"
)

func ParseTrinary(arg string) (int64, bool) {
	// Check if the string contains invalid trinary characters
	for _, char := range arg {
		if char != '0' && char != '1' && char != '2' {
			return 0, false
		}
	}

	// Calculate the decimal equivalent
	decimalValue := int64(0)
	length := len(arg)

	for i, char := range arg {
		value := int64(char - '0')
		decimalValue += value * int64(math.Pow(3, float64(length-i-1)))
	}

	return decimalValue, true
}
