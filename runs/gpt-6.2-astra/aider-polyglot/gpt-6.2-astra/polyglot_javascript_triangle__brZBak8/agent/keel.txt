export class Triangle {
  constructor(...sides) {
    const [a, b, c] = sides;
    const valid = sides.length === 3
      && sides.every(side => Number.isFinite(side) && side > 0)
      && a + b >= c && b + c >= a && a + c >= b;

    this.distinctSideCount = valid ? new Set(sides).size : 0;
  }

  get isEquilateral() {
    return this.distinctSideCount === 1;
  }

  get isIsosceles() {
    return this.isEquilateral || this.distinctSideCount === 2;
  }

  get isScalene() {
    return this.distinctSideCount === 3;
  }
}
