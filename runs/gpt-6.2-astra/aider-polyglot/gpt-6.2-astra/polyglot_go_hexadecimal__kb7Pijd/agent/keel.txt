package hexadecimal

import "errors"

// ParseHex converts a nonempty string of hexadecimal digits to an int64.
// It returns an error for invalid digits or values that do not fit.
func ParseHex(in string) (int64, error) {
	if len(in) == 0 {
		return 0, errors.New("empty hexadecimal string")
	}

	const maxInt64 = 1<<63 - 1
	var value int64
	for i := 0; i < len(in); i++ {
		c := in[i]
		var digit int64
		switch {
		case '0' <= c && c <= '9':
			digit = int64(c - '0')
		case 'a' <= c && c <= 'f':
			digit = int64(c-'a') + 10
		case 'A' <= c && c <= 'F':
			digit = int64(c-'A') + 10
		default:
			return 0, errors.New("invalid hexadecimal digit")
		}

		if value > (maxInt64-digit)/16 {
			return 0, errors.New("hexadecimal value exceeds int64")
		}
		value = value*16 + digit
	}
	return value, nil
}
EOF

	for in, want := range map[string]int64{"0": 0, "008000": 32768, "008080": 32896, "000080": 128, "10aF8c": 1093516, "7fffffffffffffff": 1<<63-1} {

}
