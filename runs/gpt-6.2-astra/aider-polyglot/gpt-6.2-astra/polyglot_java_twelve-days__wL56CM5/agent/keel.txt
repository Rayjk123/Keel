import java.util.StringJoiner;

class TwelveDays {
    private static final String[] DAYS = {
        "first", "second", "third", "fourth", "fifth", "sixth",
        "seventh", "eighth", "ninth", "tenth", "eleventh", "twelfth"
    };

    private static final String[] GIFTS = {
        "a Partridge in a Pear Tree", "two Turtle Doves", "three French Hens",
        "four Calling Birds", "five Gold Rings", "six Geese-a-Laying",
        "seven Swans-a-Swimming", "eight Maids-a-Milking", "nine Ladies Dancing",
        "ten Lords-a-Leaping", "eleven Pipers Piping", "twelve Drummers Drumming"
    };

    String verse(int verseNumber) {
        StringJoiner gifts = new StringJoiner(", ");
        for (int i = verseNumber - 1; i > 0; i--) {
            gifts.add(GIFTS[i]);
        }
        gifts.add(verseNumber == 1 ? GIFTS[0] : "and " + GIFTS[0]);
        return "On the " + DAYS[verseNumber - 1] + " day of Christmas my true love gave to me: "
                + gifts + ".";
    }

    String verses(int startVerse, int endVerse) {
        StringJoiner lyrics = new StringJoiner("\n\n");
        for (int day = startVerse; day <= endVerse; day++) {
            lyrics.add(verse(day));
        }
        return lyrics.toString();
    }

    String sing() {
        return verses(1, DAYS.length);
    }
}
EOF

class Check {
    public static void main(String[] args) {
        TwelveDays song = new TwelveDays();
        String first = "On the first day of Christmas my true love gave to me: a Partridge in a Pear Tree.";
        String second = "On the second day of Christmas my true love gave to me: two Turtle Doves, and a Partridge in a Pear Tree.";
        assert song.verses(1, 2).equals(first + "\n\n" + second);
        String[] verses = song.sing().split("\n\n", -1);
        assert verses.length == 12;
        assert verses[11].startsWith("On the twelfth day of Christmas my true love gave to me: twelve Drummers Drumming, eleven Pipers Piping, ");
        assert verses[11].endsWith(", and a Partridge in a Pear Tree.");
        for (int i = 0; i < verses.length; i++) {
            assert verses[i].equals(song.verse(i + 1));
            assert verses[i].split(", ").length == i + 1;
        }
    }
}
EOF
