export class Triangle {
  constructor(...sides) {
    this.sides = [...sides].sort((x, y) => x - y);
  }

  get isValid() {
    const [a, b, c] = this.sides;
    return (
      this.sides.length === 3 &&
      a > 0 &&
      a + b >= c
    );
  }

  get isEquilateral() {
    return this.isValid && this.sides[0] === this.sides[2];
  }

  get isIsosceles() {
    return this.isValid && (
      this.sides[0] === this.sides[1] ||
      this.sides[1] === this.sides[2]
    );
  }

  get isScalene() {
    return this.isValid &&
      this.sides[0] !== this.sides[1] &&
      this.sides[1] !== this.sides[2];
  }
}
