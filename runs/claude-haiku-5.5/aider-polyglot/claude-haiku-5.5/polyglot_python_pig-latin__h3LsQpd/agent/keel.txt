VOWELS = "aeiou"


def _translate_word(word):
    lower = word.lower()
    if lower[0] in VOWELS or lower.startswith(("xr", "yt")):
        return word + "ay"

    i = 0
    while i < len(word):
        ch = lower[i]
        if ch in VOWELS:
            break
        if ch == "q" and i + 1 < len(word) and lower[i + 1] == "u":
            i += 2
            break
        if ch == "y" and i > 0:
            break
        i += 1
    return word[i:] + word[:i] + "ay"


def translate(text):
    return " ".join(_translate_word(w) for w in text.split())
