package hexadecimal

import "errors"

// ErrInvalidHex is returned when the input is not a valid hexadecimal string.
var ErrInvalidHex = errors.New("invalid hexadecimal input")

// ErrOverflow is returned when the value does not fit into an int64.
var ErrOverflow = errors.New("hexadecimal value overflows int64")

// digitValue converts a single hex character to its numeric value.
func digitValue(c byte) (int64, bool) {
	switch {
	case c >= '0' && c <= '9':
		return int64(c - '0'), true
	case c >= 'a' && c <= 'f':
		return int64(c-'a') + 10, true
	case c >= 'A' && c <= 'F':
		return int64(c-'A') + 10, true
	}
	return 0, false
}

// ParseHex converts a hexadecimal string (e.g. "10af8c", "008000", or "0x1F")
// to its decimal value without using any conversion library.
// It returns an error for empty, malformed, or out-of-range input.
func ParseHex(in string) (int64, error) {
	s := in
	if len(s) >= 2 && s[0] == '0' && (s[1] == 'x' || s[1] == 'X') {
		s = s[2:]
	}
	if len(s) == 0 {
		return 0, ErrInvalidHex
	}

	var result int64
	for i := 0; i < len(s); i++ {
		d, ok := digitValue(s[i])
		if !ok {
			return 0, ErrInvalidHex
		}
		if result > (1<<63-1-d)/16 {
			return 0, ErrOverflow
		}
		result = result*16 + d
	}
	return result, nil
}
EOF
