package trinary

// ParseTrinary converts a trinary string to its decimal value.
// It returns the value and true when the input is a valid trinary number.
// Invalid input (empty string or any character other than '0', '1', '2')
// yields 0 and false.
func ParseTrinary(arg string) (int64, bool) {
	if len(arg) == 0 {
		return 0, false
	}
	var result int64
	for _, c := range arg {
		if c < '0' || c > '2' {
			return 0, false
		}
		result = result*3 + int64(c-'0')
	}
	return result, true
}
