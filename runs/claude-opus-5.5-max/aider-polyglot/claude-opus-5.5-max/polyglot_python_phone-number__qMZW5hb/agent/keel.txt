"""Clean up user-entered NANP phone numbers so they can be sent SMS messages."""

import string

# Characters allowed in a user-entered phone number (in addition to
# whitespace): the digits themselves plus the usual formatting punctuation,
# e.g. "+1 (613)-995-0253" or "613.995.0253".
_ALLOWED = frozenset(string.digits + "+()-.")


def _clean(raw):
    """Return the ten-digit NANP number contained in *raw*.

    Punctuation, whitespace and the country code (``1`` or ``+1``) are
    removed.  A ``ValueError`` with a descriptive message is raised if *raw*
    is not a valid NANP phone number.
    """
    if not isinstance(raw, str):
        raise TypeError(f"phone number must be a str, not {type(raw).__name__}")

    if any(char.isalpha() for char in raw):
        raise ValueError("letters not permitted")
    if not all(char in _ALLOWED or char.isspace() for char in raw):
        raise ValueError("punctuations not permitted")

    digits = "".join(char for char in raw if char in string.digits)

    if len(digits) < 10:
        raise ValueError("must not be fewer than 10 digits")
    if len(digits) > 11:
        raise ValueError("must not be greater than 11 digits")
    if len(digits) == 11:
        if digits[0] != "1":
            raise ValueError("11 digits must start with 1")
        digits = digits[1:]  # strip the country code

    # The format is NXX NXX-XXXX, where N is any digit from 2 through 9.
    if digits[0] == "0":
        raise ValueError("area code cannot start with zero")
    if digits[0] == "1":
        raise ValueError("area code cannot start with one")
    if digits[3] == "0":
        raise ValueError("exchange code cannot start with zero")
    if digits[3] == "1":
        raise ValueError("exchange code cannot start with one")

    return digits


class PhoneNumber:
    """A valid NANP phone number, e.g. ``PhoneNumber("+1 (613)-995-0253")``.

    ``number`` holds the cleaned ten-digit number (``"6139950253"``), made up
    of ``area_code`` (``"613"``), ``exchange_code`` (``"995"``) and
    ``subscriber_number`` (``"0253"``).  The constructor raises ``ValueError``
    if the input is not a valid NANP phone number.
    """

    def __init__(self, number):
        self.number = _clean(number)

    @property
    def area_code(self):
        """The three-digit Numbering Plan Area code."""
        return self.number[:3]

    @property
    def exchange_code(self):
        """The first three digits of the seven-digit local number."""
        return self.number[3:6]

    @property
    def subscriber_number(self):
        """The last four digits of the seven-digit local number."""
        return self.number[6:]

    def pretty(self):
        """Return the number formatted as ``(NXX)-NXX-XXXX``."""
        return f"({self.area_code})-{self.exchange_code}-{self.subscriber_number}"

import unittest

from phone_number import PhoneNumber


class PhoneNumberTest(unittest.TestCase):
    def assert_error(self, phrase, message):

        self.assertEqual(PhoneNumber("223.456.7890").number, "2234567890")

        self.assertEqual(PhoneNumber("223 456   7890   ").number, "2234567890")

    def test_valid_when_11_digits_and_starting_with_1(self):
        self.assertEqual(PhoneNumber("12234567890").number, "2234567890")

        self.assertEqual(PhoneNumber("+1 (223) 456-7890").number, "2234567890")

        for phrase in ["+1 (613)-995-0253", "613-995-0253", "1 613 995 0253", "613.995.0253"]:

                self.assertEqual(PhoneNumber(phrase).number, "6139950253")

    def test_invalid_when_9_digits(self):
        self.assert_error("123456789", "must not be fewer than 10 digits")

    def test_invalid_when_11_digits_does_not_start_with_a_1(self):
        self.assert_error("22234567890", "11 digits must start with 1")

I implemented `PhoneNumber` in `phone_number.py`. A 17-case test suite I wrote from the spec passes, covering every required error message. I've removed the test file and the cache directory it created, so `phone_number.py` is the only changed file. I didn't open or use the encrypted reference solution in `.oracle/`.

**What it does:**
- **`PhoneNumber(number).number`** gives the cleaned ten-digit number. For example, `+1 (613)-995-0253`, `613-995-0253`, `1 613 995 0253` and `613.995.0253` all become `6139950253`.
- **Order of checks:** an invalid number raises a `ValueError` with one of the exact messages from the instructions. Letters are checked first, then punctuation, then digit count, then the country code, area code and exchange code. The order matters for an input like `523-abc-7890`: it has only 7 digits, but it must report "letters not permitted" rather than "too few digits".
- **Allowed characters:** spaces and the usual formatting characters `+ ( ) - .` are accepted. Anything else, like `@`, `:`, `!` or `#`, gives "punctuations not permitted".
- **Country code:** a leading `1` on an 11-digit number is removed before the area and exchange codes are checked. So `1 (123) 456-7890` correctly reports "area code cannot start with one".
- **Extras:** `area_code`, `exchange_code` and `subscriber_number` properties, and a `pretty()` method that returns `(223)-456-7890`.

**One addition beyond the spec:** passing something that isn't a string, like an integer, raises a `TypeError` with a clear message instead of failing with a confusing error.
