"""Translate English text to Pig Latin."""

import re

VOWELS = frozenset("aeiou")

# Letter pairs that start with a consonant but are pronounced like a vowel.
VOWEL_SOUND_PREFIXES = ("xr", "yt")

# A word is a run of letters, optionally with internal apostrophes ("don't").
# Everything else in the text (whitespace, punctuation, ...) is kept as is.
WORD = re.compile(r"[A-Za-z]+(?:'[A-Za-z]+)*")


def translate(text):
    """Translate every word in ``text`` to Pig Latin.

    >>> translate("quick fast run")
    'ickquay astfay unray'
    """
    return WORD.sub(lambda match: translate_word(match.group()), text)


def translate_word(word):
    """Translate a single word to Pig Latin.

    The word's leading consonant sound (if any) is moved to the end of the
    word, and then "ay" is appended.
    """
    split = _consonant_sound_length(word.lower())
    return word[split:] + word[:split] + "ay"


def _consonant_sound_length(word):
    """Return how many leading letters of a lowercase ``word`` must move."""
    # Rule 1: "xr" and "yt" sound like vowels, so nothing moves.
    if word.startswith(VOWEL_SOUND_PREFIXES):
        return 0

    for index, letter in enumerate(word):
        # Rule 1 (starts with a vowel) and Rule 2 (consonants before a vowel).
        if letter in VOWELS:
            return index
        # Rule 3: "qu" moves together with any consonants before it.
        if word.startswith("qu", index):
            return index + 2
        # Rule 4: "y" after one or more consonants acts as a vowel.
        if letter == "y" and index > 0:
            return index

    # No vowel sound at all: the whole word is a single consonant cluster.
    return len(word)

## Summary

I implemented `translate` in `pig_latin.py`. The function name is unchanged and only the standard `re` module is used.

- **`translate(text)`** finds each word with a regex and translates it. Spaces and punctuation between words stay as they are.
- **`translate_word(word)`** moves the word's starting consonant sound to the end and adds `"ay"`.
- **`_consonant_sound_length(word)`** decides how many starting letters to move, using the four rules:
  1. A word starting with a vowel, `xr` or `yt` moves nothing.
  2. Otherwise, the consonants before the first vowel move.
  3. `qu` moves along with any consonants before it.
  4. A `y` after one or more consonants acts as a vowel. A `y` at the very start is a consonant.

Two behaviours go beyond the instructions:
- **Capital letters:** words with capitals are translated, but the letters keep their original case, so `"Pig"` becomes `"igPay"`, not `"Igpay"`.
- **Punctuation:** it stays in place, so `"pig, please!"` becomes `"igpay, easeplay!"`.
