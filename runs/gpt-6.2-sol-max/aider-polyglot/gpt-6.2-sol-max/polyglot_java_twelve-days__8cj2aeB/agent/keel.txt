class TwelveDays {
    private static final String[] ORDINALS = {
        "first", "second", "third", "fourth", "fifth", "sixth",
        "seventh", "eighth", "ninth", "tenth", "eleventh", "twelfth"
    };

    private static final String[] GIFTS = {
        "a Partridge in a Pear Tree",
        "two Turtle Doves",
        "three French Hens",
        "four Calling Birds",
        "five Gold Rings",
        "six Geese-a-Laying",
        "seven Swans-a-Swimming",
        "eight Maids-a-Milking",
        "nine Ladies Dancing",
        "ten Lords-a-Leaping",
        "eleven Pipers Piping",
        "twelve Drummers Drumming"
    };

    String verse(int verseNumber) {
        int dayIndex = verseNumber - 1;
        StringBuilder lyrics = new StringBuilder("On the ");
        lyrics.append(ORDINALS[dayIndex])
                .append(" day of Christmas my true love gave to me: ");

        for (int giftIndex = dayIndex; giftIndex >= 0; giftIndex--) {
            if (giftIndex < dayIndex) {
                lyrics.append(giftIndex == 0 ? ", and " : ", ");
            }
            lyrics.append(GIFTS[giftIndex]);
        }

        return lyrics.append(".\n").toString();
    }

    String verses(int startVerse, int endVerse) {
        StringBuilder lyrics = new StringBuilder();
        for (int day = startVerse; day <= endVerse; day++) {
            if (day != startVerse) {
                lyrics.append('\n');
            }
            lyrics.append(verse(day));
        }
        return lyrics.toString();
    }

    String sing() {
        return verses(1, ORDINALS.length);
    }
}
EOF

import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Paths;

class CheckTwelveDays {
    public static void main(String[] args) throws Exception {
        String expected = new String(Files.readAllBytes(Paths.get(args[0])), StandardCharsets.UTF_8);
        String[] verses = expected.substring(0, expected.length() - 1).split("\n\n");
        TwelveDays song = new TwelveDays();

        check("full song", expected, song.sing());
        for (int start = 0; start < verses.length; start++) {
            check("verse " + (start + 1), verses[start] + "\n", song.verse(start + 1));
            String range = verses[start] + "\n";
            for (int end = start; end < verses.length; end++) {
                if (end > start) {
                    range += "\n" + verses[end] + "\n";
                }
                check("range " + (start + 1) + "-" + (end + 1), range, song.verses(start + 1, end + 1));
            }
        }
        System.out.println("Full song, all 12 verses, and all 78 verse ranges match.");
    }

    private static void check(String label, String expected, String actual) {
        if (!expected.equals(actual)) {
            throw new AssertionError("Mismatch: " + label);
        }
    }
}
