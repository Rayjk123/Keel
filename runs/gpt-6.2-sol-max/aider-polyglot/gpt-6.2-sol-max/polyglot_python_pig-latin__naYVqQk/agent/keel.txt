def translate(text):
    return " ".join(_translate_word(word) for word in text.split())


def _translate_word(word):
    lower_word = word.lower()
    if lower_word.startswith(("xr", "yt")):
        return word + "ay"

    split = 0
    while split < len(lower_word):
        letter = lower_word[split]
        if letter in "aeiou" or (letter == "y" and split > 0):
            break
        if lower_word.startswith("qu", split):
            split += 2
            break
        split += 1

    return word[split:] + word[:split] + "ay"
