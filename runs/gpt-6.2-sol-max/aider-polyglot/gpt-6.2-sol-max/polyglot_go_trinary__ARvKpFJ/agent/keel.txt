package trinary

import "math"

// ParseTrinary converts a base-3 string to an int64.
// It returns zero and false if the input is invalid or the value overflows.
func ParseTrinary(arg string) (int64, bool) {
	if len(arg) == 0 {
		return 0, false
	}

	var value int64
	for i := 0; i < len(arg); i++ {
		digit := int64(arg[i]) - '0'
		if digit < 0 || digit > 2 || value > (math.MaxInt64-digit)/3 {
			return 0, false
		}
		value = value*3 + digit
	}
	return value, true
}
