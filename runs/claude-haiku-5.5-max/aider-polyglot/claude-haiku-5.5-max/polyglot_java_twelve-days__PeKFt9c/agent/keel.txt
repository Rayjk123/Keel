class TwelveDays {

    private static final String[] DAY_NAMES = {
        "first", "second", "third", "fourth", "fifth", "sixth",
        "seventh", "eighth", "ninth", "tenth", "eleventh", "twelfth"
    };

    private static final String[] GIFTS = {
        "a Partridge in a Pear Tree",
        "two Turtle Doves",
        "three French Hens",
        "four Calling Birds",
        "five Gold Rings",
        "six Geese-a-Laying",
        "seven Swans-a-Swimming",
        "eight Maids-a-Milking",
        "nine Ladies Dancing",
        "ten Lords-a-Leaping",
        "eleven Pipers Piping",
        "twelve Drummers Drumming"
    };

    String verse(int verseNumber) {
        if (verseNumber < 1 || verseNumber > DAY_NAMES.length) {
            throw new IllegalArgumentException(
                "Verse number must be between 1 and " + DAY_NAMES.length + ", but was " + verseNumber);
        }

        StringBuilder sb = new StringBuilder();
        sb.append("On the ")
          .append(DAY_NAMES[verseNumber - 1])
          .append(" day of Christmas my true love gave to me: ");

        // Gifts are listed from the current day down to the first day.
        for (int day = verseNumber; day >= 1; day--) {
            if (day < verseNumber) {
                sb.append(day == 1 ? ", and " : ", ");
            }
            sb.append(GIFTS[day - 1]);
        }

        sb.append(".\n");
        return sb.toString();
    }

    String verses(int startVerse, int endVerse) {
        StringBuilder sb = new StringBuilder();
        for (int verseNumber = startVerse; verseNumber <= endVerse; verseNumber++) {
            if (verseNumber > startVerse) {
                sb.append('\n');
            }
            sb.append(verse(verseNumber));
        }
        return sb.toString();
    }

    String sing() {
        return verses(1, 12);
    }
}
EOF
