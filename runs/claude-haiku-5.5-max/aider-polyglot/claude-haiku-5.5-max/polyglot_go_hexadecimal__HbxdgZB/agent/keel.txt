// Package hexadecimal converts hexadecimal strings into decimal integers.
//
// The conversion is implemented from first principles: each character is
// checked and mapped to its numeric value, and the running total is built up
// by multiplying by 16 (the base) and adding the next digit. No strconv,
// math/big, or other conversion helpers are used.
package hexadecimal

import "errors"

// maxInt64 is the largest value an int64 can hold. It is spelled out as a
// constant so that the overflow check needs no imports.
const maxInt64 int64 = 1<<63 - 1

// ErrInvalidHex is returned when the input is empty or contains a character
// that is not a hexadecimal digit (0-9, a-f, A-F).
var ErrInvalidHex = errors.New("invalid hexadecimal string")

// ErrOutOfRange is returned when the input is valid hexadecimal but its value
// does not fit in an int64.
var ErrOutOfRange = errors.New("hexadecimal value out of range for int64")

// ParseHex converts a hexadecimal string such as "10af8c" or "008000" into its
// decimal (int64) equivalent.
//
// Upper- and lower-case digits are both accepted. Empty input, any character
// outside the hexadecimal alphabet (including whitespace, signs, and prefixes
// such as "0x" or "#"), and values too large for an int64 are reported as
// errors, in which case the returned integer is 0.
func ParseHex(in string) (int64, error) {
	if len(in) == 0 {
		return 0, ErrInvalidHex
	}

	// Validate the entire string first so that malformed input is always
	// reported as invalid, rather than as out of range.
	for i := 0; i < len(in); i++ {
		if _, ok := hexDigitValue(in[i]); !ok {
			return 0, ErrInvalidHex
		}
	}

	var result int64
	for i := 0; i < len(in); i++ {
		digit, _ := hexDigitValue(in[i])

		// result*16 + digit must not exceed maxInt64. Comparing against the
		// bound divided by 16 performs this check without overflowing.
		if result > (maxInt64-digit)/16 {
			return 0, ErrOutOfRange
		}
		result = result*16 + digit
	}

	return result, nil
}

// hexDigitValue returns the value (0-15) of a single hexadecimal digit and
// reports whether the byte is a hexadecimal digit at all.
func hexDigitValue(c byte) (int64, bool) {
	switch {
	case '0' <= c && c <= '9':
		return int64(c - '0'), true
	case 'a' <= c && c <= 'f':
		return int64(c-'a') + 10, true
	case 'A' <= c && c <= 'F':
		return int64(c-'A') + 10, true
	default:
		return 0, false
	}
}
GOEOF
