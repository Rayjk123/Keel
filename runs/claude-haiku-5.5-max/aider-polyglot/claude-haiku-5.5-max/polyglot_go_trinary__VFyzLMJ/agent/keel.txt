package trinary

// ParseTrinary converts a trinary (base-3) number, given as a string of the
// digits '0', '1' and '2', into its decimal (base-10) value.
//
// It works from first principles: the digits are read from left to right and
// each digit d updates a running total as total = total*3 + d. Multiplying the
// total by 3 moves everything read so far one place to the left, so the final
// total is the sum of d * 3^place over all digits.
//
// Any string containing a character other than '0', '1' or '2' is an invalid
// trinary number and yields 0. The empty string also yields 0.
//
// Values too large for an int64 are not detected; they overflow silently.
func ParseTrinary(arg string) int64 {
	var total int64
	for _, c := range arg {
		if c < '0' || c > '2' {
			return 0
		}
		total = total*3 + int64(c-'0')
	}
	return total
}
EOF
