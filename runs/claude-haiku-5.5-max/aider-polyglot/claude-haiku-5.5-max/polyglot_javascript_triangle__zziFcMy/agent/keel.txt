//
// Triangle: decide whether a triangle is equilateral, isosceles or scalene.
//
// A shape is only a triangle when every side is longer than zero and the sum of
// any two sides is at least as long as the third side. For sides a <= b <= c
// that means a > 0 and a + b >= c.
//

// Returns true when `sides` (sorted ascending) describe a real triangle.
const isValidTriangle = (sides) => {
  if (sides.length !== 3) {
    return false;
  }

  const [a, b, c] = sides;

  // Since a <= b <= c and a > 0, the inequalities a + c >= b and b + c >= a
  // always hold, so only a + b >= c needs checking. A NaN side makes these
  // comparisons false, so it is rejected as well.
  return a > 0 && a + b >= c;
};

export class Triangle {
  constructor(...sides) {
    // Store the sides in ascending order so that sides[0] <= sides[1] <= sides[2].
    this.sides = sides.map(Number).sort((x, y) => x - y);
  }

  get isEquilateral() {
    const [a, b, c] = this.sides;
    return isValidTriangle(this.sides) && a === b && b === c;
  }

  get isIsosceles() {
    // With sorted sides, any pair of equal sides must be (a, b) or (b, c).
    const [a, b, c] = this.sides;
    return isValidTriangle(this.sides) && (a === b || b === c);
  }

  get isScalene() {
    // With sorted sides, all three lengths differ exactly when a !== b and b !== c.
    const [a, b, c] = this.sides;
    return isValidTriangle(this.sides) && a !== b && b !== c;
  }
}
EOF

import assert from 'assert';
import { Triangle } from './triangle.mjs';

const kind = (sides) => {
  const t = new Triangle(...sides);
  return { equilateral: t.isEquilateral, isosceles: t.isIsosceles, scalene: t.isScalene };
};

const equilateral = { equilateral: true, isosceles: true, scalene: false };
const isosceles = { equilateral: false, isosceles: true, scalene: false };
const scalene = { equilateral: false, isosceles: false, scalene: true };
const notATriangle = { equilateral: false, isosceles: false, scalene: false };

const permutations = (xs) =>
  xs.length <= 1
    ? [xs]
    : xs.flatMap((x, i) => permutations([...xs.slice(0, i), ...xs.slice(i + 1)]).map((p) => [x, ...p]));

const cases = [
  [[2, 2, 2], equilateral],
  [[3, 3, 3], equilateral],
  [[0.5, 0.5, 0.5], equilateral],
  [[2, 3, 2], isosceles],
  [[3, 4, 4], isosceles],
  [[4, 3, 4], isosceles],
  [[2, 4, 2], isosceles], // degenerate (2 + 2 == 4) is still accepted by the >= rule
  [[5, 4, 6], scalene],
  [[0.5, 0.4, 0.6], scalene],
  [[0, 0, 0], notATriangle],
  [[3, 4, -5], notATriangle],
  [[1, 1, 3], notATriangle],
  [[7, 3, 2], notATriangle],
  [[1, 2], notATriangle],
];

let failures = 0;
for (const [sides, expected] of cases) {
  for (const order of permutations(sides)) {
    try {
      assert.deepStrictEqual(kind(order), expected);
    } catch (err) {
      failures++;
      console.log('FAIL', order, kind(order), 'expected', expected);
    }
  }
}
console.log(failures === 0 ? 'all checks passed' : `${failures} failure(s)`);
