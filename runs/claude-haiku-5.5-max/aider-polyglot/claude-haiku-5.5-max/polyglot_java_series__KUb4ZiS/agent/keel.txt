import java.util.ArrayList;
import java.util.List;
import java.util.Objects;

class Series {
    private final String digits;

    Series(String string) {
        this.digits = Objects.requireNonNull(string, "series cannot be null");
    }

    List<String> slices(int num) {
        if (digits.isEmpty()) {
            throw new IllegalArgumentException("series cannot be empty");
        }
        if (num < 0) {
            throw new IllegalArgumentException("slice length cannot be negative");
        }
        if (num == 0) {
            throw new IllegalArgumentException("slice length cannot be zero");
        }
        if (num > digits.length()) {
            throw new IllegalArgumentException("slice length cannot be greater than series length");
        }

        List<String> result = new ArrayList<>(digits.length() - num + 1);
        for (int start = 0; start + num <= digits.length(); start++) {
            result.add(digits.substring(start, start + num));
        }
        return result;
    }
}

import java.util.Arrays;
import java.util.List;

public class SeriesCheck {
    private static int failures = 0;

    private static void expectSlices(String digits, int num, List<String> expected) {
        try {
            List<String> actual = new Series(digits).slices(num);
            if (actual.equals(expected)) {
                System.out.println("ok    slices(\"" + digits + "\", " + num + ") = " + actual);
            } else {
                failures++;
                System.out.println("FAIL  slices(\"" + digits + "\", " + num + ") = " + actual + ", expected " + expected);
            }
        } catch (RuntimeException e) {
            failures++;
            System.out.println("FAIL  slices(\"" + digits + "\", " + num + ") threw " + e);
        }
    }

    private static void expectError(String digits, int num, String message) {
        try {
            List<String> actual = new Series(digits).slices(num);
            failures++;
            System.out.println("FAIL  slices(\"" + digits + "\", " + num + ") returned " + actual + " instead of throwing");
        } catch (IllegalArgumentException e) {
            if (message.equals(e.getMessage())) {
                System.out.println("ok    slices(\"" + digits + "\", " + num + ") threw: " + message);
            } else {
                failures++;
                System.out.println("FAIL  slices(\"" + digits + "\", " + num + ") wrong message: " + e.getMessage());
            }
        }
    }

    public static void main(String[] args) {
        expectSlices("1", 1, Arrays.asList("1"));
        expectSlices("12", 1, Arrays.asList("1", "2"));
        expectSlices("12", 2, Arrays.asList("12"));
        expectSlices("123", 2, Arrays.asList("12", "23"));
        expectSlices("1234", 3, Arrays.asList("123", "234"));
        expectSlices("12345", 4, Arrays.asList("1234", "2345"));
        expectSlices("12345", 5, Arrays.asList("12345"));
        expectSlices("91494", 2, Arrays.asList("91", "14", "49", "94"));
        expectSlices("0123", 3, Arrays.asList("012", "123"));
        expectSlices("49142", 3, Arrays.asList("491", "914", "142"));
        expectSlices("49142", 4, Arrays.asList("4914", "9142"));
        expectError("12345", 6, "slice length cannot be greater than series length");
        expectError("12345", 42, "slice length cannot be greater than series length");
        expectError("12345", 0, "slice length cannot be zero");
        expectError("123", -1, "slice length cannot be negative");
        expectError("", 1, "series cannot be empty");

        System.out.println(failures == 0 ? "ALL CHECKS PASSED" : failures + " CHECK(S) FAILED");
        System.exit(failures == 0 ? 0 : 1);
    }
}
