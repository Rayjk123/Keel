//
// Sum Of Multiples
//
// Works out the energy points awarded for completing a level. Each magical
// item found on the level has a base value. The points are the sum of every
// distinct positive multiple of those base values that is strictly less than
// the level number.
//

/**
 * Returns the energy points awarded for a completed level.
 *
 * @param {number[]} factors - base values of the magical items collected on the level
 * @param {number} limit - the level number; only multiples strictly below it count
 * @returns {number} sum of all distinct multiples of the base values below `limit`
 */
export const sum = (factors = [], limit = 0) => {
  // A base value needs to be positive to have meaningful multiples, so zero and
  // negative values are ignored.
  const baseValues = factors.filter((factor) => factor > 0);

  let total = 0;

  // Check each candidate number once. A number that is a multiple of several base
  // values (e.g. 15 for 3 and 5) is still added only once, so duplicates are
  // removed automatically. Only a single running total is kept in memory.
  for (let candidate = 1; candidate < limit; candidate += 1) {
    if (baseValues.some((factor) => candidate % factor === 0)) {
      total += candidate;
    }
  }

  return total;
};
