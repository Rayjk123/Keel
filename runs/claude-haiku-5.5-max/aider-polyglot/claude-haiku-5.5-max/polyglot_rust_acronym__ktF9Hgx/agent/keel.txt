/// Returns the acronym of `phrase`, e.g. "Portable Network Graphics" -> "PNG".
///
/// Hyphens and whitespace separate words. All other punctuation is dropped,
/// so each word contributes its first letter (e.g. "It's" -> "I").
pub fn abbreviate(phrase: &str) -> String {
    phrase
        .split(|c: char| c == '-' || c.is_whitespace())
        .filter_map(|word| word.chars().find(|c| c.is_alphabetic()))
        .flat_map(char::to_uppercase)
        .collect()
}
EOF
