import java.util.StringJoiner;

class TwelveDays {
    private static final String[] DAYS = {
        "first", "second", "third", "fourth", "fifth", "sixth",
        "seventh", "eighth", "ninth", "tenth", "eleventh", "twelfth"
    };

    private static final String[] GIFTS = {
        "a Partridge in a Pear Tree",
        "two Turtle Doves",
        "three French Hens",
        "four Calling Birds",
        "five Gold Rings",
        "six Geese-a-Laying",
        "seven Swans-a-Swimming",
        "eight Maids-a-Milking",
        "nine Ladies Dancing",
        "ten Lords-a-Leaping",
        "eleven Pipers Piping",
        "twelve Drummers Drumming"
    };

    String verse(int verseNumber) {
        StringBuilder lyrics = new StringBuilder("On the ")
                .append(DAYS[verseNumber - 1])
                .append(" day of Christmas my true love gave to me: ");

        for (int gift = verseNumber - 1; gift > 0; gift--) {
            lyrics.append(GIFTS[gift]).append(", ");
        }
        if (verseNumber > 1) {
            lyrics.append("and ");
        }
        return lyrics.append(GIFTS[0]).append('.').toString();
    }

    String verses(int startVerse, int endVerse) {
        StringJoiner lyrics = new StringJoiner("\n\n");
        for (int day = startVerse; day <= endVerse; day++) {
            lyrics.add(verse(day));
        }
        return lyrics.toString();
    }

    String sing() {
        return verses(1, DAYS.length);
    }
}
