import re


class PhoneNumber:
    def __init__(self, number):
        compact = re.sub(r"[\s().-]", "", number)
        international = compact.startswith("+")
        digits = compact[1:] if international else compact

        if any(char.isalpha() for char in digits):
            raise ValueError("letters not permitted")
        if re.search(r"[^0-9]", digits):
            raise ValueError("punctuations not permitted")

        if len(digits) < 10:
            raise ValueError("must not be fewer than 10 digits")
        if len(digits) > 11:
            raise ValueError("must not be greater than 11 digits")
        if len(digits) == 11:
            if digits[0] != "1":
                raise ValueError("11 digits must start with 1")
            digits = digits[1:]
        elif international:
            raise ValueError("expected +1 followed by 10 digits")

        for code, first_digit in (("area", digits[0]), ("exchange", digits[3])):
            if first_digit == "0":
                raise ValueError(f"{code} code cannot start with zero")
            if first_digit == "1":
                raise ValueError(f"{code} code cannot start with one")

        self.number = digits

    @property
    def area_code(self):
        return self.number[:3]

    def pretty(self):
        return f"({self.area_code})-{self.number[3:6]}-{self.number[6:]}"

from phone_number import PhoneNumber

valid = [
    "+1 (613)-995-0253", "613-995-0253", "1 613 995 0253",
    "613.995.0253", "6139950253",
]
for raw in valid:
    phone = PhoneNumber(raw)
    assert phone.number == "6139950253", raw
    assert phone.area_code == "613", raw
    assert phone.pretty() == "(613)-995-0253", raw
assert PhoneNumber("2002000000").number == "2002000000"

invalid = {
    "123456789": "must not be fewer than 10 digits",
    "123456789012": "must not be greater than 11 digits",
    "26139950253": "11 digits must start with 1",
    "0139950253": "area code cannot start with zero",
    "+1 113-995-0253": "area code cannot start with one",
    "+1 613-095-0253": "exchange code cannot start with zero",
    "6131950253": "exchange code cannot start with one",
    "613-@!:-0253": "punctuations not permitted",
    "613-ABC-0253": "letters not permitted",
    "+44 6139950253": "must not be greater than 11 digits",
}

    except ValueError as exc:
        assert str(exc) == expected, (raw, str(exc))
    else:
        raise AssertionError(f"Unexpectedly accepted {raw!r}")
print("All normalization and validation checks passed.")
