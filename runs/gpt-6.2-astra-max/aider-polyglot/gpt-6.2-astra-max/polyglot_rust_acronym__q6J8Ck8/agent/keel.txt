/// Returns the uppercase initials of words separated by whitespace or hyphens.
pub fn abbreviate(phrase: &str) -> String {
    phrase
        .split(|c: char| c.is_whitespace() || c == '-')
        .filter_map(|word| word.chars().find(|c| c.is_alphanumeric()))
        .flat_map(char::to_uppercase)
        .collect()
}

#[cfg(test)]
mod tests {
    use super::abbreviate;

    #[test]
    fn whitespace_and_hyphens_separate_words() {
        assert_eq!(abbreviate("As Soon As Possible"), "ASAP");
        assert_eq!(abbreviate("Liquid-crystal\tdisplay"), "LCD");
    }

    #[test]
    fn other_punctuation_does_not_separate_words() {
        assert_eq!(abbreviate("Thank George It's Friday!"), "TGIF");
        assert_eq!(abbreviate("_rock'n'roll_ ... (music)"), "RM");
    }

    #[test]
    fn phrases_without_words_have_no_initials() {
        assert_eq!(abbreviate(""), "");
        assert_eq!(abbreviate(" - ! -- "), "");
    }
}
EOF
