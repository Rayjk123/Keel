import java.util.ArrayList;
import java.util.List;

class Series {
    private final String digits;

    Series(String string) {
        this.digits = string;
    }

    List<String> slices(int num) {
        if (num <= 0) {
            throw new IllegalArgumentException("Slice size is too small.");
        }
        if (num > digits.length()) {
            throw new IllegalArgumentException("Slice size is too big.");
        }

        int lastStart = digits.length() - num;
        List<String> slices = new ArrayList<>(lastStart + 1);
        for (int start = 0; start <= lastStart; start++) {
            slices.add(digits.substring(start, start + num));
        }
        return slices;
    }
}
EOF

import java.util.Arrays;

class SeriesCheck {

        if (!series.slices(size).equals(Arrays.asList(expected))) {
            throw new AssertionError("Unexpected result for slice size " + size);
        }
    }

    static void checkInvalid(Series series, int size) {
        try {

            throw new AssertionError("Expected rejection for slice size " + size);
        } catch (IllegalArgumentException expected) {
            // Invalid size was rejected.
        }
    }

        Series example = new Series("49142");
        check(example, 3, "491", "914", "142");
        check(example, 4, "4914", "9142");
        check(example, 5, "49142");
        check(new Series("0010"), 2, "00", "01", "10");
        check(new Series("777"), 1, "7", "7", "7");
        checkInvalid(example, 0);
        checkInvalid(example, -1);

        checkInvalid(new Series(""), 1);

    }
}
EOF
