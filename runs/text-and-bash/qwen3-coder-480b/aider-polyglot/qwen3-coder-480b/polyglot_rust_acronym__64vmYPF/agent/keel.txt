pub fn abbreviate(phrase: &str) -> String {
    // Replace hyphens with spaces as they're considered word separators
    let phrase = phrase.replace('-', " ");
    
    // Remove all punctuation except spaces
    let cleaned_phrase: String = phrase
        .chars()
        .filter(|c| c.is_alphanumeric() || c.is_whitespace())
        .collect();
    
    // Split by whitespace and take the first letter of each word
    cleaned_phrase
        .split_whitespace()
        .map(|word| {
            // Take the first character and make it uppercase
            word.chars().next().unwrap_or_default().to_uppercase().to_string()
        })
        .collect()
}
