export const sum = (limit, factors) => {
  // Handle edge case where limit is 0 or negative
  if (limit <= 0) return 0;

  const multiples = new Set();

  // For each factor, find all multiples less than limit
  for (const factor of factors) {
    // Skip factor 0 as it would lead to division by zero
    if (factor === 0) continue;
    
    // Add all multiples of this factor less than limit
    for (let multiple = factor; multiple < limit; multiple += factor) {
      multiples.add(multiple);
    }
  }

  // Sum all unique multiples
  let total = 0;
  for (const multiple of multiples) {
    total += multiple;
  }

  return total;
};
EOF
