export class Triangle {
  constructor(...sides) {
    this.sides = sides;
  }

  get isEquilateral() {
    const [a, b, c] = this.sides;
    
    // Check if all sides are positive
    if (a <= 0 || b <= 0 || c <= 0) {
      return false;
    }
    
    // Check triangle inequality
    if (!this._isValidTriangle()) {
      return false;
    }
    
    // All sides must be equal
    return a === b && b === c;
  }

  get isIsosceles() {
    const [a, b, c] = this.sides;
    
    // Check if all sides are positive
    if (a <= 0 || b <= 0 || c <= 0) {
      return false;
    }
    
    // Check triangle inequality
    if (!this._isValidTriangle()) {
      return false;
    }
    
    // At least two sides must be equal
    return a === b || b === c || a === c;
  }

  get isScalene() {
    const [a, b, c] = this.sides;
    
    // Check if all sides are positive
    if (a <= 0 || b <= 0 || c <= 0) {
      return false;
    }
    
    // Check triangle inequality
    if (!this._isValidTriangle()) {
      return false;
    }
    
    // All sides must be different
    return a !== b && b !== c && a !== c;
  }
  
  _isValidTriangle() {
    const [a, b, c] = this.sides;
    return (a + b >= c) && (b + c >= a) && (a + c >= b);
  }
}
