    translated_words = []

    for word in words:
        # Rule 1: begins with vowel or "xr" or "yt"

            translated_words.append(word + 'ay')

            # Check for consonant clusters followed by 'qu' (Rule 3)
            i = 0
            while i < len(word) and word[i] not in vowels:
                # Special case for 'qu'
                if i + 1 < len(word) and word[i:i+2] == 'qu':
                    # Move everything up to and including 'qu' to the end
                    translated_words.append(word[i+2:] + word[:i+2] + 'ay')

                elif i + 1 < len(word) and word[i+1] == 'y' and i > 0:
