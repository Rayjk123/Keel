class PhoneNumber:
    def __init__(self, number):
        # Remove all punctuation and spaces
        cleaned = ""
        for char in number:
            if char.isdigit():
                cleaned += char
            elif char in " .()-+":
                continue  # Skip valid punctuation
            else:
                # If we find letters or other invalid characters
                if char.isalpha():
                    raise ValueError("letters not permitted")
                else:
                    raise ValueError("punctuations not permitted")
        
        # Check length
        if len(cleaned) < 10:
            raise ValueError("must not be fewer than 10 digits")
        elif len(cleaned) > 11:
            raise ValueError("must not be greater than 11 digits")
        elif len(cleaned) == 11:
            # Must start with 1
            if cleaned[0] != '1':
                raise ValueError("11 digits must start with 1")
            cleaned = cleaned[1:]  # Remove country code
        
        # Validate area code (first digit after +1 removal)
        if cleaned[0] == '0':
            raise ValueError("area code cannot start with zero")
        if cleaned[0] == '1':
            raise ValueError("area code cannot start with one")
            
        # Validate exchange code (fourth digit)
        if cleaned[3] == '0':
            raise ValueError("exchange code cannot start with zero")
        if cleaned[3] == '1':
            raise ValueError("exchange code cannot start with one")
        
        self.number = cleaned
    
    def pretty(self):
        return f"({self.number[0:3]})-{self.number[3:6]}-{self.number[6:10]}"
    
    def __str__(self):
        return self.number
EOF
