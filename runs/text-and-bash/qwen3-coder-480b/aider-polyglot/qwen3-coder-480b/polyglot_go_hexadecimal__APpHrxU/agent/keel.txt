package hexadecimal

import (
	"errors"
)

func ParseHex(s string) (int64, error) {
	if s == "" {
		return 0, errors.New("invalid hexadecimal string")
	}

	var result int64 = 0
	
	for _, char := range s {
		// Convert character to its numeric value
		var digitValue int64
		
		switch {
		case char >= '0' && char <= '9':
			digitValue = int64(char - '0')
		case char >= 'a' && char <= 'f':
			digitValue = int64(char - 'a' + 10)
		case char >= 'A' && char <= 'F':
			digitValue = int64(char - 'A' + 10)
		default:
			return 0, errors.New("invalid hexadecimal string")
		}
		
		// Multiply result by 16 and add the new digit
		result = result*16 + digitValue
	}
	
	return result, nil
}

			if err != nil {
				t.Errorf("ParseHex(%q) unexpected error: %v", test.input, err)
			} else if result != test.expected {
				t.Errorf("ParseHex(%q) = %d, want %d", test.input, result, test.expected)
			}
		}
	}
}
