class TwelveDays {
    private static final String[] DAYS = {
        "first", "second", "third", "fourth", "fifth", "sixth",
        "seventh", "eighth", "ninth", "tenth", "eleventh", "twelfth"
    };
    
    private static final String[] GIFTS = {
        "a Partridge in a Pear Tree",
        "two Turtle Doves",
        "three French Hens",
        "four Calling Birds",
        "five Gold Rings",
        "six Geese-a-Laying",
        "seven Swans-a-Swimming",
        "eight Maids-a-Milking",
        "nine Ladies Dancing",
        "ten Lords-a-Leaping",
        "eleven Pipers Piping",
        "twelve Drummers Drumming"
    };
    
    String verse(int verseNumber) {
        if (verseNumber < 1 || verseNumber > 12) {
            throw new IllegalArgumentException("Verse number must be between 1 and 12");
        }
        
        StringBuilder sb = new StringBuilder();
        sb.append("On the ").append(DAYS[verseNumber - 1]).append(" day of Christmas my true love gave to me: ");
        
        for (int i = verseNumber - 1; i >= 0; i--) {
            if (i < verseNumber - 1 && i == 0) {
                sb.append("and ");
            }
            sb.append(GIFTS[i]);
            if (i > 0) {
                sb.append(", ");
            }
        }
        
        sb.append(".\n");
        return sb.toString();
    }
    
    String verses(int startVerse, int endVerse) {
        if (startVerse < 1 || endVerse > 12 || startVerse > endVerse) {
            throw new IllegalArgumentException("Invalid verse range");
        }
        
        StringBuilder sb = new StringBuilder();
        for (int i = startVerse; i <= endVerse; i++) {
            sb.append(verse(i));
            if (i < endVerse) {
                sb.append("\n");
            }
        }
        return sb.toString();
    }
    
    String sing() {
        return verses(1, 12);
    }
}
