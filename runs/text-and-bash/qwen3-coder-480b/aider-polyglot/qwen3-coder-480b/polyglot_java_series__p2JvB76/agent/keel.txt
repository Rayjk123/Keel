import java.util.ArrayList;
import java.util.List;

public class Series {
    private final String digits;

    public Series(String digits) {
        this.digits = digits;
    }

    public List<String> slices(int length) {
        // Validate input
        if (length <= 0) {
            throw new IllegalArgumentException("slice length must be positive");
        }
        
        if (length > digits.length()) {
            throw new IllegalArgumentException("slice length cannot exceed series length");
        }
        
        List<String> result = new ArrayList<>();
        
        // Extract all contiguous substrings of the specified length
        for (int i = 0; i <= digits.length() - length; i++) {
            result.add(digits.substring(i, i + length));
        }
        
        return result;
    }
}
