def translate(text):
    words = text.split()
    translated = []
    
    vowels = "aeiou"
    
    for word in words:
        # Rule 1: starts with vowel, or "xr", or "yt"
        if word[0] in vowels or word.startswith("xr") or word.startswith("yt"):
            translated.append(word + "ay")
            continue
        
        # Rule 3: starts with consonants followed by "qu"
        qu_pos = word.find("qu")
        if qu_pos != -1 and all(c not in vowels for c in word[:qu_pos]):
            # Move consonants + "qu" to end, then add "ay"
            translated.append(word[qu_pos+2:] + word[:qu_pos+2] + "ay")
            continue
        
        # Rule 4: starts with one or more consonants followed by "y"
        # Find 'y' that has at least one consonant before it
        y_pos = -1
        for i, c in enumerate(word):
            if c == "y":
                if i > 0 and all(c not in vowels for c in word[:i]):
                    y_pos = i
                    break
        
        if y_pos != -1:
            # Move consonants before 'y' to end, then add "ay"
            translated.append(word[y_pos:] + word[:y_pos] + "ay")
            continue
        
        # Rule 2: starts with one or more consonants
        # Find first vowel
        vowel_pos = -1
        for i, c in enumerate(word):
            if c in vowels:
                vowel_pos = i
                break
        
        if vowel_pos != -1:
            translated.append(word[vowel_pos:] + word[:vowel_pos] + "ay")
        else:
            # Edge case: no vowels found, just append "ay"
            translated.append(word + "ay")
    
    return " ".join(translated)
