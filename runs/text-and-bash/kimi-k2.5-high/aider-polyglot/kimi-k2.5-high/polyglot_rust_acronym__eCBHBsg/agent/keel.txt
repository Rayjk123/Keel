pub fn abbreviate(phrase: &str) -> String {
    phrase
        .split(|c: char| c.is_whitespace() || c == '-')
        .filter_map(|word| {
            word.chars().find(|c| c.is_ascii_alphabetic())
        })
        .collect::<String>()
        .to_ascii_uppercase()
}
