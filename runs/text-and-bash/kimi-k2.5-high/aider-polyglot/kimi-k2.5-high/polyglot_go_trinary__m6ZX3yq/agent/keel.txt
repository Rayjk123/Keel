package trinary

func ParseTrinary(s string) (int64, bool) {
	var result int64
	for _, char := range s {
		// Check if character is valid (0, 1, or 2)
		if char < '0' || char > '2' {
			return 0, false
		}
		// Multiply current result by 3 and add the digit value
		result = result*3 + int64(char-'0')
	}
	return result, true
}
