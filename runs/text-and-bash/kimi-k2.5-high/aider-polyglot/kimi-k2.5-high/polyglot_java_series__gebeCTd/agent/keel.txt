import java.util.List;
import java.util.ArrayList;

public class Series {
    private final String series;
    
    public Series(String series) {
        this.series = series;
    }
    
    public List<String> slices(int n) {
        List<String> result = new ArrayList<>();
        
        if (n > series.length()) {
            throw new IllegalArgumentException("Slice size is too large");
        }
        
        for (int i = 0; i <= series.length() - n; i++) {
            result.add(series.substring(i, i + n));
        }
        
        return result;
    }
}
