class PhoneNumber:
    def __init__(self, number):
        # Check for letters first
        if any(char.isalpha() for char in number):
            raise ValueError("letters not permitted")
        
        # Check for invalid punctuations (anything that's not a digit, letter, space, or common phone punctuation)
        # Valid punctuations: space, -, ., (, ), +
        valid_punctuation = set(' -.()+')
        for char in number:
            if not char.isdigit() and not char.isalpha() and char not in valid_punctuation:
                raise ValueError("punctuations not permitted")
        
        # Extract digits only
        digits = ''.join(char for char in number if char.isdigit())
        
        # Validate number of digits
        if len(digits) < 10:
            raise ValueError("must not be fewer than 10 digits")
        
        if len(digits) > 11:
            raise ValueError("must not be greater than 11 digits")
        
        # If 11 digits, must start with 1
        if len(digits) == 11:
            if digits[0] != '1':
                raise ValueError("11 digits must start with 1")
            # Remove the leading 1
            digits = digits[1:]
        
        # Now we have exactly 10 digits
        # Validate area code (first 3 digits)
        area_code = digits[0:3]
        if area_code[0] == '0':
            raise ValueError("area code cannot start with zero")
        if area_code[0] == '1':
            raise ValueError("area code cannot start with one")
        
        # Validate exchange code (digits 4-6, index 3-5)
        exchange_code = digits[3:6]
        if exchange_code[0] == '0':
            raise ValueError("exchange code cannot start with zero")
        if exchange_code[0] == '1':
            raise ValueError("exchange code cannot start with one")
        
        # Store the cleaned number
        self.number = digits
        self.area_code = area_code
        self.exchange_code = exchange_code
        self.subscriber_number = digits[6:10]
    
    def pretty(self):
        """Return the phone number in a pretty format: (XXX)-XXX-XXXX"""
        return f"({self.area_code})-{self.exchange_code}-{self.subscriber_number}"
EOF
