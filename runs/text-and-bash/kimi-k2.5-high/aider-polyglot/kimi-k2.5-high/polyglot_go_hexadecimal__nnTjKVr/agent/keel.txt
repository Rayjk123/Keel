package hexadecimal

import (
	"fmt"
)

// ParseHex converts a hexadecimal string to its decimal equivalent.
// Returns the parsed value and an error if the input is invalid.
func ParseHex(in string) (int64, error) {
	if len(in) == 0 {
		return 0, fmt.Errorf("empty input")
	}

	// Check for negative sign
	isNegative := false
	if in[0] == '-' {
		isNegative = true
		in = in[1:]
		if len(in) == 0 {
			return 0, fmt.Errorf("no digits after minus sign")
		}
	}

	var result uint64 = 0
	const maxUint64 = ^uint64(0)

	for i := 0; i < len(in); i++ {
		c := in[i]
		var digit uint64

		switch {
		case c >= '0' && c <= '9':
			digit = uint64(c - '0')
		case c >= 'a' && c <= 'f':
			digit = uint64(c - 'a' + 10)
		case c >= 'A' && c <= 'F':
			digit = uint64(c - 'A' + 10)
		default:
			return 0, fmt.Errorf("invalid hex digit: %c", c)
		}

		// Check for overflow before multiplication
		if result > maxUint64/16 {
			return 0, fmt.Errorf("overflow")
		}
		result = result * 16

		// Check for overflow before addition
		if result > maxUint64-digit {
			return 0, fmt.Errorf("overflow")
		}
		result = result + digit
	}

	// Convert to int64
	var finalResult int64
	if isNegative {
		// For negative numbers, we need to check if result fits in int64
		// Max int64 is 2^63 - 1, but for negative we can go to 2^63
		if result > uint64(1)<<63 {
			return 0, fmt.Errorf("overflow")
		}
		// Handle the special case of -2^63 which can't be represented as positive
		if result == uint64(1)<<63 {
			finalResult = -9223372036854775808 // math.MinInt64
		} else {
			finalResult = -int64(result)
		}
	} else {
		// For positive numbers, result must be <= MaxInt64
		if result > uint64(1)<<63-1 {
			return 0, fmt.Errorf("overflow")
		}
		finalResult = int64(result)
	}

	return finalResult, nil
}

		{"F", 15, false},

		{"008080", 32896, false},    // teal
		{"000080", 128, false},      // navy
		{"-1", -1, false},
		{"-FF", -255, false},
		{"", 0, true},      // empty
		{"xyz", 0, true},   // invalid
		{"1g", 0, true},    // invalid
	}

	allPassed := true
	for _, tc := range testCases {
		result, err := hexadecimal.ParseHex(tc.input)
		if tc.hasError {
			if err == nil {
				fmt.Printf("FAIL: '%s' expected error but got %d\n", tc.input, result)
				allPassed = false
			} else {
				fmt.Printf("PASS: '%s' correctly returned error: %v\n", tc.input, err)
			}
		} else {
			if err != nil {
				fmt.Printf("FAIL: '%s' unexpected error: %v\n", tc.input, err)
				allPassed = false
			} else if result != tc.expected {
				fmt.Printf("FAIL: '%s' expected %d but got %d\n", tc.input, tc.expected, result)
				allPassed = false
			} else {
				fmt.Printf("PASS: '%s' = %d\n", tc.input, result)
			}
		}
	}

	if allPassed {
		fmt.Println("\nAll tests passed!")
	} else {
		fmt.Println("\nSome tests failed!")
	}
}
