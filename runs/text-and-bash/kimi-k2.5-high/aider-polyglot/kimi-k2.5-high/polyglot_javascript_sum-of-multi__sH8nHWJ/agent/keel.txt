export const sum = (baseValues, level) => {
  const multiples = new Set();
  
  for (const base of baseValues) {
    if (base === 0) continue; // Skip 0 to avoid infinite loop
    
    for (let multiple = base; multiple < level; multiple += base) {
      multiples.add(multiple);
    }
  }
  
  let total = 0;
  for (const num of multiples) {
    total += num;
  }
  
  return total;
};
EOF
