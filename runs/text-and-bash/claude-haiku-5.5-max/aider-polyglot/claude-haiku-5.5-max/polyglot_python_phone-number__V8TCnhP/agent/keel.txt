DIGITS = "0123456789"
SEPARATORS = "()-."  # removed from the input, together with whitespace


class PhoneNumber:
    def __init__(self, number):
        self.number = self._clean(number)

    @staticmethod
    def _clean(raw_number):
        # Letters are checked first so they are not reported as punctuation.
        if any(char.isalpha() for char in raw_number):
            raise ValueError("letters not permitted")

        text = raw_number.strip()
        # A leading "+" marks an international number, e.g. "+1 (223) 456-7890".
        has_plus = text.startswith("+")
        if has_plus:
            text = text[1:]

        for char in text:
            if char not in DIGITS and char not in SEPARATORS and not char.isspace():
                raise ValueError("punctuations not permitted")

        digits = "".join(char for char in text if char in DIGITS)

        if has_plus and (len(digits) != 11 or digits[0] != "1"):
            raise ValueError("a leading + requires 11 digits starting with 1")
        if len(digits) < 10:
            raise ValueError("must not be fewer than 10 digits")
        if len(digits) > 11:
            raise ValueError("must not be greater than 11 digits")
        if len(digits) == 11:
            if digits[0] != "1":
                raise ValueError("11 digits must start with 1")
            digits = digits[1:]

        area_code, exchange_code = digits[:3], digits[3:6]
        if area_code[0] == "0":
            raise ValueError("area code cannot start with zero")
        if area_code[0] == "1":
            raise ValueError("area code cannot start with one")
        if exchange_code[0] == "0":
            raise ValueError("exchange code cannot start with zero")
        if exchange_code[0] == "1":
            raise ValueError("exchange code cannot start with one")

        return digits
EOF

valid_cases = {

    "223.456.7890": "2234567890",

    "": "must not be fewer than 10 digits",
    "22234567890": "11 digits must start with 1",
    "321234567890": "must not be greater than 11 digits",
    "123-abc-7890": "letters not permitted",
    "123-@:!-7890": "punctuations not permitted",
    "1+2234567890": "punctuations not permitted",

    actual = PhoneNumber(raw).number
    assert actual == expected, f"{raw!r}: got {actual!r}, expected {expected!r}"

        PhoneNumber(raw)
    except ValueError as err:
        assert str(err) == message, f"{raw!r}: got {str(err)!r}, expected {message!r}"

        raise AssertionError(f"{raw!r}: no ValueError raised")

print("all checks passed")
