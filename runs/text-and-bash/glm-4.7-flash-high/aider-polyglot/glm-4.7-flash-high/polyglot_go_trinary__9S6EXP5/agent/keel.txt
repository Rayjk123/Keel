package trinary

func ParseTrinary(arg string, want int64, ok bool) {
	decimalValue := int64(0)
	power := int64(1) // 3^0 = 1 for the rightmost digit

	// Iterate from right to left
	for i := len(arg) - 1; i >= 0; i-- {
		digit := int64(arg[i]) - '0'

		// Check if digit is valid (0, 1, or 2)
		if digit < 0 || digit > 2 {
			decimalValue = 0
			return
		}

		// Add digit_value * 3^position
		decimalValue += digit * power

		// Update power for next position: 3^1, 3^2, etc.
		power *= 3
	}
}
