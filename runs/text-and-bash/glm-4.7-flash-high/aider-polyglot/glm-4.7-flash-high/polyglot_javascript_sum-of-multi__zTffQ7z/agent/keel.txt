export const sum = (...multiples) => {
  const limit = multiples.shift();
  const result = new Set();
  
  for (const multiple of multiples) {
    if (multiple === 0) continue;
    
    for (let i = multiple; i < limit; i += multiple) {
      result.add(i);
    }
  }
  
  return Array.from(result).reduce((acc, curr) => acc + curr, 0);
};
