pub fn abbreviate(phrase: &str) -> String {
    phrase
        .to_uppercase()
        .split('-')  // Split by hyphens (word separators)
        .map(|word| {
            word.chars()
                .skip_while(|c| !c.is_alphabetic())  // Skip leading non-letters
                .take(1)  // Take first letter of each word
                .collect::<String>()
        })
        .filter(|s| !s.is_empty())  // Remove empty results
        .collect()
}

pub fn abbreviate(phrase: &str) -> String {
    let cleaned = phrase.to_uppercase()
        .chars()
        .filter(|c| c.is_alphanumeric() || *c == '-')  // Keep alphanumeric chars and hyphens
        .collect::<String>();
    
    cleaned
        .split('-')  // Split by hyphens (word separators)
        .map(|word| {
