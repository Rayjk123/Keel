export class Triangle {
  constructor(...sides) {
    this.sides = sides;
    const [a, b, c] = sides;
    // Check if it's a valid triangle: all sides > 0 and triangle inequality satisfied
    this.isValid = sides.every(s => s > 0) && 
                   (a + b >= c) && 
                   (b + c >= a) && 
                   (a + c >= b);
  }

  get isEquilateral() {
    return this.isValid && 
           this.sides[0] === this.sides[1] && 
           this.sides[1] === this.sides[2];
  }

  get isIsosceles() {
    const [a, b, c] = this.sides;
    return this.isValid && (a === b || b === c || a === c);
  }

  get isScalene() {
    const [a, b, c] = this.sides;
    return this.isValid && a !== b && b !== c && a !== c;
  }
}
