package hexadecimal

func ParseHex(in string, out int64, errCase string) {
	// Convert single character to its hex digit value, or -1 if invalid
	hexToInt := func(c byte) int {
		switch {
		case c >= '0' && c <= '9':
			return int(c - '0')
		case c >= 'a' && c <= 'f':
			return int(c - 'a' + 10)
		case c >= 'A' && c <= 'F':
			return int(c - 'A' + 10)
		default:
			return -1
		}
	}

	result := int64(0)
	
	// Iterate through each character in the hex string
	for i := 0; i < len(in); i++ {
		digit := hexToInt(in[i])
		
		// Check for invalid hex character
		if digit == -1 {
			// Invalid hexadecimal string - return without changing output
			return
		}
		
		// Shift previous result left by 4 bits and add current digit value
		// (each hex digit represents 4 bits)
		result = (result << 4) | int64(digit)
	}
	
	out = result
}

package hexadecimal

func ParseHex(in string, out int64, errCase string) {
	// Helper function to convert a single hex character to its integer value
	hexToInt := func(c byte) int {
		switch {
		case c >= '0' && c <= '9':
			return int(c - '0')
		case c >= 'a' && c <= 'f':
			return int(c - 'a' + 10)
		case c >= 'A' && c <= 'F':
			return int(c - 'A' + 10)
		default:
			return -1
		}
	}

	result := int64(0)
	
	// Process each character in the hex string from left to right
	for i := 0; i < len(in); i++ {
		digit := hexToInt(in[i])
		
		// If the character is invalid (not 0-9, a-f, A-F), return immediately
		if digit == -1 {
			return
		}
		
		// For each hex digit, multiply the current result by 16 (shift left 4 bits)
		// and add the current digit value
		result = (result * 16) + int64(digit)
	}
	
	// Assign the decimal result to the output parameter
	out = result
}
