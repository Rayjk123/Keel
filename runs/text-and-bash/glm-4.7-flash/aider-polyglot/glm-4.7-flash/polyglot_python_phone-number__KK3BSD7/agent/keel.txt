    def __init__(self, number):
        # Clean the input: extract digits
        digits = self._extract_digits(number)
        
        # Validate digit count
        if len(digits) < 10:
            raise ValueError("must not be fewer than 10 digits")
        if len(digits) > 11:
            raise ValueError("must not be greater than 11 digits")
        
        # Handle 11-digit numbers
        if len(digits) == 11 and digits[0] != '1':
            raise ValueError("11 digits must start with 1")
        
        # Validate area code (first 3 digits)
        area_code = digits[0:3]
        if area_code[0] == '0':
            raise ValueError("area code cannot start with zero")
        if area_code[0] == '1':
            raise ValueError("area code cannot start with one")
        
        # Validate exchange code (4th and 5th digits)
        exchange_code = digits[3:5]
        if exchange_code[0] == '0':
            raise ValueError("exchange code cannot start with zero")
        if exchange_code[0] == '1':
            raise ValueError("exchange code cannot start with one")
        
        # Store the cleaned 10-digit number
        self.number = digits[-10:]
    
    def _extract_digits(self, number):
        # Remove all non-digit characters and check for letters
        digits = []
        for char in str(number):
            if char.isdigit():
                digits.append(char)
            elif char.isalpha():
                raise ValueError("letters not permitted")
            else:
                raise ValueError("punctuations not permitted")
        return ''.join(digits)
