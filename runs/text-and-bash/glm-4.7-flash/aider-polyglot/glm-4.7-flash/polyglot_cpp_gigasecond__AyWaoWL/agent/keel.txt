#include "gigasecond.h"

namespace gigasecond {

boost::posix_time::ptime add(const boost::posix_time::ptime& dt) {
    // Create a gigasecond duration (1,000,000,000 seconds)
    boost::posix_time::time_duration gigasecond(1000000000);
    // Add the gigasecond duration to the given date/time
    return dt + gigasecond;
}

}  // namespace gigasecond
EOF
