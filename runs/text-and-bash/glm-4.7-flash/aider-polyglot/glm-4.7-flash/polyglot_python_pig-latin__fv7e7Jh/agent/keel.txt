def translate(text):
    """
    Translate text from English to Pig Latin.
    
    Rules:
    1. Vowel-starting words (a, e, i, o, u) start with "xr" or "yt" -> add "ay"
    2. Consonant-starting words -> move consonants to end, add "ay"
    3. Words with "qu" -> move consonants+qu to end, add "ay"
    4. Words with consonant followed by "y" -> move consonants to end, add "ay"
    """
    if not text:
        return text

    result = []
    words = text.split()
    
    for word in words:
        # Check for empty strings
        if not word:
            result.append(word)
            continue
            
        # Rule 1: Start with vowel, "xr", or "yt"
        if word[0] in vowels or word.lower().startswith(('xr', 'yt')):
            result.append(word + 'ay')
            continue
        
        # Rule 3: Consonant + "qu"
        i = 1  # Index to check from
        # Move past the first consonant to check for "qu"
        if word[0].lower() not in vowels:
            i += 1
        
        # Check for "qu" in the rest of the word
        if i + 1 < len(word) and word[i].lower() == 'q' and word[i+1].lower() == 'u':
            # Found "qu", move all consonants before and including "qu" to end
            result.append(word[i:] + word[:i] + 'ay')
            continue
        
        # Rule 2: Regular consonant start or Rule 4 with "y" but no "qu"
        # Find where the first vowel is, or the "y" if it comes before any vowel
        vowel_found = False
        y_is_consonant = True
        i = 0
        
        while i < len(word):
            char = word[i].lower()
            if char == 'y':
                # Current rule: if we encounter 'y' before any vowel, it's a consonant
                # But check if there's an earlier vowel
                # Rule 4 only applies if 'y' comes after consonants but before vowels
                # So we need to see if there was a vowel before this 'y'
                if not vowel_found:
                    # First 'y' found - it's acting as a consonant in this rule
                    break
            elif char in vowels:
                vowel_found = True
            i += 1
        
        result.append(word[i:] + word[:i] + 'ay')
    
    return ' '.join(result)

            result.append(word[2:] + word[:2] + 'ay')
            continue

        # For the 'y' rule, we take CONSONANTS only (non-vowels, including 'y' if it's a consonant)

                first_vowel_at = boundary

        else:
            result.append(word[first_vowel_at:] + word[:first_vowel_at] + 'ay')

def translate(text):
    """
    Translate text from English to Pig Latin.
    
    Rules:
    1. Vowel-starting words (a, e, i, o, u) start with "xr" or "yt" -> add "ay"
    2. Consonant-starting words -> move consonants to end, add "ay"
    3. Words with "qu" -> move consonants+qu to end, add "ay"
    4. Words with consonant followed by "y" -> move consonants to end, add "ay"
    """
    if not text:
        return text
    
    vowels = {'a', 'e', 'i', 'o', 'u'}
    result = []
    words = text.split()
    
    for word in words:
        if not word:
            result.append(word)
            continue
        
        word_lower = word.lower()
        
        # Rule 1: Start with vowel, "xr", or "yt"
        if word[0] in vowels or word_lower.startswith(('xr', 'yt')):
            result.append(word + 'ay')
            continue
        
        # Count consonants (including 'qu' patterns where u is consonant)
        consonant_cluster_end = 0
        i = 0
        while i < len(word_lower):
            char = word_lower[i]
            # Special case: skip past 'qu' pairs
            if char == 'q' and i + 1 < len(word_lower) and word_lower[i+1] == 'u':
                consonant_cluster_end = i + 2
                i += 2
                continue
            # Rule 4: 'y' before first vowel stays, but we stop counting before it
            if char == 'y':
                # 'y' is after consonants, let it stay at front
                # The consonant cluster ends just before this 'y'
                break
            if char in vowels:
                # Vowel breaks the consonant cluster
                break
            # Consonant
            consonant_cluster_end = i + 1
            i += 1
        
        suffix = word[consonant_cluster_end:] + word[:consonant_cluster_end] + 'ay'
        result.append(suffix)
    
    return ' '.join(result)
