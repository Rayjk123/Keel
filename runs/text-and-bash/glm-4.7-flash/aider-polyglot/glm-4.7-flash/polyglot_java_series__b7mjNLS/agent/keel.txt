import java.util.ArrayList;
import java.util.List;

public class Series {
    private String data;

    Series(String string) {
        this.data = string;
    }

    List<String> slices(int num) {
        List<String> result = new ArrayList<>();
        
        if (num < 1 || num > data.length()) {
            throw new IllegalArgumentException("Slice number is out of bounds.");
        }
        
        for (int i = 0; i <= data.length() - num; i++) {
            result.add(data.substring(i, i + num));
        }
        
        return result;
    }
}
