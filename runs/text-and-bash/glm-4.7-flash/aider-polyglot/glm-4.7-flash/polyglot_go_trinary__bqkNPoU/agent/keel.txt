func toDecimal(trinary string) int64 {
	if trinary == "" {
		return 0
	}
	
	decimal := int64(0)
	length := len(trinary)
	
	for i, ch := range trinary {
		// Check for valid trinary digit
		if ch < '0' || ch > '2' {
			return 0
		}
		
		// Calculate the place value (3^(position from right))
		// Position from right is (length - 1 - i)
		digit := ch - '0'
		decimal += digit * int64(pow(3, length-1-i))
	}
	
	return decimal
}

func IsTrinary(s string) bool {
	for _, ch := range s {
		if ch < '0' || ch > '2' {
			return false
		}
	}
	
	// A single character that's a digit is valid
	// Special case for empty string is handled in toDecimal()
	return s != ""
}

// Helper function to calculate 3^n
func pow(base int, exp int) int {
	result := 1
	for i := 0; i < exp; i++ {
		result *= base
	}
	return result
}
