pub fn abbreviate(phrase: &str) -> String {
    phrase
        .replace('-', ' ')
        .chars()
        .filter(|c| c.is_alphabetic())
        .filter(|c| {
            let mut chars = c.to_ascii_uppercase().collect::<String>();
            chars.chars().next().unwrap_or_default().is_uppercase()
        })
        .map(|c| c.to_ascii_uppercase())
        .collect()
}
