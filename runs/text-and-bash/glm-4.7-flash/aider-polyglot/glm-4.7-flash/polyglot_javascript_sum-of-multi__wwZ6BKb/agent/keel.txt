export const sum = (limit, numbers) => {
  const multiples = new Set();

  numbers.forEach((number) => {
    if (number === 0) return;
    for (let i = number; i < limit; i += number) {
      multiples.add(i);
    }
  });

  return Array.from(multiples).reduce((sum, num) => sum + num, 0);
};
EOF
