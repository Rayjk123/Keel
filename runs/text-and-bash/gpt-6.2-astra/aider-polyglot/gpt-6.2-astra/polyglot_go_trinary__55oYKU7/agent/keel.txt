package trinary

// ParseTrinary converts arg from base three, returning zero for invalid input.
func ParseTrinary(arg string, want int64, ok bool) int64 {
	var value int64
	for i := 0; i < len(arg); i++ {
		digit := arg[i]
		if digit < '0' || digit > '2' {
			return 0
		}
		value = value*3 + int64(digit-'0')
	}
	return value
}
EOF
