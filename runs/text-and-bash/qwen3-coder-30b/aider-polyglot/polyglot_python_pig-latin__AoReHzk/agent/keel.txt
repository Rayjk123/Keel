def translate(text):
    vowels = set('aeiou')
    words = text.split()
    result = []
    
    for word in words:
        # Rule 1: If word begins with a vowel, or starts with "xr" or "yt"
        if word[0] in vowels or word.startswith('xr') or word.startswith('yt'):
            result.append(word + 'ay')
        else:
            # Find where consonants end and vowels/other letters begin
            i = 0
            while i < len(word) and word[i] not in vowels and word[i] != 'y':
                i += 1
            
            # Special case: if we started with consonants followed by 'y' 
            # and 'y' is not at the beginning, we need to handle it properly
            if i > 0 and word[i] == 'y' and i < len(word) - 1:
                # Move all consonants before 'y' to end, but keep 'y' with following consonants
                consonant_end = i
                while consonant_end < len(word) and word[consonant_end] not in vowels and word[consonant_end] != 'y':
                    consonant_end += 1
                # Move consonants up to the 'y' and everything after it to the end
                result.append(word[consonant_end:] + word[:consonant_end] + 'ay')
            elif i > 0 and word[i] == 'y':
                # If 'y' is at the beginning, treat it like a vowel
                result.append(word[i:] + word[:i] + 'ay')
            elif i > 0 and word[i-1:i+2] == 'qu':
                # Rule 3: Handle "qu" case
                consonant_end = i
                while consonant_end < len(word) and word[consonant_end] not in vowels:
                    consonant_end += 1
                result.append(word[consonant_end:] + word[:consonant_end] + 'ay')
            else:
                # Standard case: move all initial consonants to the end
                result.append(word[i:] + word[:i] + 'ay')
    
    return ' '.join(result)
