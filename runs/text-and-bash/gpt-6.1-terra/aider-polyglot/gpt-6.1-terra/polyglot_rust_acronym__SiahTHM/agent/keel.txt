pub fn abbreviate(phrase: &str) -> String {
    let mut acronym = String::new();
    let mut at_word_start = true;

    for character in phrase.chars() {
        if character.is_whitespace() || character == '-' {
            at_word_start = true;
        } else if character.is_alphanumeric() && at_word_start {
            acronym.extend(character.to_uppercase());
            at_word_start = false;
        }
    }

    acronym
}

#[cfg(test)]
mod tests {
    use super::abbreviate;

    #[test]
    fn abbreviates_words_and_hyphens() {
        assert_eq!(abbreviate("As Soon As Possible"), "ASAP");
        assert_eq!(abbreviate("Liquid-crystal display"), "LCD");
    }

    #[test]
    fn removes_punctuation_without_splitting_words() {
        assert_eq!(abbreviate("Thank George It's Friday!"), "TGIF");
        assert_eq!(abbreviate("  ...hello,world -- _again_ "), "HA");
    }

    #[test]
    fn handles_empty_input_and_unicode() {
        assert_eq!(abbreviate(" -- ! "), "");
        assert_eq!(abbreviate("école ßeta"), "ÉSS");
    }
}
