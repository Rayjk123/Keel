def translate(text):
    vowels = set('aeiou')
    result = []
    
    for word in text.split():
        # Rule 1: starts with vowel, 'xr', or 'yt'
        if word[0] in vowels or word.startswith(('xr', 'yt')):
            result.append(word + 'ay')
            continue
        
        # Find where to split for Rules 2, 3, 4
        for i, char in enumerate(word):
            # Rule 3: 'qu' case - move consonants + 'qu' to end
            if word[i:i+2] == 'qu':
                result.append(word[i+2:] + word[:i+2] + 'ay')
                break
            # Rule 4: 'y' after consonants acts as vowel
            if char == 'y' and i > 0:
                result.append(word[i:] + word[:i] + 'ay')
                break
            # Rule 2: found a vowel - move initial consonants to end
            if char in vowels:
                result.append(word[i:] + word[:i] + 'ay')
                break
        else:
            # Fallback for edge cases
            result.append(word + 'ay')
    
    return ' '.join(result)
