package trinary

// ParseTrinary converts a trinary string to its decimal equivalent.
// Returns 0 for invalid trinary strings.
func ParseTrinary(arg string) int64 {
	// Check for empty string
	if len(arg) == 0 {
		return 0
	}

	var result int64 = 0

	// Iterate through each character
	for _, char := range arg {
		// Validate the character is a valid trinary digit
		if char < '0' || char > '2' {
			return 0
		}

		// Convert character to digit value
		digit := int64(char - '0')

		// Multiply current result by 3 and add the new digit
		result = result*3 + digit
	}

	return result
}
