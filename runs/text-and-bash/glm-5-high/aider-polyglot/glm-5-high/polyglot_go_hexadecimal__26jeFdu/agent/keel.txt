package hexadecimal

import (
	"strings"
)

// ParseHex converts a hexadecimal string to its decimal equivalent.
// Returns the decimal value and an error string if the input is invalid.
func ParseHex(in string) (int64, string) {
	// Handle empty string
	if len(in) == 0 {
		return 0, "invalid hexadecimal string"
	}

	// Convert to lowercase for case-insensitive comparison
	hex := strings.ToLower(in)

	var result int64 = 0

	// Process each character
	for _, c := range hex {
		result <<= 4 // Multiply by 16 (shift left by 4 bits)

		var digit int64
		switch {
		case c >= '0' && c <= '9':
			digit = int64(c - '0')
		case c >= 'a' && c <= 'f':
			digit = int64(c-'a') + 10
		default:
			// Invalid character
			return 0, "invalid hexadecimal string"
		}

		result += digit
	}

	return result, ""
}
EOF

		{"a", 10, false},

			}
		} else {
			if err != "" {
				t.Errorf("ParseHex(%q) unexpected error: %s", test.in, err)
			}
			if result != test.expected {
				t.Errorf("ParseHex(%q) = %d, expected %d", test.in, result, test.expected)
			}
		}
	}
}
