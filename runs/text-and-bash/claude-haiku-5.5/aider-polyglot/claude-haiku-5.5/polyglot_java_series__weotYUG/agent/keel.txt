import java.util.ArrayList;
import java.util.List;

class Series {
    private final String digits;

    Series(String string) {
        this.digits = string;
    }

    List<String> slices(int num) {
        if (num < 0) {
            throw new IllegalArgumentException("Slice length cannot be negative");
        }
        List<String> result = new ArrayList<>();
        if (num == 0) {
            for (int i = 0; i <= digits.length(); i++) {
                result.add("");
            }
            return result;
        }
        if (num > digits.length()) {
            throw new IllegalArgumentException("Slice length is larger than the series");
        }
        for (int i = 0; i + num <= digits.length(); i++) {
            result.add(digits.substring(i, i + num));
        }
        return result;
    }
}
