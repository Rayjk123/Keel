//
// Sum Of Multiples
//
// Given a list of base values (factors) and a level (limit), returns the sum
// of all unique positive multiples of each factor that are strictly less than
// the limit.
//

export const sum = (factors = [3, 5], limit = 1) => {
  const multiples = new Set();

  for (const factor of factors) {
    if (!(factor > 0)) {
      // Zero or negative bases have no meaningful positive multiples; skip them
      // (zero would otherwise loop forever).
      continue;
    }
    for (let multiple = factor; multiple < limit; multiple += factor) {
      multiples.add(multiple);
    }
  }

  let total = 0;
  for (const value of multiples) {
    total += value;
  }
  return total;
};
EOF
