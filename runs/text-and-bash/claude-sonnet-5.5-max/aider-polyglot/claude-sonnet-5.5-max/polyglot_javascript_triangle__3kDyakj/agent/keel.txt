//
// Triangle classification: equilateral, isosceles or scalene.
//

/**
 * A shape is only a triangle when it has exactly three sides, every side is
 * longer than zero, and the sum of any two sides is at least as long as the
 * remaining side (the triangle inequality).
 */
const isTriangle = (sides) => {
  if (sides.length !== 3) return false;
  if (!sides.every((side) => Number.isFinite(side) && side > 0)) return false;

  const [a, b, c] = sides;
  return a + b >= c && b + c >= a && a + c >= b;
};

export class Triangle {
  constructor(...sides) {
    this.sides = sides;
    this.valid = isTriangle(sides);
  }

  get isEquilateral() {
    if (!this.valid) return false;
    const [a, b, c] = this.sides;
    return a === b && b === c;
  }

  get isIsosceles() {
    if (!this.valid) return false;
    const [a, b, c] = this.sides;
    return a === b || b === c || a === c;
  }

  get isScalene() {
    if (!this.valid) return false;
    const [a, b, c] = this.sides;
    return a !== b && b !== c && a !== c;
  }
}
EOF_SRC

I implemented `Triangle` in `triangle.js`. A scratch script of 26 canonical and edge cases passed against it with plain Node. I couldn't run Jest or ESLint because the project has no `node_modules` and no test file.

- **Validity:** the constructor records whether the input is a real triangle. That means exactly three sides, each a finite number greater than 0, and `a + b ≥ c`, `b + c ≥ a`, `a + c ≥ b`. Degenerate triangles like `1, 1, 2` count as valid because the spec uses `≥`.
- **Invalid input:** the constructor never throws. All three getters return `false` for an invalid shape, such as `0, 0, 0`, `1, 1, 3`, a wrong number of sides, or non-numeric values.
- **`isEquilateral`:** all three sides are equal.
- **`isIsosceles`:** at least two sides are equal, so equilateral triangles count too.
- **`isScalene`:** all three sides differ.

I kept the existing class and getter names, used no external packages, and left no stray files, since the scratch copy was in `/tmp` and has been removed.
