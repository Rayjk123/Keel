// Package trinary converts trinary (base 3) numbers to their decimal value.
package trinary

import (
	"errors"
	"fmt"
	"math"
)

// ParseTrinary converts a trinary (base 3) number, given as a string of the
// digits '0', '1' and '2', to its decimal value.
//
// The most significant digit comes first and the last digit is the 1's place,
// the one before it the 3's place, then the 9's place, and so on. For example
// "102012" is 1*3^5 + 0*3^4 + 2*3^3 + 0*3^2 + 1*3^1 + 2*3^0 = 302.
//
// A string that does not specify a valid trinary number (it is empty, contains
// anything other than the digits '0', '1' and '2' - signs, spaces and
// separators included - or is too large to fit in an int64) is treated as the
// value 0 and is reported with a non-nil error.
func ParseTrinary(arg string) (int64, error) {
	if arg == "" {
		return 0, errors.New("trinary: empty string")
	}

	var value int64
	for _, r := range arg {
		if r < '0' || r > '2' {
			return 0, fmt.Errorf("trinary: invalid digit %q in %q", r, arg)
		}
		digit := int64(r - '0')

		// Make sure value*3 + digit still fits in an int64 before computing it.
		if value > (math.MaxInt64-digit)/3 {
			return 0, fmt.Errorf("trinary: %q is too large for an int64", arg)
		}
		value = value*3 + digit
	}
	return value, nil
}
EOF

import (

func TestTmpValid(t *testing.T) {

		if err != nil || got != c.want {
			t.Errorf("ParseTrinary(%q) = %d, %v; want %d, nil", c.in, got, err, c.want)
		}
	}
}

func TestTmpInvalid(t *testing.T) {
	bad := []string{
		"", "3", "4", "1234", "12jkl3", "hello", "-1", "+1", " 1", "1 ", "10\n",
		"1_0", "1.5", "0x1", "1\x002", "\xff", "\uff11", "\u0661", "1\u00e9",

	for _, in := range bad {
		got, err := ParseTrinary(in)
		if err == nil || got != 0 {
			t.Errorf("ParseTrinary(%q) = %d, %v; want 0, error", in, got, err)
		}
	}

func TestTmpLimits(t *testing.T) {
	max := strconv.FormatInt(math.MaxInt64, 3)
	if got, err := ParseTrinary(max); err != nil || got != math.MaxInt64 {
		t.Errorf("MaxInt64: got %d, %v", got, err)
	}
	over := new(big.Int).Add(big.NewInt(math.MaxInt64), big.NewInt(1)).Text(3)
	if got, err := ParseTrinary(over); err == nil || got != 0 {
		t.Errorf("MaxInt64+1: got %d, %v; want 0, error", got, err)
	}
	if got, err := ParseTrinary(max + "0"); err == nil || got != 0 {
		t.Errorf("MaxInt64*3: got %d, %v; want 0, error", got, err)
	}
	p39 := new(big.Int).Exp(big.NewInt(3), big.NewInt(39), nil)
	want := new(big.Int).Sub(p39, big.NewInt(1)).Int64()
	if got, err := ParseTrinary(strings.Repeat("2", 39)); err != nil || got != want {
		t.Errorf("39 twos: got %d, %v; want %d", got, err, want)
	}
	if got, err := ParseTrinary(strings.Repeat("2", 41)); err == nil || got != 0 {
		t.Errorf("41 twos: got %d, %v; want 0, error", got, err)
	}
	if got, err := ParseTrinary(strings.Repeat("0", 1000) + "102012"); err != nil || got != 302 {
		t.Errorf("many leading zeros: got %d, %v; want 302", got, err)
	}
}

func TestTmpDifferential(t *testing.T) {
	check := func(s string) {
		want, perr := strconv.ParseInt(s, 3, 64)
		got, err := ParseTrinary(s)
		if perr != nil {
			if err == nil || got != 0 {
				t.Fatalf("ParseTrinary(%q) = %d, %v; want 0, error", s, got, err)
			}
		} else if err != nil || got != want {
			t.Fatalf("ParseTrinary(%q) = %d, %v; want %d, nil", s, got, err, want)
		}
	}

	rng := rand.New(rand.NewSource(42))
	for i := 0; i < 200000; i++ {
		b := make([]byte, 1+rng.Intn(45))
		for j := range b {
			b[j] = "012"[rng.Intn(3)]
		}
		check(string(b))
	}

	// Walk across the int64 boundary.
	for k := int64(-1000); k <= 1000; k++ {
		v := new(big.Int).Add(big.NewInt(math.MaxInt64), big.NewInt(k))
		check(v.Text(3))
	}

	// Any string containing a non-trinary byte must be rejected.
	for i := 0; i < 100000; i++ {
		b := make([]byte, 1+rng.Intn(12))
		hasBad := false
		for j := range b {
			b[j] = byte(rng.Intn(256))
			if b[j] < '0' || b[j] > '2' {
				hasBad = true
			}
		}
		if !hasBad {
			continue
		}
		if got, err := ParseTrinary(string(b)); err == nil || got != 0 {
			t.Fatalf("ParseTrinary(%q) = %d, %v; want 0, error", b, got, err)
		}
	}
}

`ParseTrinary` in `trinary.go` is implemented. The final temporary test run, `gofmt`, `go vet` and the build were all clean.

- **How it works:** it reads the digits left to right and computes `value = value*3 + digit`, which is the place-value sum from the instructions. It uses no standard-library parsing.
- **Invalid input:** an empty string, any character other than `0`, `1` or `2`, or a value too large for `int64` returns `0` and a non-nil error. That covers signs, whitespace, underscores, non-ASCII digits and bad UTF-8. Overflow is rejected before it can wrap.
- **Signature change:** the stub `ParseTrinary(arg string, want int64, ok bool)` can't work, because it has no return value and takes test-table fields as parameters. I kept the name and used `ParseTrinary(arg string) (int64, error)`. If your tests expect a different shape, that is the one thing to adjust.
