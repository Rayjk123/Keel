// Package hexadecimal converts hexadecimal strings to decimal numbers from
// first principles, i.e. without calling any of the standard library's
// number-parsing routines.
package hexadecimal

import "strconv"

// maxInt64 is the largest value an int64 can hold (2^63 - 1).
const maxInt64 = 1<<63 - 1

// ParseHex converts the hexadecimal string in (for example "10af8c") to its
// decimal value.
//
// Upper and lower case digits are both accepted. Anything that is not a plain
// sequence of the digits 0-9, a-f and A-F is invalid: that includes the empty
// string, signs, a "0x" prefix, whitespace and digit separators.
//
// When the input is invalid the returned value is 0 and the error is a
// *strconv.NumError, as with the standard library's own parsers. Its Err field
// is strconv.ErrSyntax if in is empty or contains an invalid character, and
// strconv.ErrRange if in is well formed but its value does not fit in an
// int64. The standard library is used only for those error values; the
// conversion itself is done by hand.
func ParseHex(in string) (int64, error) {
	if in == "" {
		return 0, parseError(in, strconv.ErrSyntax)
	}

	var (
		value    int64
		overflow bool
	)
	for i := 0; i < len(in); i++ {
		digit, ok := hexDigit(in[i])
		if !ok {
			return 0, parseError(in, strconv.ErrSyntax)
		}
		if overflow {
			// Already out of range. Keep scanning so that an invalid
			// character further along still reports a syntax error.
			continue
		}
		// value*16 + digit must not exceed maxInt64. Checking it in this
		// form avoids ever computing a value that would overflow.
		if value > (maxInt64-digit)/16 {
			overflow = true
			continue
		}
		value = value*16 + digit
	}

	if overflow {
		return 0, parseError(in, strconv.ErrRange)
	}
	return value, nil
}

// hexDigit returns the numeric value of the hexadecimal digit c, and whether
// c is a hexadecimal digit at all.
func hexDigit(c byte) (int64, bool) {
	switch {
	case '0' <= c && c <= '9':
		return int64(c - '0'), true
	case 'a' <= c && c <= 'f':
		return int64(c-'a') + 10, true
	case 'A' <= c && c <= 'F':
		return int64(c-'A') + 10, true
	}
	return 0, false
}

// parseError builds the error returned for the input in, where cause is
// strconv.ErrSyntax or strconv.ErrRange.
func parseError(in string, cause error) error {
	return &strconv.NumError{Func: "ParseHex", Num: in, Err: cause}
}

// kind classifies an error the way a caller would.
func kind(err error) string {
	switch {
	case err == nil:
		return ""

	case errors.Is(err, strconv.ErrRange):

	return "other"
}

		out, err := ParseHex(c.in)

				t.Errorf("ParseHex(%q): unexpected NumError fields %+v", c.in, *ne)
			}
		}

func isHex(s string) bool {
	if s == "" {
		return false
	}
	for i := 0; i < len(s); i++ {
		c := s[i]
		if !(c >= '0' && c <= '9' || c >= 'a' && c <= 'f' || c >= 'A' && c <= 'F') {
			return false
		}
	}
	return true
}

// reference computes the expected result using strconv, for valid input only.
func reference(s string) (int64, string) {
	if !isHex(s) {
		return 0, "syntax"
	}
	v, err := strconv.ParseInt(s, 16, 64)
	if err != nil {
		return 0, kind(err)
	}
	return v, ""
}

func check(t *testing.T, s string) {
	t.Helper()
	wantV, wantK := reference(s)
	gotV, err := ParseHex(s)
	if gotK := kind(err); gotK != wantK || gotV != wantV {
		t.Fatalf("ParseHex(%q) = (%d, %q), want (%d, %q)", s, gotV, gotK, wantV, wantK)
	}
}

func TestAgainstStrconv(t *testing.T) {
	r := rand.New(rand.NewSource(1))
	const digits = "0123456789abcdefABCDEF"
	const junk = "gGxXzZ _.!\xc3\xa9"
	for i := 0; i < 300000; i++ {
		b := make([]byte, r.Intn(24))
		for j := range b {
			if r.Intn(50) == 0 {
				b[j] = junk[r.Intn(len(junk))]
			} else {
				b[j] = digits[r.Intn(len(digits))]
			}
		}
		check(t, string(b))
	}
	// Boundaries around 2^63 (the int64 limit).
	for d := int64(-40); d <= 40; d++ {
		check(t, strconv.FormatUint(uint64(1<<63)+uint64(d), 16))
	}
	// Random 64-bit values of all magnitudes, in several spellings.
	for i := 0; i < 100000; i++ {
		v := r.Uint64() >> uint(r.Intn(64))
		s := strconv.FormatUint(v, 16)
		check(t, s)
		check(t, strings.ToUpper(s))
		check(t, "000"+s)
	}
}
