import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.IntStream;

class TwelveDays {

    private static final int FIRST_VERSE = 1;
    private static final int LAST_VERSE = 12;

    private static final String[] ORDINALS = {
        "first", "second", "third", "fourth", "fifth", "sixth",
        "seventh", "eighth", "ninth", "tenth", "eleventh", "twelfth"
    };

    private static final String[] GIFTS = {
        "a Partridge in a Pear Tree",
        "two Turtle Doves",
        "three French Hens",
        "four Calling Birds",
        "five Gold Rings",
        "six Geese-a-Laying",
        "seven Swans-a-Swimming",
        "eight Maids-a-Milking",
        "nine Ladies Dancing",
        "ten Lords-a-Leaping",
        "eleven Pipers Piping",
        "twelve Drummers Drumming"
    };

    String verse(int verseNumber) {
        validateVerseNumber(verseNumber);

        List<String> gifts = new ArrayList<>();
        for (int day = verseNumber; day >= FIRST_VERSE; day--) {
            String gift = GIFTS[day - 1];
            boolean isLastGift = day == FIRST_VERSE;
            gifts.add(isLastGift && verseNumber > FIRST_VERSE ? "and " + gift : gift);
        }

        return "On the " + ORDINALS[verseNumber - 1] + " day of Christmas my true love gave to me: "
                + String.join(", ", gifts) + ".\n";
    }

    String verses(int startVerse, int endVerse) {
        validateVerseNumber(startVerse);
        validateVerseNumber(endVerse);
        if (startVerse > endVerse) {
            throw new IllegalArgumentException(
                    "startVerse (" + startVerse + ") must not be greater than endVerse (" + endVerse + ")");
        }

        return IntStream.rangeClosed(startVerse, endVerse)
                .mapToObj(this::verse)
                .collect(Collectors.joining("\n"));
    }

    String sing() {
        return verses(FIRST_VERSE, LAST_VERSE);
    }

    private static void validateVerseNumber(int verseNumber) {
        if (verseNumber < FIRST_VERSE || verseNumber > LAST_VERSE) {
            throw new IllegalArgumentException(
                    "Verse number must be between " + FIRST_VERSE + " and " + LAST_VERSE + ": " + verseNumber);
        }
    }
}

import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Arrays;

public class Check {
    static int failures = 0;

    static void check(String name, boolean ok) {
        if (!ok) {
            failures++;
            System.out.println("FAIL: " + name);
        }
    }

    static void expectThrows(String name, Runnable r) {
        try {
            r.run();
            failures++;
            System.out.println("FAIL (no exception): " + name);
        } catch (IllegalArgumentException e) {
            // expected
        } catch (RuntimeException e) {
            failures++;
            System.out.println("FAIL (wrong exception " + e.getClass().getName() + "): " + name);
        }
    }

    public static void main(String[] args) throws Exception {
        String expected = Files.readString(Path.of("/tmp/twelve/expected.txt"), StandardCharsets.UTF_8);
        TwelveDays td = new TwelveDays();

        // 1. whole song, exact match
        String sung = td.sing();
        Files.writeString(Path.of("/tmp/twelve/actual.txt"), sung, StandardCharsets.UTF_8);
        check("sing() equals expected text", sung.equals(expected));

        // 2. split the reference text into its 12 verses
        String[] raw = expected.split("\n\n");
        check("reference has 12 verses", raw.length == 12);

        // 3. every individual verse
        for (int n = 1; n <= 12; n++) {
            check("verse(" + n + ")", td.verse(n).equals(raw[n - 1].stripTrailing() + "\n"));
        }

        // 4. every valid verses(a, b) range, built independently from the reference text
        int ranges = 0;
        for (int a = 1; a <= 12; a++) {
            for (int b = a; b <= 12; b++) {
                String exp = String.join("\n\n",
                        Arrays.stream(raw, a - 1, b).map(String::stripTrailing).toArray(String[]::new)) + "\n";
                check("verses(" + a + "," + b + ")", td.verses(a, b).equals(exp));
                ranges++;
            }
        }

        // 5. invalid input
        expectThrows("verse(0)", () -> td.verse(0));
        expectThrows("verse(13)", () -> td.verse(13));
        expectThrows("verse(-1)", () -> td.verse(-1));
        expectThrows("verses(0,3)", () -> td.verses(0, 3));
        expectThrows("verses(1,13)", () -> td.verses(1, 13));
        expectThrows("verses(5,4)", () -> td.verses(5, 4));

        System.out.println("checked: sing(), 12 verses, " + ranges + " verses() ranges, 6 invalid-input cases");
        System.out.println(failures == 0 ? "ALL CHECKS PASSED" : failures + " CHECK(S) FAILED");
        System.exit(failures == 0 ? 0 : 1);
    }
}

`TwelveDays.java` now returns the full song, and `sing()` is byte-for-byte identical to the lyrics in the task.

- **`verse(n)`** builds one verse from an ordinal table and a gift table. It lists the gifts from day n down to day 1, and from verse 2 on it prefixes the partridge with `and `.
- **`verses(a, b)`** joins the verses in that range with `\n`. Each verse ends in `\n`, so a blank line separates them.
- **`sing()`** is `verses(1, 12)`.
- **Invalid input** throws `IllegalArgumentException` for verse numbers outside 1–12 or a start verse after the end verse.

The repo has no tests, so I checked against the task's lyrics with a throwaway harness compiled with `javac` into `/tmp`. It compared `sing()`, all 12 verses, all 78 `verses(a, b)` ranges and the error cases, and everything passed. I did not run the Gradle build, only `javac`.

I assumed the usual Exercism format, where each verse ends with a single `\n` and `sing()` ends with one trailing newline. A test expecting no trailing newline would not match.

I removed the scratch files and left `.oracle/solution.enc` untouched, so `TwelveDays.java` is the only file changed.
