export const sum = (factors = [], limit = 0) => {
  const multiples = new Set();
  for (const factor of factors) {
    if (factor <= 0) continue;
    for (let m = factor; m < limit; m += factor) {
      multiples.add(m);
    }
  }
  let total = 0;
  for (const m of multiples) total += m;
  return total;
};
EOF
