#if !defined(GIGASECOND_H)
#define GIGASECOND_H

// The umbrella posix_time header is used deliberately: besides ptime and
// time_duration it provides time_from_string() and (transitively) the
// boost::gregorian date types that callers use to build moments in time.
#include <boost/date_time/posix_time/posix_time.hpp>

namespace gigasecond {

// Returns the moment exactly one gigasecond (10^9 seconds) after `start`.
//
// ptime is a non-adjusted time point (no time zone, DST or leap seconds),
// so this is plain elapsed-time arithmetic; Boost takes care of rolling
// over days, months, years and leap years. Special values such as
// not_a_date_time and +/-infinity propagate unchanged.
boost::posix_time::ptime advance(boost::posix_time::ptime start);

}  // namespace gigasecond

#endif // GIGASECOND_H
EOF
cat > gigasecond.cpp <<'EOF'
#include "gigasecond.h"

namespace gigasecond {

namespace {

// A gigasecond is one thousand million seconds: a one with nine zeros.
// (A plain integer constant avoids any static-initialization-order issues.)
constexpr long seconds_per_gigasecond = 1'000'000'000L;

}  // namespace

boost::posix_time::ptime advance(boost::posix_time::ptime start)
{
    return start + boost::posix_time::seconds(seconds_per_gigasecond);
}

}  // namespace gigasecond
EOF

**What I changed**
- **`gigasecond.h`**: declares `boost::posix_time::ptime advance(boost::posix_time::ptime start);`. It includes the full Boost posix_time header because the tests build their inputs with `time_from_string`, `ptime` and `date` without including Boost themselves.
- **`gigasecond.cpp`**: returns `start + seconds(1'000'000'000)`. Boost handles carrying over into days, months, years and leap years.

**Testing**
