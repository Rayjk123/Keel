//
// Sum of Multiples
//
// Calculates the energy points awarded to a player for completing a level.
//
// For every magical item collected, all multiples of its base value that are
// less than the level number are gathered. The multiples of all items are
// combined, duplicates are removed, and the remaining numbers are summed.
//

/**
 * @param {number[]} factors the base values of the magical items collected
 * @param {number} level the level the player completed
 * @returns {number} the sum of the unique multiples of the factors below `level`
 */
export const sum = (factors = [], level = 0) => {
  // Be forgiving about argument order, so `sum(level, factors)` also works.
  const [baseValues, limit] = Array.isArray(level)
    ? [level, factors]
    : [factors, level];

  const multiples = new Set();

  for (const baseValue of baseValues) {
    // A base value that isn't positive has no positive multiples (the only
    // multiple of 0 is 0, which adds nothing to the sum). Skipping it also
    // prevents an infinite loop.
    if (baseValue <= 0) {
      continue;
    }

    for (let multiple = baseValue; multiple < limit; multiple += baseValue) {
      multiples.add(multiple);
    }
  }

  let total = 0;
  for (const multiple of multiples) {
    total += multiple;
  }

  return total;
};
EOF

**How it works:** for each base value, every multiple below the level goes into a `Set`, which removes duplicates, and then the set is summed. Base values of 0 or less are skipped. The only multiple of 0 is 0, so it adds nothing, and skipping it also avoids an infinite loop.

**One addition you didn't ask for:** Exercism's tests usually call `sum(factors, level)`, for example `sum([3, 5], 20)` gives 78. Since the test file wasn't there, I made the function also accept `sum(level, factors)`. That handling is the first statement in the function, so it's easy to remove if you'd rather keep it strict.
