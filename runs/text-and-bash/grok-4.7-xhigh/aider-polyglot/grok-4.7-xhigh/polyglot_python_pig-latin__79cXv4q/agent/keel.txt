def translate(text):
    return " ".join(_translate_word(word) for word in text.split())


def _translate_word(word):
    vowels = set("aeiou")

    # Rule 1: begins with a vowel, or starts with "xr" or "yt"
    if word[0] in vowels or word.startswith(("xr", "yt")):
        return word + "ay"

    # Rule 3: zero or more consonants followed by "qu"
    qu_index = word.find("qu")
    if qu_index != -1 and all(c not in vowels for c in word[:qu_index]):
        return word[qu_index + 2 :] + word[: qu_index + 2] + "ay"

    # Rule 4: one or more consonants followed by "y"
    y_index = word.find("y")
    if y_index > 0 and all(c not in vowels for c in word[:y_index]):
        return word[y_index:] + word[:y_index] + "ay"

    # Rule 2: begins with one or more consonants
    consonant_count = 0
    for char in word:
        if char in vowels:
            break
        consonant_count += 1
    return word[consonant_count:] + word[:consonant_count] + "ay"
