pub fn abbreviate(phrase: &str) -> String {
    let mut acronym = String::new();
    let mut start_of_word = true;
    let mut prev_was_lowercase = false;

    for c in phrase.chars() {
        if c.is_alphabetic() {
            if start_of_word || (prev_was_lowercase && c.is_uppercase()) {
                acronym.push(c.to_ascii_uppercase());
            }
            start_of_word = false;
            prev_was_lowercase = c.is_lowercase();
        } else if c.is_whitespace() || c == '-' {
            // Hyphens are word separators, just like whitespace.
            start_of_word = true;
            prev_was_lowercase = false;
        }
        // All other punctuation is ignored (removed) and does not split words.
    }

    acronym
}
EOF

#[cfg(test)]
mod tests {

        assert_eq!(abbreviate("GNU Image Manipulation Program"), "GIMP");
    }

        assert_eq!(abbreviate("Complementary metal-oxide semiconductor"), "CMOS");

        );

    #[test]
    fn consecutive_delimiters() {
        assert_eq!(abbreviate("Something - I made up from thin air"), "SIMUFTA");

}
