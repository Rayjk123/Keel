package trinary

import (
	"errors"
	"math"
)

// ParseTrinary converts a trinary string to its decimal equivalent.
// Invalid trinary strings and values that overflow int64 return 0 and an error.
func ParseTrinary(s string) (int64, error) {
	const third = math.MaxInt64 / 3
	overflow := false
	var n int64

	for _, r := range s {
		if r < '0' || r > '2' {
			return 0, errors.New("invalid trinary")
		}
		if overflow {
			continue
		}
		digit := int64(r - '0')
		if n > third || (n == third && digit > math.MaxInt64%3) {
			overflow = true
			continue
		}
		n = n*3 + digit
	}

	if overflow {
		return 0, errors.New("overflow")
	}
	return n, nil
}
EOF
