package hexadecimal

import "errors"

// ParseHex converts a hexadecimal string to its decimal equivalent.
// The conversion is done from first principles (no strconv or similar).
// Invalid hexadecimal strings return an error.
func ParseHex(in string) (int64, error) {
	if len(in) == 0 {
		return 0, errors.New("invalid hexadecimal string")
	}

	const maxInt64 = int64(^uint64(0) >> 1)

	var result int64
	for i := 0; i < len(in); i++ {
		digit, ok := hexDigit(in[i])
		if !ok {
			return 0, errors.New("invalid hexadecimal string")
		}
		if result > (maxInt64-digit)/16 {
			return 0, errors.New("hexadecimal value overflows int64")
		}
		result = result*16 + digit
	}
	return result, nil
}

// hexDigit maps a single ASCII hex character to its numeric value.
func hexDigit(c byte) (int64, bool) {
	switch {
	case c >= '0' && c <= '9':
		return int64(c - '0'), true
	case c >= 'a' && c <= 'f':
		return int64(c-'a') + 10, true
	case c >= 'A' && c <= 'F':
		return int64(c-'A') + 10, true
	default:
		return 0, false
	}
}
EOF

package hexadecimal

import "testing"

func TestParseHex(t *testing.T) {
	cases := []struct {
		in      string
		want    int64
		wantErr bool
	}{
		{"1", 1, false},
		{"10", 16, false},
		{"2d", 45, false},
		{"cfcfcf", 13619151, false},
		{"cf", 207, false},
		{"G", 0, true},
		{"", 0, true},
		{"10af8c", 1093516, false},
		{"10AF8C", 1093516, false},
		{"000000", 0, false},
		{"ffffff", 16777215, false},
		{"ffff00", 16776960, false},
		{"carrot", 0, true},
		{"008000", 32768, false},
		{"008080", 32896, false},
		{"000080", 128, false},
		{"7fffffffffffffff", 9223372036854775807, false},
		{"8000000000000000", 0, true},
		{"0x10", 0, true},
		{" 10", 0, true},
	}
	for _, tc := range cases {
		got, err := ParseHex(tc.in)
		if tc.wantErr {
			if err == nil {
				t.Errorf("ParseHex(%q) = %d, want error", tc.in, got)
			}
			continue
		}
		if err != nil {
			t.Errorf("ParseHex(%q) unexpected error: %v", tc.in, err)
		}
		if got != tc.want {
			t.Errorf("ParseHex(%q) = %d, want %d", tc.in, got, tc.want)
		}
	}
}
EOF
