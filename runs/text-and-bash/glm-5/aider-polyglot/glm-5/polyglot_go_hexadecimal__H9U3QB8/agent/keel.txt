package hexadecimal

import (
	"fmt"
)

// ParseHex converts a hexadecimal string to its decimal equivalent.
// If the input contains invalid hex characters, it returns an error.
// out and errCase are used for test verification.
func ParseHex(in string, out int64, errCase string) {
	result, err := parseHexString(in)
	
	// If errCase is provided, we expect an error
	if errCase != "" {
		if err == nil {
			panic(fmt.Sprintf("expected error for case: %s", errCase))
		}
		return
	}
	
	// If no errCase, we expect successful parsing
	if err != nil {
		panic(fmt.Sprintf("unexpected error: %v", err))
	}
	
	// Verify the result matches expected output
	if result != out {
		panic(fmt.Sprintf("expected %d, got %d", out, result))
	}
}

// parseHexString converts a hex string to int64 using first principles
func parseHexString(hex string) (int64, error) {
	if len(hex) == 0 {
		return 0, fmt.Errorf("empty string")
	}
	
	var result int64 = 0
	
	for i, char := range hex {
		// Skip the "0x" or "0X" prefix if present
		if i == 1 && (hex[0] == '0') && (char == 'x' || char == 'X') {
			continue
		}
		if i == 0 && char == '0' && len(hex) > 1 && (hex[1] == 'x' || hex[1] == 'X') {
			continue
		}
		
		var digit int64
		
		switch {
		case char >= '0' && char <= '9':
			digit = int64(char - '0')
		case char >= 'a' && char <= 'f':
			digit = int64(char-'a') + 10
		case char >= 'A' && char <= 'F':
			digit = int64(char-'A') + 10
		default:
			return 0, fmt.Errorf("invalid hex character: %c", char)
		}
		
		result = result*16 + digit
	}
	
	return result, nil
}
