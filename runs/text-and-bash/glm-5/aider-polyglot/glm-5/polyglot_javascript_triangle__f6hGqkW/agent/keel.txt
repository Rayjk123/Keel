// convenience to get you started writing code faster.

export class Triangle {
  constructor(...sides) {
    this.sides = sides.sort((a, b) => a - b);
    
    // Validate that all sides are greater than 0
    if (this.sides.some(side => side <= 0)) {
      throw new Error('All sides must be greater than 0');
    }
    
    // Validate triangle inequality
    // After sorting: sides[0] + sides[1] must be >= sides[2]
    if (this.sides[0] + this.sides[1] < this.sides[2]) {
      throw new Error('Triangle inequality violated');
    }
  }

  get isEquilateral() {
    return this.sides[0] === this.sides[1] && this.sides[1] === this.sides[2];
  }

  get isIsosceles() {
    return this.sides[0] === this.sides[1] || 
           this.sides[1] === this.sides[2] || 
           this.sides[0] === this.sides[2];
  }

  get isScalene() {
    return this.sides[0] !== this.sides[1] && 
           this.sides[1] !== this.sides[2] && 
           this.sides[0] !== this.sides[2];
  }
}
