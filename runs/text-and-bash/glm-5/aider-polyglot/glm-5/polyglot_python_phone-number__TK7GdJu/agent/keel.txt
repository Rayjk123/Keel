class PhoneNumber:
    def __init__(self, number):
        self._number = self._clean(number)
    
    def _clean(self, number):
        # Check for letters first
        for char in number:
            if char.isalpha():
                raise ValueError("letters not permitted")
        
        # Extract digits only
        digits = ''.join(char for char in number if char.isdigit())
        
        # Check length constraints
        if len(digits) < 10:
            raise ValueError("must not be fewer than 10 digits")
        if len(digits) > 11:
            raise ValueError("must not be greater than 11 digits")
        
        # Handle 11-digit numbers
        if len(digits) == 11:
            if digits[0] != '1':
                raise ValueError("11 digits must start with 1")
            digits = digits[1:]  # Remove country code
        
        # Check for invalid characters (punctuation where digits should be)
        # This is after extracting digits, so we need original checks
        # Actually we need to verify that non-digit chars are only punctuation, not letters (already done)
        # and that we have valid area/exchange codes
        
        # Check area code (first 3 digits)
        area_code = digits[0:3]
        if area_code[0] == '0':
            raise ValueError("area code cannot start with zero")
        if area_code[0] == '1':
            raise ValueError("area code cannot start with one")
        
        # Check exchange code (digits 4-6)
        exchange_code = digits[3:6]
        if exchange_code[0] == '0':
            raise ValueError("exchange code cannot start with zero")
        if exchange_code[0] == '1':
            raise ValueError("exchange code cannot start with one")
        
        return digits
    
    @property
    def number(self):
        return self._number
    
    def pretty(self):
        return f"({self._number[0:3]})-{self._number[3:6]}-{self._number[6:10]}"
    
    @property
    def area_code(self):
        return self._number[0:3]
