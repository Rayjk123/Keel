def translate(text):
    vowels = "aeiou"
    words = text.split()
    translated_words = []
    
    for word in words:
        # Rule 1: starts with vowel or "xr" or "yt"
        if word[0] in vowels or word.startswith("xr") or word.startswith("yt"):
            translated_words.append(word + "ay")
            continue
        
        # Find the consonant cluster at the beginning
        i = 0
        while i < len(word) and word[i] not in vowels and word[i] != 'y':
            i += 1
        
        # Rule 3: consonants followed by "qu"
        if i > 0 and i < len(word) - 1 and word[i] == 'q' and word[i+1] == 'u':
            # Move consonants + "qu" to end
            translated = word[i+2:] + word[:i+2] + "ay"
            translated_words.append(translated)
            continue
        
        # Rule 4: consonants followed by "y"
        if i > 0 and i < len(word) and word[i] == 'y':
            # Move consonants before 'y' to end
            translated = word[i:] + word[:i] + "ay"
            translated_words.append(translated)
            continue
        
        # Rule 2: regular consonant cluster
        translated = word[i:] + word[:i] + "ay"
        translated_words.append(translated)
    
    return " ".join(translated_words)

def translate(text):
    vowels = "aeiou"
    words = text.split()
    translated_words = []
    
    for word in words:
        # Rule 1: starts with vowel or "xr" or "yt"
        if word[0] in vowels or word.startswith("xr") or word.startswith("yt"):
            translated_words.append(word + "ay")
            continue
        
        # Find the consonant cluster at the beginning
        # consonants are letters that are not vowels and not 'y' (y can be special)
        i = 0
        while i < len(word) and word[i] not in vowels:
            # Handle 'y' - stop if we see a 'y' after at least one consonant
            if word[i] == 'y' and i > 0:
                break
            i += 1
        
        # Rule 3: check if we have "qu" in the consonant cluster
        # The 'q' must be followed by 'u' at position i or within the cluster
        if i <= len(word) - 1 and word[i-1:i+1] == "qu" if i > 0 else False:
            # "qu" is at the end of consonant cluster, include both
            translated = word[i+1:] + word[:i+1] + "ay"
            translated_words.append(translated)
            continue
        
        # Rule 4: consonants followed by "y"
        if i > 0 and i < len(word) and word[i] == 'y':
            # Move consonants before 'y' to end
            translated = word[i:] + word[:i] + "ay"
            translated_words.append(translated)
            continue
        
        # Rule 2: regular consonant cluster
        translated = word[i:] + word[:i] + "ay"
        translated_words.append(translated)
    
    return " ".join(translated_words)
