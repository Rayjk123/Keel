package trinary

import (
	"math"
)

// ParseTrinary converts a trinary number string to its decimal equivalent.
// Returns 0 for invalid trinary strings (containing characters other than 0, 1, 2).
func ParseTrinary(trinary string) int64 {
	var result int64 = 0
	length := len(trinary)
	
	for i, char := range trinary {
		// Check if character is valid (0, 1, or 2)
		if char < '0' || char > '2' {
			return 0
		}
		
		// Convert char to digit
		digit := int64(char - '0')
		
		// Calculate position from right (0-indexed from right)
		position := length - i - 1
		
		// Add digit * 3^position to result
		result += digit * int64(math.Pow(3, float64(position)))
	}
	
	return result
}
