class PhoneNumber:
    def __init__(self, number):
        # Check for letters first
        if any(c.isalpha() for c in number):
            raise ValueError("letters not permitted")
        
        # Check for punctuation (anything that's not a digit, not space, and not common phone separators)
        allowed_chars = set("0123456789 .-+()")
        if any(c not in allowed_chars for c in number):
            raise ValueError("punctuations not permitted")
        
        # Extract only digits
        digits = ''.join(c for c in number if c.isdigit())
        
        # Check length constraints
        if len(digits) < 10:
            raise ValueError("must not be fewer than 10 digits")
        
        if len(digits) > 11:
            raise ValueError("must not be greater than 11 digits")
        
        # If 11 digits, must start with 1
        if len(digits) == 11:
            if digits[0] != '1':
                raise ValueError("11 digits must start with 1")
            # Strip the country code
            digits = digits[1:]
        
        # Now we have 10 digits
        # Get area code and exchange code
        area_code = digits[0:3]
        exchange_code = digits[3:6]
        
        # Validate area code
        if area_code[0] == '0':
            raise ValueError("area code cannot start with zero")
        if area_code[0] == '1':
            raise ValueError("area code cannot start with one")
        
        # Validate exchange code
        if exchange_code[0] == '0':
            raise ValueError("exchange code cannot start with zero")
        if exchange_code[0] == '1':
            raise ValueError("exchange code cannot start with one")
        
        # Store the cleaned number
        self.number = digits
        self.area_code = area_code
        self.exchange_code = exchange_code
        self.subscriber_number = digits[6:]
    
    def pretty(self):
        return f"({self.area_code})-{self.exchange_code}-{self.subscriber_number}"
EOF
