export const sum = (items, level) => {
  const uniqueMultiples = new Set();
  
  for (const baseValue of items) {
    if (baseValue > 0) {
      for (let multiple = baseValue; multiple < level; multiple += baseValue) {
        uniqueMultiples.add(multiple);
      }
    }
  }
  
  let total = 0;
  for (const num of uniqueMultiples) {
    total += num;
  }
  
  return total;
};
EOF
