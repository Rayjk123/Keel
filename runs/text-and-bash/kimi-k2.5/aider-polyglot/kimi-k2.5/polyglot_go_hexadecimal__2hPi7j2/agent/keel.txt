package hexadecimal

import (
	"fmt"
)

// ParseHex converts a hexadecimal string to its decimal equivalent.
// Returns the parsed value and an error if the string is invalid.
func ParseHex(in string) (int64, error) {
	if len(in) == 0 {
		return 0, fmt.Errorf("empty hexadecimal string")
	}
	
	var result int64 = 0
	
	for _, char := range in {
		var value int64
		
		switch {
		case char >= '0' && char <= '9':
			value = int64(char - '0')
		case char >= 'a' && char <= 'f':
			value = int64(char - 'a' + 10)
		case char >= 'A' && char <= 'F':
			value = int64(char - 'A' + 10)
		default:
			return 0, fmt.Errorf("invalid hexadecimal character: %c", char)
		}
		
		// Check for overflow before multiplying
		if result > (1<<63-1)/16 {
			return 0, fmt.Errorf("hexadecimal value overflows int64")
		}
		
		result = result*16 + value
	}
	
	return result, nil
}
