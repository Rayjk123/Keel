            # Rule 3: Word starts with consonants followed by "qu"
            return word[2:] + word[:2] + "ay"
        elif word[0] in vowels or word.startswith("xr") or word.startswith("yt"):
            # Rule 1: Word starts with a vowel, "xr" or "yt"
            return word + "ay"
        elif "qu" in word:
            # Rule 3: Word starts with consonants followed by "qu"
            match = re.match(r"^([^qu]*qu)", word)
            if match:
                return word[match.end():] + match.group(1) + "ay"
        elif word.endswith("y"):
            # Rule 4: Word starts with consonants followed by "y"
            match = re.match(r"^([^y]*)y", word)
            if match:
                return word[match.end():] + match.group(1) + "ay"
        else:
            # Rule 2: Word starts with one or more consonants

        return word

    translated_words = [translate_word(word) for word in words]
