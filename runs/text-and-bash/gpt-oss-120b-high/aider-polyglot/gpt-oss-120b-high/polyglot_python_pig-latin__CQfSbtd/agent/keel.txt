#!/usr/bin/env python3
import re

def translate(text):
    """Translate a string of English words into Pig Latin.

    This follows the four translation rules described in the problem
    statement. Words are assumed to be separated by whitespace.
    """
    def translate_word(word: str) -> str:
        # Guard against empty strings
        if not word:
            return word

        lower = word.lower()

        # Rule 1: word starts with a vowel, "xr", or "yt"
        if lower[0] in "aeiou" or lower.startswith("xr") or lower.startswith("yt"):
            return word + "ay"

        # Rule 3: leading consonants (maybe none) followed by "qu"
        # Capture the whole leading block that ends with "qu"
        m = re.match(r"^([^aeiou]*?qu)(.*)", lower)
        if m:
            prefix_len = len(m.group(1))
            return word[prefix_len:] + word[:prefix_len] + "ay"

        # Find the index of the first vowel (including a 'y' that acts as a vowel)
        first_vowel_idx = None
        for i, ch in enumerate(lower):
            if ch in "aeiou":
                first_vowel_idx = i
                break
            # 'y' counts as a vowel only when it is not the first character
            if ch == "y" and i != 0:
                first_vowel_idx = i
                break

        # If no vowel is found, just append "ay"
        if first_vowel_idx is None:
            return word + "ay"

        # Rule 4: the first vowel encountered is a 'y' after some consonants
        if lower[first_vowel_idx] == "y":
            return word[first_vowel_idx:] + word[:first_vowel_idx] + "ay"

        # Rule 2: normal consonant cluster before a vowel
        return word[first_vowel_idx:] + word[:first_vowel_idx] + "ay"

    # Translate each word separated by whitespace and reassemble the result.
    return " ".join(translate_word(w) for w in text.split())
