import java.util.List;
import java.util.ArrayList;

/**
 * The {@code Series} class provides functionality to extract contiguous substrings
 * (slices) of a given length from a string of digits.
 *
 * <p>The input string is validated to contain only digit characters.
 * The {@code slices} method returns all possible slices of the requested length
 * in the order they appear. If the requested length is invalid, an
 * {@link IllegalArgumentException} is thrown.
 */
class Series {
    /** The underlying series of digits. */
    private final String digits;

    /**
     * Constructs a {@code Series} from the provided string.
     *
     * @param string a string containing only digit characters
     * @throws IllegalArgumentException if {@code string} contains any non‑digit characters
     */
    Series(String string) {
        if (string == null || !string.chars().allMatch(Character::isDigit)) {
            throw new IllegalArgumentException("Series must contain only digits");
        }
        this.digits = string;
    }

    /**
     * Returns a list of all contiguous substrings of length {@code num}.
     *
     * @param num the length of each slice; must be between 1 and the length of the series (inclusive)
     * @return a {@link List} containing each slice in the order they appear in the series
     * @throws IllegalArgumentException if {@code num} is less than 1 or greater than the length of the series
     */
    List<String> slices(int num) {
        if (num <= 0 || num > digits.length()) {
            throw new IllegalArgumentException("Slice size must be between 1 and the length of the series");
        }
        List<String> result = new ArrayList<>();
        for (int i = 0; i <= digits.length() - num; i++) {
            result.add(digits.substring(i, i + num));
        }
        return result;
    }
}
EOF
