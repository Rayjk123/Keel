// This file implements the 'Sum Of Multiples' exercise.
//
// The exported function `sum` calculates the sum of all unique multiples of the
// provided base values that are less than the given limit.
//
// Example:
//   sum(20, [3, 5]) => 78
//
// The algorithm:
//   1. Validate inputs.
//   2. For each base value, generate its multiples less than the limit.
//   3. Store each multiple in a Set to ensure uniqueness.
//   4. Sum the values in the Set and return the total.

export const sum = (limit, bases) => {
  // Guard against non‑positive limits: no multiples can be less than such a limit.
  if (typeof limit !== 'number' || limit <= 0) {
    return 0;
  }

  // Ensure we have a valid array of base values.
  if (!Array.isArray(bases) || bases.length === 0) {
    return 0;
  }

  const uniqueMultiples = new Set();

  for (const base of bases) {
    // Ignore non‑positive base values (zero or negative) as they would generate
    // an infinite or meaningless series of multiples.
    if (typeof base !== 'number' || base <= 0) {
      continue;
    }

    // Generate multiples of `base` that are strictly less than `limit`.
    for (let multiple = base; multiple < limit; multiple += base) {
      uniqueMultiples.add(multiple);
    }
  }

  // Sum all unique multiples.
  let total = 0;
  for (const value of uniqueMultiples) {
    total += value;
  }
  return total;
};
