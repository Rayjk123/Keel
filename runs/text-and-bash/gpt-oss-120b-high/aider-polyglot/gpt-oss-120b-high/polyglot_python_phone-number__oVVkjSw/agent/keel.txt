"""Phone Number cleaning and validation according to NANP rules."""

import re

class PhoneNumber:
    """Represents a sanitized NANP phone number.

    The number is stored in the ``number`` attribute as a ten‑digit string.
    Invalid input raises :class:`ValueError` with a descriptive message.
    """

    def __init__(self, raw_number: str):
        # First, check for letters – they are not allowed.
        if any(ch.isalpha() for ch in raw_number):
            raise ValueError("letters not permitted")

        # Allowed non‑digit characters: space, hyphen, period, parentheses and plus sign.
        allowed = set(" -().+")
        for ch in raw_number:
            if not (ch.isdigit() or ch in allowed):
                raise ValueError("punctuations not permitted")

        # Extract only digits.
        digits = re.sub(r'\D', '', raw_number)

        # Length checks.
        if len(digits) < 10:
            raise ValueError("must not be fewer than 10 digits")
        if len(digits) > 11:
            raise ValueError("must not be greater than 11 digits")

        # If we have 11 digits, the first must be the country code '1'.
        if len(digits) == 11:
            if digits[0] != '1':
                raise ValueError("11 digits must start with 1")
            digits = digits[1:]  # strip the leading country code

        # At this point we must have exactly ten digits.
        # Area code validation – first digit cannot be 0 or 1.
        if digits[0] == '0':
            raise ValueError("area code cannot start with zero")
        if digits[0] == '1':
            raise ValueError("area code cannot start with one")

        # Exchange code validation – the first digit of the exchange code (fourth digit overall) cannot be 0 or 1.
        if digits[3] == '0':
            raise ValueError("exchange code cannot start with zero")
        if digits[3] == '1':
            raise ValueError("exchange code cannot start with one")

        # Store the cleaned 10‑digit number.
        self.number = digits

    def __repr__(self) -> str:
        return f"PhoneNumber('{self.number}')"
EOF
