def translate(text):
    return " ".join(_translate_word(word) for word in text.split())


def _translate_word(word):
    if word[0] in "aeiou" or word.startswith(("xr", "yt")):
        return word + "ay"

    for i, letter in enumerate(word):
        if letter in "aeiou":
            # "qu" is treated as a single consonant unit, even after other consonants.
            if letter == "u" and i > 0 and word[i - 1] == "q":
                i += 1
            break
        # "y" acts as a vowel when it follows one or more consonants.
        if letter == "y" and i > 0:
            break
    else:
        i = len(word)

    return word[i:] + word[:i] + "ay"
