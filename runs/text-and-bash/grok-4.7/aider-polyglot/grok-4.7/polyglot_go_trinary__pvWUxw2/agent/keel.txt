package trinary

import "fmt"

// ParseTrinary converts a trinary string to its decimal value.
// Invalid trinary strings and values that overflow int64 yield 0 and an error.
func ParseTrinary(s string) (int64, error) {
	const maxInt64 = int64(^uint64(0) >> 1)
	const third = maxInt64 / 3

	overflow := false
	var n int64
	for _, r := range s {
		if r < '0' || r > '2' {
			return 0, fmt.Errorf("invalid trinary: %s", s)
		}
		if overflow {
			continue
		}
		if n > third {
			overflow = true
			continue
		}
		n *= 3
		digit := int64(r - '0')
		if digit > maxInt64-n {
			overflow = true
			continue
		}
		n += digit
	}
	if overflow {
		return 0, fmt.Errorf("overflow: %s", s)
	}
	return n, nil
}
EOF
