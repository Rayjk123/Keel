package hexadecimal

import "errors"

// ParseHex converts a hexadecimal string to its decimal (base-10) equivalent.
// Conversion is done from first principles: each digit is shifted into an int64
// accumulator. Built-in parsing helpers are intentionally not used.
// Invalid digits, an empty string, or a value that does not fit in int64 yield an error.
func ParseHex(in string) (int64, error) {
	if in == "" {
		return 0, errors.New("invalid hexadecimal string")
	}

	const maxInt64 = int64(^uint64(0) >> 1)

	var result int64
	for _, c := range in {
		var digit int64
		switch {
		case c >= '0' && c <= '9':
			digit = int64(c - '0')
		case c >= 'a' && c <= 'f':
			digit = int64(c-'a') + 10
		case c >= 'A' && c <= 'F':
			digit = int64(c-'A') + 10
		default:
			return 0, errors.New("invalid hexadecimal string")
		}

		if result > (maxInt64-digit)/16 {
			return 0, errors.New("hexadecimal value overflows int64")
		}
		result = result*16 + digit
	}
	return result, nil
}

package hexadecimal

import (
	"errors"
	"strings"
)

// ParseHex converts a hexadecimal string to its decimal equivalent.
// The conversion is done from first principles (repeated multiply-by-16),
// without using built-in integer parsing. An empty string or any character
// that is not a hexadecimal digit produces an error whose message contains
// "syntax". A value that does not fit in an int64 produces an error whose
// message contains "range".
func ParseHex(in string) (int64, error) {
	if in == "" {
		return 0, errors.New("syntax")
	}

	const maxInt64 = int64(^uint64(0) >> 1)

	var result int64
	for _, c := range in {
		var digit int64
		switch {
		case c >= '0' && c <= '9':
			digit = int64(c - '0')
		case c >= 'a' && c <= 'f':
			digit = int64(c-'a') + 10
		case c >= 'A' && c <= 'F':
			digit = int64(c-'A') + 10
		default:
			return 0, errors.New("syntax")
		}

		if result > (maxInt64-digit)/16 {
			return 0, errors.New("range")
		}
		result = result*16 + digit
	}
	return result, nil
}

// HandleErrors reports the error case for each input passed to ParseHex.
// Each result is one of "none", "syntax", or "range".
func HandleErrors(inputs []string) []string {
	out := make([]string, len(inputs))
	for i, in := range inputs {
		_, err := ParseHex(in)
		switch {
		case err == nil:
			out[i] = "none"
		case strings.Contains(strings.ToLower(err.Error()), "syntax"):
			out[i] = "syntax"
		case strings.Contains(strings.ToLower(err.Error()), "range"):
			out[i] = "range"
		default:
			out[i] = "syntax"
		}
	}
	return out
}
EOF

package hexadecimal

import (
	"strings"
	"testing"
)

var testCases = []struct {
	in      string
	out     int64
	errCase string
}{
	{"1", 1, "none"},
	{"10", 0x10, "none"},
	{"2d", 0x2d, "none"},
	{"012", 0x12, "none"},
	{"cfcfcf", 0xcfcfcf, "none"},
	{"CFCFCF", 0xcfcfcf, "none"},
	{"", 0, "syntax"},
	{"peanut", 0, "syntax"},
	{"2cg134", 0, "syntax"},
	{"8000000000000000", 0, "range"},
	{"9223372036854775809", 0, "range"},
	{"10af8c", 0x10af8c, "none"},
	{"008000", 0x008000, "none"},
	{"008080", 0x008080, "none"},
	{"000080", 0x000080, "none"},
	{"7fffffffffffffff", 0x7fffffffffffffff, "none"},
	{"8000000000000001", 0, "range"},
	{"0", 0, "none"},
	{"a", 10, "none"},
	{"F", 15, "none"},
	{"g", 0, "syntax"},
	{" 1", 0, "syntax"},
}

func TestParseHex(t *testing.T) {
	for _, test := range testCases {
		out, err := ParseHex(test.in)
		if test.errCase != "none" {
			var _ error = err
			if err == nil {
				t.Errorf("ParseHex(%q): expected an error, but error is nil", test.in)
				continue
			}
			if !strings.Contains(strings.ToLower(err.Error()), test.errCase) {
				t.Errorf("ParseHex(%q) returned error %q. Expected error containing %q.", test.in, err, test.errCase)
			}
		} else {
			if out != test.out {
				t.Errorf("ParseHex(%q) = %d. Expected %d.", test.in, out, test.out)
			}
			if err != nil {
				t.Errorf("ParseHex(%q) returned error %q.  Error not expected.", test.in, err)
			}
		}
	}
}

func TestHandleErrors(t *testing.T) {
	tests := make([]string, len(testCases))
	for i, test := range testCases {
		tests[i] = test.in
	}
	er := HandleErrors(tests)
	if len(er) != len(tests) {
		t.Fatalf("For %d tests, HandleErrors returned %d results, want %d", len(tests), len(er), len(tests))
	}
	for i, e := range er {
		if e != testCases[i].errCase {
			t.Errorf("For ParseHex(%q), HandleErrors reports %q, want %q", tests[i], e, testCases[i].errCase)
		}
	}
}
EOF
