/// Convert a phrase to its acronym.
///
/// Hyphens and whitespace separate words. All other punctuation is ignored.
/// A letter contributes to the acronym when it starts a word, or when an
/// uppercase letter follows a lowercase letter (camel case).
pub fn abbreviate(phrase: &str) -> String {
    let mut acronym = String::new();
    let mut start_of_word = true;
    let mut prev_was_lowercase = false;

    for c in phrase.chars() {
        if c.is_whitespace() || c == '-' {
            start_of_word = true;
            prev_was_lowercase = false;
            continue;
        }

        if !c.is_alphabetic() {
            // Drop punctuation (apostrophes, commas, underscores, etc.)
            // without starting a new word.
            continue;
        }

        if start_of_word || (c.is_uppercase() && prev_was_lowercase) {
            acronym.extend(c.to_uppercase());
        }

        start_of_word = false;
        prev_was_lowercase = c.is_lowercase();
    }

    acronym
}

use acronym::abbreviate;

#[test]
fn empty() {
    assert_eq!(abbreviate(""), "");
}

#[test]
fn basic() {
    assert_eq!(abbreviate("Portable Network Graphics"), "PNG");
}

#[test]
fn lowercase_words() {
    assert_eq!(abbreviate("Ruby on Rails"), "ROR");
}

#[test]
fn camelcase() {
    assert_eq!(abbreviate("HyperText Markup Language"), "HTML");
}

#[test]
fn punctuation() {
    assert_eq!(abbreviate("First In, First Out"), "FIFO");
}

#[test]
fn all_caps_word() {
    assert_eq!(abbreviate("GNU Image Manipulation Program"), "GIMP");
}

#[test]
fn punctuation_without_whitespace() {
    assert_eq!(abbreviate("Complementary metal-oxide semiconductor"), "CMOS");
}

#[test]
fn very_long_abbreviation() {
    assert_eq!(
        abbreviate("Rolling On The Floor Laughing So Hard That My Dogs Came Over And Licked Me"),
        "ROTFLSHTMDCOALM"
    );
}

#[test]
fn consecutive_delimiters() {
    assert_eq!(abbreviate("Something - I made up from thin air"), "SIMUFTA");
}

#[test]
fn apostrophes() {
    assert_eq!(abbreviate("Halley's Comet"), "HC");
}

#[test]
fn underscore_emphasis() {
    assert_eq!(abbreviate("The Road _Not_ Taken"), "TRNT");
}

#[test]
fn prompt_examples() {
    assert_eq!(abbreviate("As Soon As Possible"), "ASAP");
    assert_eq!(abbreviate("Liquid-crystal display"), "LCD");
    assert_eq!(abbreviate("Thank George It's Friday!"), "TGIF");
}
EOF
