package trinary

import (
	"errors"
	"math"
)

// ParseTrinary converts a trinary string to its decimal value.
// Invalid input (empty, bad characters, or overflow) yields 0 and an error.
func ParseTrinary(arg string) (int64, error) {
	if arg == "" {
		return 0, errors.New("trinary: empty input")
	}
	var result int64
	for _, c := range arg {
		if c < '0' || c > '2' {
			return 0, errors.New("trinary: invalid digit")
		}
		d := int64(c - '0')
		if result > (math.MaxInt64-d)/3 {
			return 0, errors.New("trinary: value out of range")
		}
		result = result*3 + d
	}
	return result, nil
}
EOF
