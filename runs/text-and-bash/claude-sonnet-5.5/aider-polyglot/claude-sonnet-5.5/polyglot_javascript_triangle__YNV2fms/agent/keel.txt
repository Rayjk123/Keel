//
// Triangle classification exercise.

export class Triangle {
  constructor(...sides) {
    this.sides = sides;
    const [a, b, c] = sides;
    this.valid =
      sides.length === 3 &&
      sides.every((s) => typeof s === 'number' && Number.isFinite(s) && s > 0) &&
      a + b >= c &&
      b + c >= a &&
      a + c >= b;
  }

  get isEquilateral() {
    if (!this.valid) return false;
    const [a, b, c] = this.sides;
    return a === b && b === c;
  }

  get isIsosceles() {
    if (!this.valid) return false;
    const [a, b, c] = this.sides;
    return a === b || b === c || a === c;
  }

  get isScalene() {
    if (!this.valid) return false;
    const [a, b, c] = this.sides;
    return a !== b && b !== c && a !== c;
  }
}

I implemented the `Triangle` class in `triangle.js`:
- **Validation:** The constructor stores the sides and marks the triangle valid only if there are exactly three sides, each a finite number greater than 0, and the three triangle inequalities (`a + b ≥ c`, `b + c ≥ a`, `a + c ≥ b`) hold.
- **`isEquilateral`:** True when the triangle is valid and all three sides are equal.
- **`isIsosceles`:** True when the triangle is valid and at least two sides are equal.
- **`isScalene`:** True when the triangle is valid and all sides differ.
- **Invalid triangles:** All three getters return `false`.
