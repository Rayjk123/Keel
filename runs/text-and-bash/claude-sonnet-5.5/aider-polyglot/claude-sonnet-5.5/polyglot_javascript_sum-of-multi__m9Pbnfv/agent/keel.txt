//
// Sum Of Multiples
//

export const sum = (items = [], level = 0) => {
  const multiples = new Set();
  for (const base of items) {
    if (!(base > 0)) continue;
    for (let m = base; m < level; m += base) {
      multiples.add(m);
    }
  }
  let total = 0;
  for (const m of multiples) total += m;
  return total;
};
EOF
