#include <chrono>

namespace gigasecond {

boost::posix_time::ptime advance(const boost::posix_time::ptime& start) {
    constexpr std::chrono::seconds one_gigasecond{1'000'000'000};
    return start + boost::posix_time::seconds(one_gigasecond.count());
}

}  // namespace gigasecond
""")

#include "gigasecond.h"

int main() {
    using boost::posix_time::time_from_string;
    const auto check = [](const char* start, const char* expected) {

    };
    check("2011-04-25 00:00:00", "2043-01-01 01:46:40");
