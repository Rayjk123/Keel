pub fn abbreviate(phrase: &str) -> String {
    let mut acronym = String::new();
    let mut previous: Option<char> = None;

    for ch in phrase.chars() {
        if ch.is_whitespace() || ch == '-' {
            previous = None;
        } else if ch.is_alphanumeric() {
            let is_initial = match previous {
                None => true,
                Some(last) => last.is_lowercase() && ch.is_uppercase(),
            };
            if is_initial {
                acronym.extend(ch.to_uppercase());
            }
            previous = Some(ch);
        }
    }

    acronym
}

#[cfg(test)]
mod tests {
    use super::abbreviate;

    #[test]
    fn takes_uppercase_initials() {
        assert_eq!(abbreviate("Portable Network Graphics"), "PNG");
        assert_eq!(abbreviate("As Soon As Possible"), "ASAP");
        assert_eq!(abbreviate("Ruby on Rails"), "ROR");
        assert_eq!(abbreviate("GNU Image Manipulation Program"), "GIMP");
    }

    #[test]
    fn splits_on_whitespace_and_hyphens() {
        assert_eq!(abbreviate("\tLiquid--crystal\n display "), "LCD");
        assert_eq!(abbreviate(" - \t"), "");
    }

    #[test]
    fn ignores_other_punctuation() {
        assert_eq!(abbreviate("Thank George It's Friday!"), "TGIF");
        assert_eq!(abbreviate("_Non_ or not _so_ much"), "NONSM");
        assert_eq!(abbreviate("front.end controller"), "FC");
    }

    #[test]
    fn takes_initials_from_camel_case() {
        assert_eq!(abbreviate("HyperText Markup Language"), "HTML");
    }
}
