package hexadecimal

import (
	"errors"
	"math"
)

// ErrRange indicates that a value is out of range for the target type.
var ErrRange = errors.New("value out of range")

// ErrSyntax indicates that a value does not have the right syntax for the target type.
var ErrSyntax = errors.New("invalid syntax")

// ParseError records a failed conversion.
type ParseError struct {
	Num string // the input
	Err error  // the reason the conversion failed (ErrRange, ErrSyntax)
}

func (e *ParseError) Error() string {
	return "hexadecimal.ParseHex: parsing \"" + e.Num + "\": " + e.Err.Error()
}

// ParseHex interprets a hexadecimal string s and returns the corresponding int64.
// Empty strings and invalid digits produce a syntax error. Values that cannot
// be represented by an int64 produce a range error.
func ParseHex(s string) (int64, error) {
	if len(s) == 0 {
		return 0, &ParseError{Num: s, Err: ErrSyntax}
	}

	var n int64
	for i := 0; i < len(s); i++ {
		d := s[i]
		var v int64
		switch {
		case d >= '0' && d <= '9':
			v = int64(d - '0')
		case d >= 'a' && d <= 'f':
			v = int64(d-'a') + 10
		case d >= 'A' && d <= 'F':
			v = int64(d-'A') + 10
		default:
			return 0, &ParseError{Num: s, Err: ErrSyntax}
		}

		// Multiplying by 16 or adding the next digit would overflow int64.
		if n > (math.MaxInt64-v)/16 {
			return math.MaxInt64, &ParseError{Num: s, Err: ErrRange}
		}
		n = n*16 + v
	}
	return n, nil
}

// HandleErrors runs ParseHex on each input and reports "none", "syntax", or "range".
func HandleErrors(inputs []string) []string {
	results := make([]string, len(inputs))
	for i, s := range inputs {
		_, err := ParseHex(s)
		switch pe, ok := err.(*ParseError); {
		case err == nil:
			results[i] = "none"
		case ok && pe.Err == ErrSyntax:
			results[i] = "syntax"
		case ok && pe.Err == ErrRange:
			results[i] = "range"
		}
	}
	return results
}

package hexadecimal

import (
	"strings"
	"testing"
)

var testCases = []struct {
	in      string
	out     int64
	errCase string
}{
	{"1", 1, "none"},
	{"10", 0x10, "none"},
	{"2d", 0x2d, "none"},
	{"012", 0x12, "none"},
	{"cfcfcf", 0xcfcfcf, "none"},
	{"CFCFCF", 0xcfcfcf, "none"},
	{"", 0, "syntax"},
	{"peanut", 0, "syntax"},
	{"2cg134", 0, "syntax"},
	{"8000000000000000", 0, "range"},
	{"9223372036854775809", 0, "range"},
	{"7fffffffffffffff", 0x7fffffffffffffff, "none"},
	{"008000", 0x008000, "none"},
	{"008080", 0x008080, "none"},
	{"000080", 0x000080, "none"},
}

func TestParseHex(t *testing.T) {
	for _, test := range testCases {
		out, err := ParseHex(test.in)
		if test.errCase != "none" {
			var _ error = err
			if err == nil {
				t.Errorf("ParseHex(%q): expected an error, but error is nil", test.in)
				continue
			}
			if !strings.Contains(strings.ToLower(err.Error()), test.errCase) {
				t.Errorf("ParseHex(%q) returned error %q. Expected error containing %q.", test.in, err, test.errCase)
			}
		} else {
			if out != test.out {
				t.Errorf("ParseHex(%q) = %d. Expected %d.", test.in, out, test.out)
			}
			if err != nil {
				t.Errorf("ParseHex(%q) returned error %q.  Error not expected.", test.in, err)
			}
		}
	}
}

func TestHandleErrors(t *testing.T) {
	tests := make([]string, len(testCases))
	for i, test := range testCases {
		tests[i] = test.in
	}
	er := HandleErrors(tests)
	if len(er) != len(tests) {
		t.Fatalf("For %d tests, HandleErrors returned %d results, want %d", len(tests), len(er), len(tests))
	}
	for i, e := range er {
		if e != testCases[i].errCase {
			t.Errorf("For ParseHex(%q), HandleErrors reports %q, want %q", tests[i], e, testCases[i].errCase)
		}
	}
}
EOF
