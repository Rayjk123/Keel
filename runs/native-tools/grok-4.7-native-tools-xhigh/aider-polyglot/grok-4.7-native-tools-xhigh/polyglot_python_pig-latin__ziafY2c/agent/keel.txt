from pig_latin import translate

cases = [
    ("apple", "appleay"),
    ("xray", "xrayay"),
    ("yttria", "yttriaay"),
    ("pig", "igpay"),
    ("chair", "airchay"),
    ("thrush", "ushthray"),
    ("quick", "ickquay"),
    ("square", "aresquay"),
    ("my", "ymay"),
    ("rhythm", "ythmrhay"),
    ("hello world", "ellohay orldway"),
    ("queen", "eenquay"),
    ("yellow", "ellowyay"),
    ("school", "oolschay"),
    ("quiet", "ietquay"),
    ("the quick brown fox", "ethay ickquay ownbray oxfay"),
]
for src, expected in cases:
    got = translate(src)
    print(f"{'OK' if got == expected else 'FAIL'}: {src!r} -> {got!r} (expected {expected!r})")
EOF
`translate` in `pig_latin.py` now turns English phrases into Pig Latin word by word. It keeps vowel, `xr`, and `yt` starts and adds `ay`; otherwise it moves the leading consonants—including `qu` and consonants before `y`—to the end, then adds `ay`.
