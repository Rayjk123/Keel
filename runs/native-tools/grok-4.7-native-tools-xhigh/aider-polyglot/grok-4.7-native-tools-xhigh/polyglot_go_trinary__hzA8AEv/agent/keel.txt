package trinary

import "testing"

func TestParseTrinary(t *testing.T) {
	cases := []struct {
		in   string
		want int64
		ok   bool
	}{
		{"1", 1, true},
		{"2", 2, true},
		{"10", 3, true},
		{"11", 4, true},
		{"100", 9, true},
		{"112", 14, true},
		{"222", 26, true},
		{"1122000120", 32091, true},
		{"102012", 302, true},
		{"0", 0, true},
		{"", 0, false},
		{"3", 0, false},
		{"6", 0, false},
		{"9", 0, false},
		{"124578", 0, false},
		{"abc", 0, false},
		{" 1", 0, false},
		{"1 ", 0, false},
	}
	for _, tc := range cases {
		got, err := ParseTrinary(tc.in)
		if tc.ok && err != nil {
			t.Errorf("%q: unexpected error %v", tc.in, err)
		}
		if !tc.ok && err == nil {
			t.Errorf("%q: expected error", tc.in)
		}
		if got != tc.want {
			t.Errorf("%q: got %d want %d", tc.in, got, tc.want)
		}
	}
}
EOF
go test -v

`ParseTrinary` converts a trinary string to decimal by accumulating `result = result*3 + digit` for each `0`, `1`, or `2`. Invalid strings return `0` and an error.
