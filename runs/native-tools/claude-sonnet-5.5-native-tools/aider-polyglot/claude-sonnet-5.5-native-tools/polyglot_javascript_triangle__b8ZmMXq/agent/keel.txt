export class Triangle {
  constructor(...sides) {
    this.sides = sides;
    const [a, b, c] = sides;
    this.valid =
      sides.length === 3 &&
      sides.every((s) => typeof s === 'number' && s > 0) &&
      a + b >= c &&
      b + c >= a &&
      a + c >= b;
  }

  get isEquilateral() {
    if (!this.valid) return false;
    const [a, b, c] = this.sides;
    return a === b && b === c;
  }

  get isIsosceles() {
    if (!this.valid) return false;
    const [a, b, c] = this.sides;
    return a === b || b === c || a === c;
  }

  get isScalene() {
    if (!this.valid) return false;
    return !this.isIsosceles;
  }
}
EOF

I implemented `Triangle` in `triangle.js`. The class and getter names are unchanged: `isEquilateral`, `isIsosceles` and `isScalene`.

- **Validity:** A triangle needs exactly three sides. Each must be a number greater than 0, and all three triangle inequalities must hold. The equal case counts, so a degenerate triangle like 1, 2, 3 is valid. An invalid triangle returns `false` for all three checks.
- **Equilateral:** All three sides are equal.
- **Isosceles:** At least two sides are equal.
- **Scalene:** The triangle is valid and not isosceles, so all sides differ.

I checked six side sets by hand with a short script, and the results were as expected. They were 2,2,2, 3,3,4, 3,4,5, 0,0,0, 1,1,3 and 1,2,3. No unit tests were in the folder, so I didn't run any.
