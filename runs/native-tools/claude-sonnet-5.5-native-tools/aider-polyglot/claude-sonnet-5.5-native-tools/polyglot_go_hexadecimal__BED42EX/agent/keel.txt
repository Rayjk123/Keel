package hexadecimal

import "errors"

// ErrSyntax is returned for strings that are not valid hexadecimal.
var ErrSyntax = errors.New("invalid hexadecimal syntax")

// ErrRange is returned when the value does not fit in an int64.
var ErrRange = errors.New("hexadecimal value out of range")

// ParseHex converts a hexadecimal string to its decimal value.
func ParseHex(in string) (int64, error) {
	if len(in) == 0 {
		return 0, ErrSyntax
	}
	const max = int64(^uint64(0) >> 1)
	var out int64
	for _, c := range in {
		var d int64
		switch {
		case c >= '0' && c <= '9':
			d = int64(c - '0')
		case c >= 'a' && c <= 'f':
			d = int64(c-'a') + 10
		case c >= 'A' && c <= 'F':
			d = int64(c-'A') + 10
		default:
			return 0, ErrSyntax
		}
		if out > (max-d)/16 {
			return 0, ErrRange
		}
		out = out*16 + d
	}
	return out, nil
}
EOF

- **Conversion:** it reads each digit by hand (`0-9`, `a-f`, `A-F`) and multiplies the running total by 16 per digit, with no parsing libraries.
- **Invalid input:** an empty string or any other character returns `ErrSyntax`.
- **Overflow:** a value too large for an `int64` returns `ErrRange`.
- **Test cases:** the throwaway test covered `1`, `10af8c`, `FF`, `008080`, `7fffffffffffffff`, and the failures `""`, `g`, `1z` and `8000000000000000`.
