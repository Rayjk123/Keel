export const sum = (bases, level) => {
  const seen = new Set();
  for (const b of bases) {
    if (b <= 0) continue;
    for (let m = b; m < level; m += b) seen.add(m);
  }
  let total = 0;
  for (const n of seen) total += n;
  return total;
};
