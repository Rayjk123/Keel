import re

_RULE1 = re.compile(r"^(?:[aeiou]|xr|yt)")
_RULE3 = re.compile(r"^([^aeiou]*qu)(.*)$")
_RULE2 = re.compile(r"^([^aeiouy]+)(.*)$")


def _word(word):
    if _RULE1.match(word):
        return word + "ay"
    m = _RULE3.match(word)
    if m:
        return m.group(2) + m.group(1) + "ay"
    m = _RULE2.match(word)
    if m:
        return m.group(2) + m.group(1) + "ay"
    return word + "ay"


def translate(text):
    return " ".join(_word(w) for w in text.split())
