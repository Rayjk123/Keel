#include "gigasecond.h"
#include <boost/date_time/posix_time/posix_time.hpp>
#include <cassert>
#include <chrono>
#include <type_traits>

int main() {
    using namespace boost::gregorian;
    using namespace boost::posix_time;
    assert(gigasecond::advance(ptime(date(2015, 1, 24), hours(22))) ==
           ptime(date(2046, 10, 2), hours(23) + minutes(46) + seconds(40)));
    assert(gigasecond::advance(ptime(date(2011, 4, 25))) ==
           ptime(date(2043, 1, 1), hours(1) + minutes(46) + seconds(40)));
    assert(gigasecond::advance(ptime(date(2015, 1, 24), hours(22) + milliseconds(123))) ==
           ptime(date(2046, 10, 2), hours(23) + minutes(46) + seconds(40) + milliseconds(123)));
    using namespace std::chrono;
    const system_clock::time_point start{milliseconds{123}};
    assert(gigasecond::advance(start) - start == std::chrono::seconds{1000000000});
    const time_point<system_clock, milliseconds> precise{milliseconds{123}};
    assert(gigasecond::advance(precise).time_since_epoch() == milliseconds{1000000000123LL});
}
EOF
