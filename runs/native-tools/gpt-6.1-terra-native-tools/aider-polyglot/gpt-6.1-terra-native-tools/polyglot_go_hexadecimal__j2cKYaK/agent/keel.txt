package hexadecimal

import "testing"

func TestParseHex(t *testing.T) {
	valid := map[string]int64{
		"0": 0, "008000": 32768, "008080": 32896, "000080": 128,
		"10af8c": 1093516, "ABCDEF": 11259375,
		"7fffffffffffffff": 1<<63 - 1, "00000000000000000001": 1,
	}
	for input, want := range valid {
		got, err := ParseHex(input)
		if err != nil || got != want { t.Errorf("ParseHex(%q) = %d, %v; want %d", input, got, err, want) }
	}
	for _, input := range []string{"", "g", "1g", "0x10", "-1", " 1", "é", "8000000000000000", "ffffffffffffffff"} {
		if got, err := ParseHex(input); err == nil { t.Errorf("ParseHex(%q) = %d, want error", input, got) }
	}
}
EOF
