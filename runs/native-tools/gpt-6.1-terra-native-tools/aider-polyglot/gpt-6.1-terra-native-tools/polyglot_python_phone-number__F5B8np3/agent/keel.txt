from phone_number import PhoneNumber

for raw in ('+1 (613)-995-0253', '613-995-0253', '1 613 995 0253', '613.995.0253'):
    phone = PhoneNumber(raw)
    assert phone.number == '6139950253'
    assert phone.area_code == '613'
    assert phone.pretty() == '(613) 995-0253'

cases = {
    '123456789': 'must not be fewer than 10 digits',
    '123456789012': 'must not be greater than 11 digits',
    '22345678901': '11 digits must start with 1',
    '0234567890': 'area code cannot start with zero',
    '1234567890': 'area code cannot start with one',
    '2230567890': 'exchange code cannot start with zero',
    '2231567890': 'exchange code cannot start with one',
    '613-995-025x': 'letters not permitted',
    '613-995-025@': 'punctuations not permitted',
}
for raw, message in cases.items():
    try:
        PhoneNumber(raw)
    except ValueError as exc:
        assert str(exc) == message, (raw, exc)
    else:
        raise AssertionError(f'{raw!r} was accepted')
print('All phone number checks passed.')
