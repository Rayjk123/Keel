package hexadecimal

import (
	"testing"
)

func TestParseHex(t *testing.T) {
	tests := []struct {
		input    string
		expected int64
		hasError bool
	}{
		{"0", 0, false},
		{"1", 1, false},
		{"a", 10, false},
		{"A", 10, false},
		{"f", 15, false},
		{"F", 15, false},
		{"10", 16, false},
		{"ff", 255, false},
		{"FF", 255, false},
		{"10af8c", 1093580, false},
		{"008000", 32768, false}, // green
		{"008080", 32896, false}, // teal
		{"000080", 128, false},   // navy
		{"", 0, true},            // empty string
		{"g", 0, true},           // invalid char
		{"1g", 0, true},          // invalid char
		{"xyz", 0, true},         // invalid chars
	}

	for _, test := range tests {
		result, err := ParseHex(test.input)
		if test.hasError {
			if err == nil {
				t.Errorf("ParseHex(%q) expected error, got nil", test.input)
			}
		} else {
			if err != nil {
				t.Errorf("ParseHex(%q) unexpected error: %v", test.input, err)
			}
			if result != test.expected {
				t.Errorf("ParseHex(%q) = %d, expected %d", test.input, result, test.expected)
			}
		}
	}
}
EOF
