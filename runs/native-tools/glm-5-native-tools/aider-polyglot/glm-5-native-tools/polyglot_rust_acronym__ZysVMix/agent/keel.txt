pub fn abbreviate(phrase: &str) -> String {
    phrase
        .split(|c: char| c.is_whitespace() || c == '-')
        .filter(|word| !word.is_empty())
        .filter_map(|word| word.chars().find(|c| c.is_alphanumeric()))
        .collect::<String>()
        .to_uppercase()
}

fn main() {
    // Test cases from the instructions
    assert_eq!(abbreviate("As Soon As Possible"), "ASAP");
    assert_eq!(abbreviate("Liquid-crystal display"), "LCD");
    assert_eq!(abbreviate("Thank George It's Friday!"), "TGIF");
    
    // Additional test cases
    assert_eq!(abbreviate("Portable Network Graphics"), "PNG");
    assert_eq!(abbreviate("Three Letter Acronyms"), "TLA");
    
    println!("All tests passed!");
}
