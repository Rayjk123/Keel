package main

import (
	"fmt"
	"trinary"
)

func main() {
	// Test case from the problem: "102012" should equal 302
	result, ok := trinary.ParseTrinary("102012")
	fmt.Printf("102012 -> %d, %v (expected 302, true)\n", result, ok)
	
	// Test valid inputs
	testCases := map[string]int64{
		"0":       0,
		"1":       1,
		"2":       2,
		"10":      3,
		"11":      4,
		"12":      5,
		"20":      6,
		"21":      7,
		"22":      8,
		"100":     9,
		"102012":  302,
	}
	
	for input, expected := range testCases {
		result, ok := trinary.ParseTrinary(input)
		if !ok || result != expected {
			fmt.Printf("FAIL: %s -> %d, %v (expected %d, true)\n", input, result, ok, expected)
		} else {
			fmt.Printf("PASS: %s -> %d\n", input, result)
		}
	}
	
	// Test invalid inputs (should return 0, false)
	invalidInputs := []string{"3", "10201b", "abc", "", "123"}
	for _, input := range invalidInputs {
		result, ok := trinary.ParseTrinary(input)
		if ok || result != 0 {
			fmt.Printf("FAIL: invalid input %q should return (0, false), got (%d, %v)\n", input, result, ok)
		} else {
			fmt.Printf("PASS: invalid input %q correctly returns (0, false)\n", input)
		}
	}
}
