#include <iostream>
#include <boost/date_time/posix_time/posix_time.hpp>

boost::posix_time::ptime advance(const boost::posix_time::ptime& time) {
    return time + boost::posix_time::seconds(1000000000LL);
}

int main() {
    // Test case from the problem: January 24th, 2015 at 22:00 (10:00:00pm)
    // should give October 2nd, 2046 at 23:46:40 (11:46:40pm)
    boost::posix_time::ptime birth_time(
        boost::gregorian::date(2015, 1, 24),
        boost::posix_time::time_duration(22, 0, 0)
    );
    
    boost::posix_time::ptime gigasecond_old = advance(birth_time);
    
    std::cout << "Birth time: " << birth_time << std::endl;
    std::cout << "Gigasecond old: " << gigasecond_old << std::endl;
    
    // Expected: 2046-Oct-02 23:46:40
    boost::posix_time::ptime expected(
        boost::gregorian::date(2046, 10, 2),
        boost::posix_time::time_duration(23, 46, 40)
    );
    
    if (gigasecond_old == expected) {
        std::cout << "Test PASSED!" << std::endl;
        return 0;
    } else {
        std::cout << "Test FAILED!" << std::endl;
        std::cout << "Expected: " << expected << std::endl;
        return 1;
    }
}
EOF
