package trinary

import (
	"testing"
)

func TestParseTrinary(t *testing.T) {
	tests := []struct {
		input string
		want  int64
		ok    bool
	}{
		{"1", 1, true},
		{"2", 2, true},
		{"10", 3, true},
		{"11", 4, true},
		{"100", 9, true},
		{"112", 14, true},
		{"222", 26, true},
		{"1122000120", 32091, true},
		{"102012", 302, true},
		{"0", 0, true},
		{"", 0, false},
		{"3", 0, false},
		{"6", 0, false},
		{"7", 0, false},
		{"8", 0, false},
		{"9", 0, false},
		{"123", 0, false},
		{"a", 0, false},
		{"carrot", 0, false},
		{"102012a", 0, false},
	}
	for _, tt := range tests {
		got, err := ParseTrinary(tt.input)
		ok := err == nil
		if got != tt.want || ok != tt.ok {
			t.Errorf("ParseTrinary(%q) = (%d, %v), want (%d, ok=%v)", tt.input, got, err, tt.want, tt.ok)
		}
	}
}
EOF

The `ParseTrinary` function now converts a trinary string to decimal by walking each digit and accumulating `result*3 + digit`. Invalid input (empty string, non-`0`/`1`/`2` digits, or overflow) returns `0` and an error.
