$ cat > /tmp/hex_test.go << 'EOF'
package hexadecimal

import "testing"

func TestParseHex(t *testing.T) {
	cases := []struct {
		in   string
		want int64
		err  bool
	}{
		{"1", 1, false},
		{"10", 16, false},
		{"10af8c", 1093516, false},
		{"10AF8C", 1093516, false},
		{"c", 12, false},
		{"a", 10, false},
		{"f", 15, false},
		{"A", 10, false},
		{"F", 15, false},
		{"abcdef", 11259375, false},
		{"ABCDEF", 11259375, false},
		{"0", 0, false},
		{"00", 0, false},
		{"008000", 32768, false},
		{"008080", 32896, false},
		{"000080", 128, false},
		{"ffffff", 16777215, false},
		{"ffffffff", 4294967295, false},
		{"7fffffffffffffff", 9223372036854775807, false},
		{"8000000000000000", 0, true},
		{"ffffffffffffffff", 0, true},
		{"", 0, true},
		{"g", 0, true},
		{"10af8g", 0, true},
		{"0x10", 0, true},
		{" 1", 0, true},
		{"1 ", 0, true},
		{"-1", 0, true},
	}
	for _, tc := range cases {
		got, err := ParseHex(tc.in)
		if tc.err {
			if err == nil {
				t.Errorf("ParseHex(%q) expected error, got %d", tc.in, got)
			}
		} else {
			if err != nil {
				t.Errorf("ParseHex(%q) unexpected error: %v", tc.in, err)
			} else if got != tc.want {
				t.Errorf("ParseHex(%q) = %d, want %d", tc.in, got, tc.want)
			}
		}
	}
}
EOF
cp /tmp/hex_test.go ./hexadecimal_test.go && go test -v . && rm ./hexadecimal_test.go
`ParseHex` now converts a hexadecimal string to an `int64` by hand: each character is mapped to a digit (`0-9`, `a-f`, `A-F`) and accumulated as `result * 16 + digit`. Empty input, invalid characters, and values that overflow `int64` return an error.
