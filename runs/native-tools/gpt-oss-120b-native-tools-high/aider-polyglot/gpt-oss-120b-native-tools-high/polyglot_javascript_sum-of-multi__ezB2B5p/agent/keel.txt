import { sum } from './sum-of-multiples.js';
function naive(multiples, limit){
  const s = new Set();
  for(let i=1;i<limit;i++){
    for(const m of multiples){
      if(m>0 && i % m ===0) s.add(i);
    }
  }
  let total=0;
  for(const val of s) total+=val;
  return total;
}
const arr = [3,5,6];
const limit = 100;
console.log('our sum:', sum(arr, limit));
console.log('naive sum:', naive(arr, limit));

import { sum } from './sum-of-multiples.js';
function naive(multiples, limit) {
  const set = new Set();
  for (let i=1; i<limit; i++) {
    for (const m of multiples) {
      if (typeof m === 'number' && m>0 && i % m === 0) {
        set.add(i);
        break; // once added, break; but not needed
      }
    }
  }
  let total = 0;
  for (const v of set) total += v;
  return total;
}
function testRandom() {
  for (let t=0; t<100; t++) {
    const len = Math.floor(Math.random()*5);
    const arr = [];
    for (let i=0; i<len; i++) {
      const val = Math.floor(Math.random()*20) - 5; // can be negative or zero
      arr.push(val);
    }
    const limit = Math.floor(Math.random()*50) + 1; // 1 to 50
    const a = sum(arr, limit);
    const b = naive(arr, limit);
    if (a !== b) {
      console.error('Mismatch', arr, limit, a, b);
      return false;
    }
  }
  console.log('All random tests passed');
  return true;
}

testRandom();
