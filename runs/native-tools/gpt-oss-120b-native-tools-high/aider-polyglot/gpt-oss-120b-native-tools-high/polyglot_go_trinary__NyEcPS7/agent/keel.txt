package trinary

import "testing"

func TestParseTrinaryValid(t *testing.T) {
    cases := []struct {
        input    string
        expected int64
        ok       bool
    }{
        {"", 0, true},
        {"0", 0, true},
        {"1", 1, true},
        {"2", 2, true},
        {"10", 3, true},
        {"102", 11, true},
        {"21201", 500, true}, // verify: 2*3^4=162,1*27=27,2*9=18,0*3=0,1*1=1 tot=162+27+18+0+1=208? That seems wrong.
    }
    for _, c := range cases {
        got, ok := ParseTrinary(c.input)
        if got != c.expected || ok != c.ok {
            t.Errorf("ParseTrinary(%q) = (%d,%v), expected (%d,%v)", c.input, got, ok, c.expected, c.ok)
        }
    }
}

func TestParseTrinaryInvalid(t *testing.T) {
    invalid := []string{"3", "1023", "abc", "1a2", "-1"}
    for _, s := range invalid {
        got, ok := ParseTrinary(s)
        if ok != false || got != 0 {
            t.Errorf("ParseTrinary(%q) = (%d,%v), expected (0,false)", s, got, ok)
        }
    }
}
EOF
