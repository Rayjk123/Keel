package hexadecimal

import (
	"errors"
	"unicode"
)

// ParseHex converts a hexadecimal string to its decimal equivalent.
// Returns the decimal value and an empty string on success,
// or 0 and an error message on failure.
func ParseHex(in string) (int64, string) {
	if len(in) == 0 {
		return 0, "empty string"
	}
	
	var result int64
	for _, c := range in {
		result *= 16
		switch {
		case c >= '0' && c <= '9':
			result += int64(c - '0')
		case c >= 'a' && c <= 'f':
			result += int64(c - 'a' + 10)
		case c >= 'A' && c <= 'F':
			result += int64(c - 'A' + 10)
		default:
			return 0, "invalid character"
		}
	}
	return result, ""
}
