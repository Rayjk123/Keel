fn abbreviate(phrase: &str) -> String {
    let mut acronym = String::new();
    let mut prev_char: Option<char> = None;
    
    for c in phrase.chars() {
        // Check if this character is the start of a word
        let is_word_start = match prev_char {
            None => true, // First character of the string
            Some(prev) => {
                // Word starts after whitespace, hyphen, or other separator
                prev.is_whitespace() || prev == '-' || prev == '_'
            }
        };
        
        // Add to acronym if:
        // 1. It's a letter and starts a word, OR
        // 2. It's an uppercase letter and the previous was lowercase (camelCase)
        if c.is_alphabetic() {
            if is_word_start {
                acronym.push(c.to_ascii_uppercase());
            } else if let Some(prev) = prev_char {
                // Handle camelCase: uppercase letter after lowercase
                if c.is_uppercase() && prev.is_lowercase() {
                    acronym.push(c.to_ascii_uppercase());
                }
            }
        }
        
        prev_char = Some(c);
    }
    
    acronym
}

fn main() {
    println!("Portable Network Graphics -> {}", abbreviate("Portable Network Graphics"));
    println!("Liquid-crystal display -> {}", abbreviate("Liquid-crystal display"));
    println!("Thank George It's Friday! -> {}", abbreviate("Thank George It's Friday!"));
    println!("As Soon As Possible -> {}", abbreviate("As Soon As Possible"));
    println!("HyperText Markup Language -> {}", abbreviate("HyperText Markup Language"));
}
