import java.util.List;

public class Smoke {
    static int failures = 0;

    static void check(String label, Object actual, Object expected) {
        boolean ok = expected.equals(actual);
        if (!ok) failures++;
        System.out.println((ok ? "PASS " : "FAIL ") + label + " -> " + actual + (ok ? "" : "   (expected " + expected + ")"));
    }

    static void checkThrows(String label, Runnable r, String expectedMsg) {
        try {
            r.run();
            failures++;
            System.out.println("FAIL " + label + " -> no exception");
        } catch (IllegalArgumentException e) {
            boolean ok = expectedMsg.equals(e.getMessage());
            if (!ok) failures++;
            System.out.println((ok ? "PASS " : "FAIL ") + label + " -> IAE(\"" + e.getMessage() + "\")");
        }
    }

    public static void main(String[] args) {
        // Examples from the instructions
        check("49142 / 3", new Series("49142").slices(3), List.of("491", "914", "142"));
        check("49142 / 4", new Series("49142").slices(4), List.of("4914", "9142"));
        // Edge cases
        check("1 / 1", new Series("1").slices(1), List.of("1"));
        check("12 / 1", new Series("12").slices(1), List.of("1", "2"));
        check("35 / 2 (whole string)", new Series("35").slices(2), List.of("35"));
        check("9142 / 2", new Series("9142").slices(2), List.of("91", "14", "42"));
        check("777777 / 3 (duplicates kept)", new Series("777777").slices(3), List.of("777", "777", "777", "777"));
        check("918493904243 / 5 (leading zero kept)", new Series("918493904243").slices(5),
              List.of("91849", "18493", "84939", "49390", "93904", "39042", "90424", "04243"));
        // Same Series object can be sliced repeatedly with different sizes
        Series s = new Series("49142");
        check("reuse / 5", s.slices(5), List.of("49142"));
        check("reuse / 1", s.slices(1), List.of("4", "9", "1", "4", "2"));
        // Error cases ("6-digit series from a 5-digit string")
        checkThrows("12345 / 6", () -> new Series("12345").slices(6), "Slice size is too big.");
        checkThrows("12345 / 42", () -> new Series("12345").slices(42), "Slice size is too big.");
        checkThrows("12345 / MAX_VALUE", () -> new Series("12345").slices(Integer.MAX_VALUE), "Slice size is too big.");
        checkThrows("12345 / 0", () -> new Series("12345").slices(0), "Slice size is too small.");
        checkThrows("123 / -1", () -> new Series("123").slices(-1), "Slice size is too small.");
        checkThrows("123 / MIN_VALUE", () -> new Series("123").slices(Integer.MIN_VALUE), "Slice size is too small.");
        checkThrows("empty series", () -> new Series(""), "series cannot be empty");

        System.out.println(failures == 0 ? "\nALL SMOKE CHECKS PASSED" : "\n" + failures + " FAILURE(S)");
        System.exit(failures == 0 ? 0 : 1);
    }
}
