from pig_latin import translate

cases = {
    # Every example given in the instructions
    "apple": "appleay", "xray": "xrayay", "yttria": "yttriaay",
    "pig": "igpay", "chair": "airchay", "thrush": "ushthray",
    "quick": "ickquay", "square": "aresquay",
    "my": "ymay", "rhythm": "ythmrhay",
    # Small or unusual words
    "": "", "a": "aay", "y": "yay", "q": "qay", "qu": "quay",
    "squ": "squay", "nth": "nthay", "aqua": "aquaay", "squeal": "ealsquay",
    # Capitalisation is kept
    "Pig": "Igpay", "PIG": "IGPAY", "Queen": "Eenquay", "SQUARE": "ARESQUAY",
    "Xray": "Xrayay", "Rhythm": "Ythmrhay", "I": "Iay", "OK": "OKAY",
    # Whitespace, punctuation, contractions and hyphens are kept
    "  pig  latin  ": "  igpay  atinlay  ",
    "Hello, world!": "Ellohay, orldway!",
    "don't stop": "on'tday opstay",
    "don\u2019t": "on\u2019tday",
    "well-known": "ellway-ownknay",
    "line one\nline two\t3": "inelay oneay\ninelay otway\t3",
    "quick fast run": "ickquay astfay unray",
}

failures = 0
for given, expected in cases.items():
    actual = translate(given)
    status = "ok  " if actual == expected else "FAIL"
    failures += actual != expected
    print(f"{status} {given!r:24} -> {actual!r}" + ("" if actual == expected else f"   (expected {expected!r})"))
print(f"\n{len(cases) - failures}/{len(cases)} passed")
EOF
