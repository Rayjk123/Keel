use acronym::abbreviate;

#[test]
fn empty() {
    assert_eq!(abbreviate(""), "");
}

#[test]
fn basic() {
    assert_eq!(abbreviate("Portable Network Graphics"), "PNG");
}

#[test]
fn lowercase_words() {
    assert_eq!(abbreviate("Ruby on Rails"), "ROR");
}

#[test]
fn camelcase() {
    assert_eq!(abbreviate("HyperText Markup Language"), "HTML");
}

#[test]
fn punctuation() {
    assert_eq!(abbreviate("First In, First Out"), "FIFO");
}

#[test]
fn all_caps_word() {
    assert_eq!(abbreviate("GNU Image Manipulation Program"), "GIMP");
}

#[test]
fn all_caps_word_with_punctuation() {
    assert_eq!(abbreviate("PHP: Hypertext Preprocessor"), "PHP");
}

#[test]
fn punctuation_without_whitespace() {
    assert_eq!(abbreviate("Complementary metal-oxide semiconductor"), "CMOS");
}

#[test]
fn very_long_abbreviation() {
    assert_eq!(
        abbreviate("Rolling On The Floor Laughing So Hard That My Dogs Came Over And Licked Me"),
        "ROTFLSHTMDCOALM"
    );
}

#[test]
fn consecutive_delimiters() {
    assert_eq!(abbreviate("Something - I made up from thin air"), "SIMUFTA");
}

#[test]
fn apostrophes() {
    assert_eq!(abbreviate("Halley's Comet"), "HC");
}

#[test]
fn underscore_emphasis() {
    assert_eq!(abbreviate("The Road _Not_ Taken"), "TRNT");
}

// Examples from the instructions.
#[test]
fn instruction_examples() {
    assert_eq!(abbreviate("As Soon As Possible"), "ASAP");
    assert_eq!(abbreviate("Liquid-crystal display"), "LCD");
    assert_eq!(abbreviate("Thank George It's Friday!"), "TGIF");
}

// Extra edge cases.
#[test]
fn edge_cases() {
    assert_eq!(abbreviate("   "), "");
    assert_eq!(abbreviate("---"), "");
    assert_eq!(abbreviate("!!! ?? ..."), "");
    assert_eq!(abbreviate("Rock & Roll"), "RR");
    assert_eq!(abbreviate("  leading and trailing  "), "LAT");
    assert_eq!(abbreviate("tabs\tand\nnewlines"), "TAN");
    assert_eq!(abbreviate("ab-cd--ef"), "ACE");
    assert_eq!(abbreviate("-leading-hyphen"), "LH");
    assert_eq!(abbreviate("GNU's Not Unix"), "GNU");
    assert_eq!(abbreviate("O'Neil"), "O");
    assert_eq!(abbreviate("McDonald's"), "MD");
    assert_eq!(abbreviate("Hyper_Text"), "HT");
    assert_eq!(abbreviate("Mozilla 3D Engine"), "M3E");
    assert_eq!(abbreviate("x"), "X");
    // Unicode hyphens/dashes and whitespace
    assert_eq!(abbreviate("Liquid\u{2010}crystal display"), "LCD");
    assert_eq!(abbreviate("Liquid\u{2014}crystal display"), "LCD");
    assert_eq!(abbreviate("non\u{a0}breaking space"), "NBS");
    // Soft hyphen is not a separator: it's dropped like other punctuation
    assert_eq!(abbreviate("Liq\u{ad}uid crystal"), "LC");
    // Non-ASCII letters
    assert_eq!(abbreviate("élan vital"), "ÉV");
    assert_eq!(abbreviate("Über alles"), "ÜA");
}

The task is finished. `abbreviate` in `src/lib.rs` now turns a phrase into its acronym:

- **Splitting:** words are separated by whitespace and hyphens, including Unicode dashes.
- **Punctuation:** everything else is ignored, so "It's" stays one word.
- **Letters used:** each word gives its first letter or digit. A capital right after a lowercase letter also counts, so "HyperText" gives "HT".
- **Case:** the result is uppercased.
