cat > extra_check.cpp <<'EOF'
#include "gigasecond.h"
#include <boost/date_time/posix_time/posix_time.hpp>
#include <fstream>
#include <iostream>
#include <string>
using namespace boost::posix_time;
using namespace boost::gregorian;

int main() {
    int bad = 0;
    // 1) random cases vs Python oracle (includes microseconds, leap years, century rules)
    std::ifstream in("cases.txt");
    std::string d1, t1, d2, t2; long n = 0;
    while (in >> d1 >> t1 >> d2 >> t2) {
        ++n;
        const ptime got = gigasecond::advance(time_from_string(d1 + " " + t1));
        const ptime want = time_from_string(d2 + " " + t2);
        if (got != want) { if (++bad < 5) std::cerr << "MISMATCH " << d1 << " " << t1 << " got " << got << " want " << want << "\n"; }
    }
    std::cout << "random cases checked: " << n << ", mismatches: " << bad << "\n";

    // 2) fractional seconds are preserved
    const ptime frac = time_from_string("2015-01-24 22:00:00.123456");
    const ptime fr = gigasecond::advance(frac);
    std::cout << "fractional: " << fr << "\n";
    if (fr.time_of_day().fractional_seconds() != frac.time_of_day().fractional_seconds()) { ++bad; std::cerr << "fraction lost\n"; }

    // 3) exact difference is 10^9 seconds
    const ptime s = time_from_string("2015-01-24 22:00:00");
    if ((gigasecond::advance(s) - s).total_seconds() != 1000000000) { ++bad; std::cerr << "diff wrong\n"; }

    // 4) special values are passed through unchanged
    if (!gigasecond::advance(ptime(not_a_date_time)).is_not_a_date_time()) { ++bad; std::cerr << "nadt\n"; }
    if (!gigasecond::advance(ptime(pos_infin)).is_pos_infinity())          { ++bad; std::cerr << "+inf\n"; }
    if (!gigasecond::advance(ptime(neg_infin)).is_neg_infinity())          { ++bad; std::cerr << "-inf\n"; }

    // 5) the example from the instructions
    const ptime ex = gigasecond::advance(ptime(date(2015, Jan, 24), hours(22)));
    std::cout << "example: " << ex << "\n";
    if (ex != ptime(date(2046, Oct, 2), time_duration(23, 46, 40))) { ++bad; std::cerr << "example wrong\n"; }

    std::cout << (bad ? "FAILED" : "ALL EXTRA CHECKS PASSED") << "\n";
    return bad ? 1 : 0;
}
