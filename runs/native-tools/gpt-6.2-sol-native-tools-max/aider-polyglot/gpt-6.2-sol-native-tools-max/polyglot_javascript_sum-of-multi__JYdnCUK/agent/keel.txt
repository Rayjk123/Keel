import assert from 'node:assert/strict';
import { sum } from './sum-of-multiples.js';

assert.equal(sum([3, 5], 20), 78);
assert.equal(sum([3, 5], 1000), 233168);
assert.equal(sum([3], 6), 3);
assert.equal(sum([20, 21], 20), 0);
assert.equal(sum([], 20), 0);
assert.equal(sum([0], 20), 0);
assert.equal(sum([0, 3, 3, 6], 7), 9);
assert.equal(sum([1], 1), 0);

for (let limit = 0; limit < 50; limit++) {
  for (let a = 0; a < 10; a++) {
    for (let b = 0; b < 10; b++) {
      let expected = 0;
      for (let i = 1; i < limit; i++) {
        if (i % a === 0 || i % b === 0) expected += i;
      }
      assert.equal(sum([a, b], limit), expected);
    }
  }
}
console.log('All sum-of-multiples checks passed.');
