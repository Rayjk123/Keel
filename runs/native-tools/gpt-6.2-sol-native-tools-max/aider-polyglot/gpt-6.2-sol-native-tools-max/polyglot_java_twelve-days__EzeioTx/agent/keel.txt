import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Arrays;

class TwelveDaysCheck {
    public static void main(String[] args) throws Exception {
        String expectedSong = Files.readString(Path.of(args[0]));
        TwelveDays song = new TwelveDays();
        check("full song", expectedSong, song.sing());
        String[] expectedVerses = expectedSong.stripTrailing().split("\n\n");
        for (int first = 1; first <= expectedVerses.length; first++) {
            check("verse " + first, expectedVerses[first - 1] + "\n", song.verse(first));
            for (int last = first; last <= expectedVerses.length; last++) {
                String expectedRange = String.join("\n\n", Arrays.copyOfRange(expectedVerses, first - 1, last));
                check("verses " + first + " through " + last, expectedRange + "\n", song.verses(first, last));
            }
        }
        System.out.println("Full song, all 12 individual verses, and all 78 inclusive ranges match.");
    }

    private static void check(String label, String expected, String actual) {
        if (!expected.equals(actual)) {
            throw new AssertionError(label + " did not match.\nExpected: " + expected + "\nActual: " + actual);
        }
    }
}
EOF
