import assert from 'node:assert/strict';
import { Triangle } from './triangle.js';

const cases = [
  [[3, 3, 3], [true, true, false]],
  [[0.5, 0.5, 0.5], [true, true, false]],
  [[4, 4, 3], [false, true, false]],
  [[4, 3, 4], [false, true, false]],
  [[3, 4, 4], [false, true, false]],
  [[3, 4, 5], [false, false, true]],
  [[0.5, 0.4, 0.6], [false, false, true]],
  [[2, 2, 4], [false, true, false]],
  [[2, 5, 3], [false, false, true]],
  [[0, 0, 0], [false, false, false]],
  [[-1, 2, 2], [false, false, false]],
  [[7, 3, 3], [false, false, false]],
  [[3, 7, 3], [false, false, false]],
  [[3, 3, 7], [false, false, false]],
  [[Infinity, Infinity, Infinity], [false, false, false]],
  [[1, 1, NaN], [false, false, false]],
  [['1', 1, 1], [false, false, false]],
  [[1, 1], [false, false, false]],
  [[1, 1, 1, 1], [false, false, false]],
];

for (const [sides, expected] of cases) {
  const { isEquilateral, isIsosceles, isScalene } = new Triangle(...sides);
  assert.deepEqual([isEquilateral, isIsosceles, isScalene], expected, `sides: ${sides.join(', ')}`);
}
console.log(`Passed ${cases.length} triangle classification cases.`);
