import java.util.List;

class SeriesCheck {
    public static void main(String[] args) {
        Series series = new Series("49142");
        expect(List.of("491", "914", "142"), series.slices(3));
        expect(List.of("4914", "9142"), series.slices(4));
        expect(List.of("49142"), series.slices(5));
        expect(List.of("4", "9", "1", "4", "2"), series.slices(1));
        expect(List.of("01", "10", "01"), new Series("0101").slices(2));
        expectInvalid(series, 6, "Slice size is too big.");
        expectInvalid(series, 0, "Slice size is too small.");
        expectInvalid(series, -1, "Slice size is too small.");
        expectInvalid(new Series(""), 1, "Slice size is too big.");
        System.out.println("All Series checks passed.");
    }

    private static void expect(List<String> expected, List<String> actual) {
        if (!expected.equals(actual)) {
            throw new AssertionError("Expected " + expected + ", got " + actual);
        }
    }

    private static void expectInvalid(Series series, int size, String message) {
        try {
            series.slices(size);
            throw new AssertionError("Expected an exception for slice size " + size);
        } catch (IllegalArgumentException error) {
            if (!message.equals(error.getMessage())) {
                throw new AssertionError(error);
            }
        }
    }
}
EOF
