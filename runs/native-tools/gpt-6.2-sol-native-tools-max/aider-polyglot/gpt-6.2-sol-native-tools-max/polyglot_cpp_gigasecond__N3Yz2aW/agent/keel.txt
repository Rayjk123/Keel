int main() {
    using namespace std::chrono;
    using gigasecond::advance;
    using SecondsPoint = time_point<system_clock, seconds>;
    assert(advance(SecondsPoint{seconds{1'422'136'800}}) == SecondsPoint{seconds{2'422'136'800}});
    assert(advance(SecondsPoint{seconds{-329'961'600}}) == SecondsPoint{seconds{670'038'400}});
    assert(advance(SecondsPoint{seconds{1'422'143'999}}) == SecondsPoint{seconds{2'422'143'999}});

    const time_point<system_clock, milliseconds> fractional{milliseconds{1'422'136'800'123}};
    assert(advance(fractional).time_since_epoch().count() == 2'422'136'800'123);
    static_assert(std::is_same_v<decltype(advance(time_point<system_clock, hours>{})), SecondsPoint>);
    assert(advance(time_point<system_clock, hours>{hours{1}}) == SecondsPoint{seconds{1'000'003'600}});

    const auto start = system_clock::from_time_t(1'422'136'800);
    assert(system_clock::to_time_t(advance(start)) == 2'422'136'800);
}

    assert(advance(ptime(date(1977, 6, 13))) == ptime(date(2009, 2, 19), time_duration(1, 46, 40)));

    assert(advance(ptime(date(2015, 1, 24), time_duration(23, 59, 59))) == ptime(date(2046, 10, 3), time_duration(1, 46, 39)));
    assert(advance(ptime(date(2000, 2, 29), hours(12) + microseconds(123))) == ptime(date(2031, 11, 7), time_duration(13, 46, 40) + microseconds(123)));
}
