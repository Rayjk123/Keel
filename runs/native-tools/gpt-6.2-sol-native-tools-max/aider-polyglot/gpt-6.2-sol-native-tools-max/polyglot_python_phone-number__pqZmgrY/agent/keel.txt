import unittest

from phone_number import PhoneNumber


class PhoneNumberTest(unittest.TestCase):
    def test_normalizes_supported_formats(self):
        for text in (
            '+1 (613)-995-0253', '613-995-0253', '1 613 995 0253',
            '613.995.0253', '16139950253', '6139950253',
            '  (613)\t995 0253  ',
        ):
            with self.subTest(text=text):
                phone = PhoneNumber(text)
                self.assertEqual(phone.number, '6139950253')
                self.assertEqual(phone.area_code, '613')
                self.assertEqual(phone.pretty(), '(613) 995-0253')

    def test_rejects_invalid_numbers_with_specific_messages(self):
        for text, message in (
            ('', 'must not be fewer than 10 digits'),
            ('613-995-025', 'must not be fewer than 10 digits'),
            ('12 613 995 0253', 'must not be greater than 11 digits'),
            ('+2 613 995 0253', '11 digits must start with 1'),
            ('013-995-0253', 'area code cannot start with zero'),
            ('+1 113-995-0253', 'area code cannot start with one'),
            ('+1 613-095-0253', 'exchange code cannot start with zero'),
            ('613-195-0253', 'exchange code cannot start with one'),
            ('613-99A-0253', 'letters not permitted'),
            ('613-@95-0253', 'punctuations not permitted'),
            ('613-+995-0253', 'punctuations not permitted'),
            ('++1 613-995-0253', 'punctuations not permitted'),
            ('+613-995-0253', 'punctuations not permitted'),
            ('613-995-025３', 'punctuations not permitted'),
        ):
            with self.subTest(text=text):
                with self.assertRaises(ValueError) as error:
                    PhoneNumber(text)
                self.assertEqual(str(error.exception), message)


unittest.main(verbosity=2)
