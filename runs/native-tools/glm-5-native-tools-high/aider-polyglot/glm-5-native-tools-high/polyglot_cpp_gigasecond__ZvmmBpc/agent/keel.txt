#include "gigasecond.h"
#include <iostream>
#include <cassert>

int main() {
    // Test 1: January 24th, 2015 at 22:00 
    {
        boost::posix_time::ptime birth(boost::gregorian::date(2015, 1, 24), 
                                        boost::posix_time::time_duration(22, 0, 0));
        boost::posix_time::ptime result = gigasecond::advance(birth);
        boost::posix_time::ptime expected(boost::gregorian::date(2046, 10, 2),
                                           boost::posix_time::time_duration(23, 46, 40));
        assert(result == expected);
        std::cout << "Test 1 PASSED" << std::endl;
    }
    
    // Test 2: Simple case at midnight
    {
        boost::posix_time::ptime start(boost::gregorian::date(2000, 1, 1), 
                                        boost::posix_time::time_duration(0, 0, 0));
        boost::posix_time::ptime result = gigasecond::advance(start);
        // 1 gigasecond = 1,000,000,000 seconds
        // = 11574 days + 1 hour + 46 minutes + 40 seconds
        boost::posix_time::ptime expected(boost::gregorian::date(2031, 9, 9),
                                           boost::posix_time::time_duration(1, 46, 40));
        assert(result == expected);
        std::cout << "Test 2 PASSED" << std::endl;
    }
    
    // Test 3: Testing the number of seconds added
    {
        boost::posix_time::ptime start(boost::gregorian::date(1970, 1, 1), 
                                        boost::posix_time::time_duration(0, 0, 0));
        boost::posix_time::ptime result = gigasecond::advance(start);
        boost::posix_time::time_duration diff = result - start;
        assert(diff.total_seconds() == 1000000000LL);
        std::cout << "Test 3 PASSED - exactly 1,000,000,000 seconds added" << std::endl;
    }
    
    std::cout << "\nAll tests passed!" << std::endl;
    return 0;
}
EOF
