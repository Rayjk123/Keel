package main

import (
	"fmt"
)

func parseTrinary(arg string) int64 {
	if len(arg) == 0 {
		return 0
	}
	
	var result int64
	for _, c := range arg {
		result *= 3
		switch c {
		case '0':
			// add 0
		case '1':
			result += 1
		case '2':
			result += 2
		default:
			// Invalid trinary digit
			return 0
		}
	}
	return result
}

func main() {
	// Test case from the problem
	fmt.Printf("parseTrinary(\"102012\") = %d (expected 302)\n", parseTrinary("102012"))
	
	// Additional tests
	fmt.Printf("parseTrinary(\"0\") = %d (expected 0)\n", parseTrinary("0"))
	fmt.Printf("parseTrinary(\"1\") = %d (expected 1)\n", parseTrinary("1"))
	fmt.Printf("parseTrinary(\"2\") = %d (expected 2)\n", parseTrinary("2"))
	fmt.Printf("parseTrinary(\"10\") = %d (expected 3)\n", parseTrinary("10"))
	fmt.Printf("parseTrinary(\"11\") = %d (expected 4)\n", parseTrinary("11"))
	fmt.Printf("parseTrinary(\"12\") = %d (expected 5)\n", parseTrinary("12"))
	fmt.Printf("parseTrinary(\"20\") = %d (expected 6)\n", parseTrinary("20"))
	
	// Invalid trinary tests
	fmt.Printf("parseTrinary(\"3\") = %d (expected 0 - invalid)\n", parseTrinary("3"))
	fmt.Printf("parseTrinary(\"abc\") = %d (expected 0 - invalid)\n", parseTrinary("abc"))
	fmt.Printf("parseTrinary(\"10a\") = %d (expected 0 - invalid)\n", parseTrinary("10a"))

}
