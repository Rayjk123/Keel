package trinary

// ParseTrinary converts a trinary string to its decimal value.
// Any string containing a character other than 0, 1, or 2 (including an
// empty string) is treated as invalid and yields 0.
func ParseTrinary(arg string) int64 {
	if arg == "" {
		return 0
	}
	var value int64
	for _, r := range arg {
		if r < '0' || r > '2' {
			return 0
		}
		value = value*3 + int64(r-'0')
	}
	return value
}
EOF
