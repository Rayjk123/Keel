import java.util.ArrayList;
import java.util.List;

class Series {
    private final String digits;

    Series(String string) {
        this.digits = string;
    }

    List<String> slices(int num) {
        if (num < 0 || num > digits.length()) {
            throw new IllegalArgumentException("slice length " + num + " is invalid for string of length " + digits.length());
        }
        List<String> result = new ArrayList<>();
        for (int i = 0; i + num <= digits.length(); i++) {
            result.add(digits.substring(i, i + num));
        }
        return result;
    }
}
EOF
