VOWELS = "aeiou"


def _translate_word(word):
    if word[:1] in VOWELS or word[:2] in ("xr", "yt"):
        return word + "ay"

    i = 0
    while i < len(word):
        if word[i:i + 2] == "qu":
            i += 2
            break
        if word[i] in VOWELS or (word[i] == "y" and i > 0):
            break
        i += 1
    return word[i:] + word[:i] + "ay"


def translate(text):
    return " ".join(_translate_word(word) for word in text.split())
