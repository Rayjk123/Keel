package hexadecimal

import "fmt"

// ParseHex converts a hexadecimal string (e.g. "10af8c") to its decimal value
// without using any conversion library. An optional "0x"/"0X" prefix is
// accepted. Invalid input (empty string, non-hex characters, or overflow of
// int64) returns an error.
//
// The in, out and errCase parameters are kept from the original stub; only
// in is used.
func ParseHex(in string, out int64, errCase string) (int64, error) {
	s := in
	if len(s) >= 2 && s[0] == '0' && (s[1] == 'x' || s[1] == 'X') {
		s = s[2:]
	}
	if len(s) == 0 {
		return 0, fmt.Errorf("invalid hexadecimal string: %q", in)
	}

	var result int64
	for i := 0; i < len(s); i++ {
		d, ok := digitValue(s[i])
		if !ok {
			return 0, fmt.Errorf("invalid hexadecimal string: %q", in)
		}
		if result > (1<<63-1-int64(d))/16 {
			return 0, fmt.Errorf("hexadecimal value out of range: %q", in)
		}
		result = result*16 + int64(d)
	}
	return result, nil
}

func digitValue(c byte) (byte, bool) {
	switch {
	case c >= '0' && c <= '9':
		return c - '0', true
	case c >= 'a' && c <= 'f':
		return c - 'a' + 10, true
	case c >= 'A' && c <= 'F':
		return c - 'A' + 10, true
	}
	return 0, false
}
EOF
