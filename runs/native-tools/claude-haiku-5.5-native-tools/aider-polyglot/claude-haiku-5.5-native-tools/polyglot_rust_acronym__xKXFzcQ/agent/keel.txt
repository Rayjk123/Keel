pub fn abbreviate(phrase: &str) -> String {
    // Hyphens and whitespace separate words. For each word, the acronym
    // letter is its first alphabetic character (other punctuation is ignored).
    phrase
        .split(|c: char| c.is_whitespace() || c == '-' || c == '_')
        .filter_map(|word| word.chars().find(|c| c.is_alphabetic()))
        .flat_map(|c| c.to_uppercase())
        .collect()
}

#[cfg(test)]
mod tests {
    use super::abbreviate;

    #[test]
    fn examples() {
        assert_eq!(abbreviate("As Soon As Possible"), "ASAP");
        assert_eq!(abbreviate("Liquid-crystal display"), "LCD");
        assert_eq!(abbreviate("Thank George It's Friday!"), "TGIF");
        assert_eq!(abbreviate("Portable Network Graphics"), "PNG");
        assert_eq!(abbreviate("First In, First Out"), "FIFO");
        assert_eq!(abbreviate("GNU Image Manipulation Program"), "GIMP");
        assert_eq!(abbreviate("Complementary metal-oxide semiconductor"), "CMOS");
        assert_eq!(abbreviate("hyper-text   markup language"), "HTML");
    }
}
