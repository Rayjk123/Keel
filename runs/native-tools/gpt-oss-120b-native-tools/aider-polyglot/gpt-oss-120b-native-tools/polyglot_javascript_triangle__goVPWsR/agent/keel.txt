import { Triangle } from './triangle.js';
const cases = [
  {s:[2,2,2], eq:true, iso:true, sca:false},
  {s:[2,2,3], eq:false, iso:true, sca:false},
  {s:[2,3,2], eq:false, iso:true, sca:false},
  {s:[3,2,2], eq:false, iso:true, sca:false},
  {s:[3,4,5], eq:false, iso:false, sca:true},
  {s:[0,0,0], eq:false, iso:false, sca:false},
  {s:[-1,2,2], eq:false, iso:false, sca:false},
  {s:[5,5,10], eq:false, iso:false, sca:false}, // degenerate not a triangle
  {s:[1,2,3], eq:false, iso:false, sca:false}, // 1+2=3 not triangle
];
let allPass=true;
for(const c of cases){
  const t = new Triangle(...c.s);
  const pass = t.isEquilateral===c.eq && t.isIsosceles===c.iso && t.isScalene===c.sca;
  if(!pass){
    console.log('FAIL', c.s, t.isEquilateral, t.isIsosceles, t.isScalene);
    allPass=false;
  }
}
console.log('All pass?', allPass);
