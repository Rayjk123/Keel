from phone_number import PhoneNumber
for number in ('+1 (613)-995-0253', '613-995-0253', '1 613 995 0253', '613.995.0253'):
    phone = PhoneNumber(number)
    assert phone.number == '6139950253'
    assert phone.area_code == '613'
    assert phone.pretty() == '(613)-995-0253'
for number, message in {
    '613995025': 'must not be fewer than 10 digits',
    '116139950253': 'must not be greater than 11 digits',
    '+26139950253': '11 digits must start with 1',
    '6130950253': 'exchange code cannot start with zero',
    '6131950253': 'exchange code cannot start with one',
    '0 13 995 0253': 'area code cannot start with zero',
    '1 (113) 995 0253': 'area code cannot start with one',
    '613-995-02@3': 'punctuations not permitted',
    '613-995-02a3': 'letters not permitted',
    '613+9950253': 'punctuations not permitted',
}.items():
    try:
        PhoneNumber(number)
    except ValueError as error:
        assert str(error) == message, (number, error)
    else:
        raise AssertionError(number)
print('All normalization and validation checks passed.')
