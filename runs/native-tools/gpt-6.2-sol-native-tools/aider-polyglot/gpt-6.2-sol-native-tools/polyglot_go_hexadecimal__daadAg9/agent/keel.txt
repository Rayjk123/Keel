package hexadecimal

import "testing"

func TestParseHex(t *testing.T) {
	for input, want := range map[string]int64{
		"10af8c": 1093516, "008000": 32768, "008080": 32896,
		"000080": 128, "aF": 175, "0": 0,
		"00007fffffffffffffff": 1<<63 - 1,
	} {
		got, err := ParseHex(input)
		if err != nil || got != want {
			t.Errorf("ParseHex(%q) = (%d, %v), want (%d, nil)", input, got, err, want)
		}
	}
	for _, input := range []string{"", "0x1", "-1", "+1", "1 ", "g", "é", "8000000000000000", "10000000000000000"} {
		if got, err := ParseHex(input); err == nil || got != 0 {
			t.Errorf("ParseHex(%q) = (%d, %v), want (0, error)", input, got, err)
		}
	}
}
EOF
