package trinary

import (

  {"102012", 302, true}, {"0", 0, true}, {"00012", 5, true},
  {"", 0, false}, {"123", 0, false}, {"-12", 0, false}, {" 12", 0, false},
  {strconv.FormatInt(math.MaxInt64, 3), math.MaxInt64, true},
  {strconv.FormatUint(uint64(math.MaxInt64)+1, 3), 0, false},
 } {
  got, err := ParseTrinary(tc.input)
  if got != tc.value || (err == nil) != tc.valid { t.Errorf("ParseTrinary(%q) = %d, %v", tc.input, got, err) }
 }
}
