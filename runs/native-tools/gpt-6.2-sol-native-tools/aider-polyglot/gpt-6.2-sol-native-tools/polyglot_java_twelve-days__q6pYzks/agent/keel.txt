class CheckTwelveDays {
    public static void main(String[] args) {
        TwelveDays song = new TwelveDays();
        String first = "On the first day of Christmas my true love gave to me: a Partridge in a Pear Tree.\n";
        String second = "On the second day of Christmas my true love gave to me: two Turtle Doves, and a Partridge in a Pear Tree.\n";
        String third = "On the third day of Christmas my true love gave to me: three French Hens, two Turtle Doves, and a Partridge in a Pear Tree.\n";
        String last = "On the twelfth day of Christmas my true love gave to me: twelve Drummers Drumming, eleven Pipers Piping, ten Lords-a-Leaping, nine Ladies Dancing, eight Maids-a-Milking, seven Swans-a-Swimming, six Geese-a-Laying, five Gold Rings, four Calling Birds, three French Hens, two Turtle Doves, and a Partridge in a Pear Tree.\n";
        assert song.verse(1).equals(first);
        assert song.verses(2, 3).equals(second + "\n" + third);
        assert song.verses(12, 12).equals(last);
        String full = song.sing();
        assert full.startsWith(first + "\n" + second + "\n" + third);
        assert full.endsWith("\n\n" + last);
        String[] verses = full.split("\n\n");
        assert verses.length == 12;
        String[] days = {"first", "second", "third", "fourth", "fifth", "sixth", "seventh", "eighth", "ninth", "tenth", "eleventh", "twelfth"};
        for (int i = 0; i < verses.length; i++) {
            String prefix = "On the " + days[i] + " day of Christmas my true love gave to me: ";
            assert verses[i].startsWith(prefix);
            if (i >= 2) {
                String previousGifts = verses[i - 1].split(": ", 2)[1];
                assert verses[i].stripTrailing().endsWith(", " + previousGifts);
            }
        }
    }
}
EOF
javac -d "$tmp" src/main/java/TwelveDays.java "$tmp/CheckTwelveDays.java" && java -ea -cp "$tmp" CheckTwelveDays
