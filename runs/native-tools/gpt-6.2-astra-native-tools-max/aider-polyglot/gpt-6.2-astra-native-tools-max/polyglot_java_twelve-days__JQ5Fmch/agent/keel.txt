                """;
        TwelveDays song = new TwelveDays();
        if (!song.sing().equals(expected)) throw new AssertionError("full song");
        String[] lines = expected.stripTrailing().split("\n\n");
        for (int start = 1; start <= 12; start++) {
            if (!song.verse(start).equals(lines[start - 1] + "\n")) throw new AssertionError("verse " + start);
            for (int end = start; end <= 12; end++) {
                String range = String.join("\n\n", java.util.Arrays.copyOfRange(lines, start - 1, end)) + "\n";
                if (!song.verses(start, end).equals(range)) throw new AssertionError("range " + start + "-" + end);
            }
        }
        System.out.println("Verified the full song, all 12 verses, and all 78 inclusive ranges.");
