import assert from 'node:assert/strict';
import { Triangle } from './triangle.js';

const cases = [
  [[2, 2, 2], [true, true, false]],
  [[3, 4, 4], [false, true, false]],
  [[4, 3, 4], [false, true, false]],
  [[4, 4, 3], [false, true, false]],
  [[3, 4, 5], [false, false, true]],
  [[0.25, 0.375, 0.5], [false, false, true]],
  [[1, 1, 2], [false, true, false]],
  [[3, 1, 2], [false, false, true]],
];
for (const sides of [[0, 0, 0], [0, 2, 2], [-1, 2, 2], [1, 2, 4], [4, 1, 2], [1, 4, 2], [NaN, 2, 2], [Infinity, Infinity, 1], [3, '3', 3], [3, 3], [2, 2, 2, 2]]) {
  cases.push([sides, [false, false, false]]);
}
for (const [sides, expected] of cases) {
  const triangle = new Triangle(...sides);
  assert.deepEqual([triangle.isEquilateral, triangle.isIsosceles, triangle.isScalene], expected, `sides: ${sides}`);
}
console.log(`${cases.length} triangle cases passed`);

Implemented triangle classification in `triangle.js`, including invalid inputs and cases where two sides sum exactly to the third. Equilateral triangles also count as isosceles. All 19 test cases passed.
