package hexadecimal

import (
    "math"
    "strings"
    "testing"
)

func TestValid(t *testing.T) {
    for input, want := range map[string]int64{
        "0": 0, "008000": 32768, "008080": 32896, "000080": 128,
        "10af8c": 1093516, "ABcDeF": 11259375,
        "7fffffffffffffff": math.MaxInt64, strings.Repeat("0", 50) + "1": 1,
    } {
        got, err := ParseHex(input)
        if got != want || err != nil { t.Errorf("ParseHex(%q) = %d, %v; want %d, nil", input, got, err, want) }
    }
}

func TestInvalid(t *testing.T) {
    for _, input := range []string{"", "10ag", " 12", "12 ", "-1", "0x12", "f０", "a\xff", "8000000000000000", "10000000000000000"} {
        got, err := ParseHex(input)
        if got != 0 || err == nil { t.Errorf("ParseHex(%q) = %d, %v; want 0 and an error", input, got, err) }
    }
}
