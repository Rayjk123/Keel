import java.util.List;

class SeriesCheck {
    static void check(Series series, int length, String... expected) {
        if (!series.slices(length).equals(List.of(expected))) {
            throw new AssertionError("Unexpected slices for length " + length);
        }
    }

    static void checkInvalid(Series series, int length) {
        try {
            series.slices(length);
            throw new AssertionError("Accepted invalid length " + length);
        } catch (IllegalArgumentException expected) {
            // Expected for lengths outside the valid range.
        }
    }

    public static void main(String[] args) {
        Series series = new Series("49142");
        check(series, 3, "491", "914", "142");
        check(series, 4, "4914", "9142");
        check(series, 5, "49142");
        check(series, 1, "4", "9", "1", "4", "2");
        check(new Series("0001"), 2, "00", "00", "01");
        checkInvalid(series, 0);
        checkInvalid(series, -1);
        checkInvalid(series, 6);
        checkInvalid(new Series(""), 1);
        System.out.println("Series checks passed");
    }
}
EOF
