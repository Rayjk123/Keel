import unittest
from phone_number import PhoneNumber


class TestPhoneNumber(unittest.TestCase):
    def test_formats(self):
        for text in ('+1 (613)-995-0253', '613-995-0253', '1 613 995 0253',
                     '613.995.0253', '6139950253', '  (613) 995-0253  '):
            with self.subTest(text=text):
                phone = PhoneNumber(text)
                self.assertEqual(phone.number, '6139950253')
                self.assertEqual(phone.area_code, '613')
                self.assertEqual(phone.pretty(), '(613)-995-0253')

    def test_boundary_digits(self):
        for number in ('2002000000', '9999999999'):
            self.assertEqual(PhoneNumber('+1' + number).number, number)

    def test_errors(self):
        for text, message in (
            ('123456789', 'must not be fewer than 10 digits'),
            ('123456789012', 'must not be greater than 11 digits'),
            ('2 (613) 995-0253', '11 digits must start with 1'),
            ('013-995-0253', 'area code cannot start with zero'),
            ('+1 (113)-995-0253', 'area code cannot start with one'),
            ('613-095-0253', 'exchange code cannot start with zero'),
            ('1 613 195 0253', 'exchange code cannot start with one'),
            ('613-99A-0253', 'letters not permitted'),
            ('613-99@-0253', 'punctuations not permitted'),
        ):
            with self.subTest(text=text):
                with self.assertRaises(ValueError) as raised:
                    PhoneNumber(text)
                self.assertEqual(str(raised.exception), message)

unittest.main()
