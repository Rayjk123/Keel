fn abbreviate(phrase: &str) -> String {
    let mut result = String::new();
    let mut in_word = false;

    for c in phrase.chars() {
        println!("Char: '{}', is_whitespace: {}, is ascii_punctuation: {}, in_word: {}", 
                 c, c.is_whitespace(), c.is_ascii_punctuation(), in_word);
        
        if c.is_whitespace() || c == '-' || c.is_ascii_punctuation() {
            in_word = false;
            println!("  -> Setting in_word to false");
        } else if !in_word {
            in_word = true;
            let upper = c.to_uppercase().next().unwrap();
            result.push(upper);
            println!("  -> Adding '{}'", upper);
        }
    }

    result
}

fn main() {
    let result = abbreviate("Thank George It's Friday!");
    println!("Result: {:?}", result);
}
EOF
