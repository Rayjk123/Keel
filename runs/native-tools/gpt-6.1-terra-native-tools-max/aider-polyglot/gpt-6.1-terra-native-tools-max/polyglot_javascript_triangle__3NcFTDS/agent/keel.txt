import assert from 'node:assert/strict';

const cases = [
  { sides: [2, 2, 2], expected: [true, true, false], name: 'equilateral is also isosceles' },
  { sides: [3, 4, 4], expected: [false, true, false], name: 'isosceles with last two sides equal' },
  { sides: [4, 3, 4], expected: [false, true, false], name: 'isosceles with first and last sides equal' },
  { sides: [4, 4, 3], expected: [false, true, false], name: 'isosceles with first two sides equal' },
  { sides: [3, 4, 5], expected: [false, false, true], name: 'scalene' },
  { sides: [0.4, 0.6, 0.3], expected: [false, false, true], name: 'fractional lengths' },
  { sides: [1, 1, 2], expected: [false, true, false], name: 'degenerate isosceles' },
  { sides: [1, 2, 3], expected: [false, false, true], name: 'degenerate scalene' },
  { sides: [0, 0, 0], expected: [false, false, false], name: 'zero sides' },
  { sides: [0, 1, 1], expected: [false, false, false], name: 'one zero side' },
  { sides: [-1, 1, 1], expected: [false, false, false], name: 'negative side' },
  { sides: [1, 1, 3], expected: [false, false, false], name: 'triangle inequality violation' },
  { sides: [1, 3, 1], expected: [false, false, false], name: 'triangle inequality violation in another order' },
  { sides: [3, 1, 1], expected: [false, false, false], name: 'triangle inequality violation in third order' },
  { sides: [NaN, 1, 1], expected: [false, false, false], name: 'NaN side' },
  { sides: [Infinity, Infinity, Infinity], expected: [false, false, false], name: 'infinite sides' },
  { sides: [1, 1], expected: [false, false, false], name: 'too few sides' },
  { sides: [1, 1, 1, 1], expected: [false, false, false], name: 'too many sides' },
  { sides: ['1', '1', '1'], expected: [false, false, false], name: 'non-numeric sides' },
];

for (const { sides, expected, name } of cases) {
  const triangle = new Triangle(...sides);
  assert.deepEqual([triangle.isEquilateral, triangle.isIsosceles, triangle.isScalene], expected, name);
}

let count = 0;
for (let a = -1; a <= 15; a++) {
  for (let b = -1; b <= 15; b++) {
    for (let c = -1; c <= 15; c++) {
      const ordered = [a, b, c].sort((x, y) => x - y);
      const valid = ordered[0] > 0 && ordered[0] + ordered[1] >= ordered[2];
      const distinct = new Set([a, b, c]).size;
      const triangle = new Triangle(a, b, c);
      assert.deepEqual(
        [triangle.isEquilateral, triangle.isIsosceles, triangle.isScalene],
        [valid && distinct === 1, valid && distinct < 3, valid && distinct === 3],
        `${a}, ${b}, ${c}`,
      );
      count++;
    }
  }
}
console.log(`Passed ${cases.length} targeted cases and ${count} exhaustive integer cases.`);
