import random
import unittest
from phone_number import PhoneNumber


class PhoneNumberTests(unittest.TestCase):
    def test_examples_and_formatting(self):
        for text in (
            '+1 (613)-995-0253',
            '613-995-0253',
            '1 613 995 0253',
            '613.995.0253',
            '(613) 995-0253',
            '6139950253',
            '\t1 (613)\n995-0253  ',
        ):
            with self.subTest(text=text):
                phone = PhoneNumber(text)
                self.assertEqual(phone.number, '6139950253')
                self.assertEqual(phone.area_code, '613')
                self.assertEqual(phone.pretty(), '(613) 995-0253')

    def test_validation_messages(self):
        cases = (
            ('', 'must not be fewer than 10 digits'),
            ('123456789', 'must not be fewer than 10 digits'),
            ('123456789012', 'must not be greater than 11 digits'),
            ('22345678901', '11 digits must start with 1'),
            ('0234567890', 'area code cannot start with zero'),
            ('1234567890', 'area code cannot start with one'),
            ('2230567890', 'exchange code cannot start with zero'),
            ('2231567890', 'exchange code cannot start with one'),
            ('1 023 456 7890', 'area code cannot start with zero'),
            ('1 123 456 7890', 'area code cannot start with one'),
            ('1 223 056 7890', 'exchange code cannot start with zero'),
            ('1 223 156 7890', 'exchange code cannot start with one'),
            ('223-abc-7890', 'letters not permitted'),
            ('223-456-789x', 'letters not permitted'),
            ('223-@:!-7890', 'punctuations not permitted'),
            ('223/456/7890', 'punctuations not permitted'),
            ('223_456_7890', 'punctuations not permitted'),
            ('２234567890', 'punctuations not permitted'),
        )
        for text, message in cases:
            with self.subTest(text=text):
                with self.assertRaises(ValueError) as caught:
                    PhoneNumber(text)
                self.assertEqual(str(caught.exception), message)

    def test_many_valid_numbers(self):
        rng = random.Random(7142)
        for _ in range(10000):
            area = rng.choice('23456789') + ''.join(rng.choices('0123456789', k=2))
            exchange = rng.choice('23456789') + ''.join(rng.choices('0123456789', k=2))
            subscriber = ''.join(rng.choices('0123456789', k=4))
            expected = area + exchange + subscriber
            for formatted in (expected, '1' + expected, f'+1 ({area}) {exchange}-{subscriber}'):
                phone = PhoneNumber(formatted)
                self.assertEqual(phone.number, expected)
                self.assertEqual(phone.area_code, area)
                self.assertEqual(phone.pretty(), f'({area}) {exchange}-{subscriber}')


unittest.main(verbosity=2)
