import java.nio.file.Files;
import java.nio.file.Path;
import java.util.Arrays;

class TwelveDaysCheck {
    private static void check(String label, String expected, String actual) {
        if (!expected.equals(actual)) {
            throw new AssertionError(label + "\nExpected: " + expected + "\nActual:   " + actual);
        }
    }

    public static void main(String[] args) throws Exception {
        String song = Files.readString(Path.of(args[0]));
        String[] verses = song.stripTrailing().split("\n\n");
        TwelveDays twelveDays = new TwelveDays();
        if (verses.length != 12) throw new AssertionError("Expected 12 verses");
        check("full song", song, twelveDays.sing());
        for (int start = 1; start <= 12; start++) {
            check("verse " + start, verses[start - 1] + "\n", twelveDays.verse(start));
            for (int end = start; end <= 12; end++) {
                String expected = String.join("\n\n", Arrays.copyOfRange(verses, start - 1, end)) + "\n";
                check("verses " + start + " through " + end, expected, twelveDays.verses(start, end));
            }
        }
        System.out.println("Passed: full song, all 12 verses, and all 78 valid verse ranges.");
    }
}
EOF
