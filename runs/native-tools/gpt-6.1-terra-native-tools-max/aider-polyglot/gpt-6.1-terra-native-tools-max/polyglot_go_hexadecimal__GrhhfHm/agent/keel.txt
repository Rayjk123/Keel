package hexadecimal_test

import (
	"errors"
	"fmt"
	"math"
	"math/rand"
	"reflect"
	"strings"
	"testing"

	"hexadecimal"
)

func TestAdditionalParseHex(t *testing.T) {
	cases := []struct {
		input string
		want  int64
		err   error
	}{
		{"0", 0, nil},
		{"a", 10, nil},
		{"f", 15, nil},
		{"A", 10, nil},
		{"F", 15, nil},
		{"10af8c", 1093516, nil},
		{"008000", 32768, nil},
		{"008080", 32896, nil},
		{"000080", 128, nil},
		{"7fffffffffffffff", math.MaxInt64, nil},
		{"00000000000000000007fFfFfFfFfFfFfFf", math.MaxInt64, nil},
		{strings.Repeat("0", 1024), 0, nil},
		{"8000000000000000", math.MaxInt64, hexadecimal.ErrRange},
		{"10000000000000000", math.MaxInt64, hexadecimal.ErrRange},
		{"00008000000000000000", math.MaxInt64, hexadecimal.ErrRange},
		{"", 0, hexadecimal.ErrSyntax},
		{" ", 0, hexadecimal.ErrSyntax},
		{" 1", 0, hexadecimal.ErrSyntax},
		{"1\n", 0, hexadecimal.ErrSyntax},
		{"0x10", 0, hexadecimal.ErrSyntax},
		{"-1", 0, hexadecimal.ErrSyntax},
		{"+1", 0, hexadecimal.ErrSyntax},
		{"f_f", 0, hexadecimal.ErrSyntax},
		{"10g", 0, hexadecimal.ErrSyntax},
		{"é", 0, hexadecimal.ErrSyntax},
		{"\x00", 0, hexadecimal.ErrSyntax},
	}
	for _, tc := range cases {
		t.Run(tc.input, func(t *testing.T) {
			got, err := hexadecimal.ParseHex(tc.input)
			if got != tc.want || !errors.Is(err, tc.err) {
				t.Fatalf("ParseHex(%q) = %d, %v; want %d, %v", tc.input, got, err, tc.want, tc.err)
			}
			if err != nil {
				var parseErr *hexadecimal.ParseError
				if !errors.As(err, &parseErr) || parseErr.Num != tc.input || parseErr.Err != tc.err {
					t.Fatalf("ParseHex(%q): unexpected error detail %#v", tc.input, err)
				}
			}
		})
	}
}

func TestAdditionalHandleErrors(t *testing.T) {
	inputs := []string{"ff", "", "8000000000000000", "bad!", "7fffffffffffffff"}
	want := []string{"none", "syntax", "range", "syntax", "none"}
	if got := hexadecimal.HandleErrors(inputs); !reflect.DeepEqual(got, want) {
		t.Fatalf("HandleErrors(%q) = %q, want %q", inputs, got, want)
	}
	if got := hexadecimal.HandleErrors(nil); len(got) != 0 {
		t.Fatalf("HandleErrors(nil) = %q, want empty", got)
	}
}

func TestRoundTrips(t *testing.T) {
	r := rand.New(rand.NewSource(42))
	for i := 0; i < 10000; i++ {
		want := r.Int63()
		input := fmt.Sprintf("%x", want)
		if i%2 == 0 {
			input = strings.ToUpper(input)
		}
		got, err := hexadecimal.ParseHex(input)
		if err != nil || got != want {
			t.Fatalf("ParseHex(%q) = %d, %v; want %d, nil", input, got, err, want)
		}
	}
}
