import assert from 'node:assert/strict';
import { sum } from './sum-of-multiples.js';

const cases = [
  { factors: [3, 5], limit: 20, expected: 78 },
  { factors: [3, 5], limit: 1, expected: 0 },
  { factors: [3, 5], limit: 4, expected: 3 },
  { factors: [3], limit: 7, expected: 9 },
  { factors: [3, 5], limit: 10, expected: 23 },
  { factors: [3, 5], limit: 100, expected: 2318 },
  { factors: [3, 5], limit: 1000, expected: 233168 },
  { factors: [7, 13, 17], limit: 20, expected: 51 },
  { factors: [4, 6], limit: 15, expected: 30 },
  { factors: [5, 6, 8], limit: 150, expected: 4419 },
  { factors: [5, 25], limit: 51, expected: 275 },
  { factors: [43, 47], limit: 10000, expected: 2203160 },
  { factors: [1], limit: 100, expected: 4950 },
  { factors: [], limit: 10000, expected: 0 },
  { factors: [0], limit: 100, expected: 0 },
  { factors: [3, 0], limit: 4, expected: 3 },
  { factors: [0, 1], limit: 1, expected: 0 },
  { factors: [2, 2], limit: 100, expected: 2450 },
  { factors: [-3, 5], limit: 20, expected: 78 },
  { factors: [3, 5], limit: 0, expected: 0 },
];

for (const { factors, limit, expected } of cases) {
  assert.equal(sum(factors, limit), expected, `sum(${JSON.stringify(factors)}, ${limit})`);
}

function reference(factors, limit) {
  let result = 0;
  for (let n = 1; n < limit; n++) {
    if (factors.some(factor => factor !== 0 && n % factor === 0)) result += n;
  }
  return result;
}

for (let i = 0; i < 1000; i++) {
  const factors = Array.from({ length: Math.floor(Math.random() * 8) }, () => Math.floor(Math.random() * 31) - 15);
  const limit = Math.floor(Math.random() * 200);
  assert.equal(sum(factors, limit), reference(factors, limit), `sum(${JSON.stringify(factors)}, ${limit})`);
}
console.log(`${cases.length} examples and 1000 randomized checks passed`);
NODE
