#include "gigasecond.h"
#include <boost/date_time/posix_time/posix_time.hpp>
#include <cassert>
#include <chrono>
#include <type_traits>
#include <iostream>

int main() {
    using namespace boost::posix_time;
    struct TestCase { const char* input; const char* expected; };
    const TestCase cases[] = {
        {"2011-04-25 00:00:00", "2043-01-01 01:46:40"},
        {"1977-06-13 00:00:00", "2009-02-19 01:46:40"},
        {"1959-07-19 00:00:00", "1991-03-27 01:46:40"},
        {"2015-01-24 22:00:00", "2046-10-02 23:46:40"},
        {"2015-01-24 23:59:59", "2046-10-03 01:46:39"},
        {"2000-02-29 12:34:56.123456", "2031-11-09 14:21:36.123456"},
        {"1900-01-01 23:59:59", "1931-09-10 01:46:39"}
    };
    for (const auto& test : cases) {
        const auto start = time_from_string(test.input);
        const auto copy = start;
        const auto actual = gigasecond::advance(start);
        const auto expected = time_from_string(test.expected);
        if (actual != expected) {
            std::cerr << test.input << ": got " << actual << ", expected " << expected << '\n';
            return 1;
        }
        assert(start == copy);
        assert((actual - start).total_seconds() == 1000000000);
    }
    // Check that a standard function pointer may be resolved from the template.
    ptime (*advance_fn)(const ptime&) = &gigasecond::advance;
    assert(advance_fn(time_from_string(cases[0].input)) == time_from_string(cases[0].expected));

    using namespace std::chrono;
    const system_clock::time_point epoch{};
    assert(gigasecond::advance(epoch).time_since_epoch() == seconds{1000000000});
    const time_point<system_clock, milliseconds> millis{milliseconds{123456789}};
    const auto millis_end = gigasecond::advance(millis);
    static_assert(std::is_same<decltype(millis_end), const time_point<system_clock, milliseconds>>::value, "precision retained");
    assert(millis_end - millis == seconds{1000000000});
    const time_point<system_clock, minutes> minute_start{minutes{42}};
    const auto minute_end = gigasecond::advance(minute_start);
    static_assert(std::is_same<decltype(minute_end), const time_point<system_clock, seconds>>::value, "no fractional minute discarded");
    assert(minute_end - minute_start == seconds{1000000000});
    const time_point<steady_clock, microseconds> steady_start{microseconds{-123456789}};
    assert(gigasecond::advance(steady_start) - steady_start == seconds{1000000000});
    std::cout << "All gigasecond checks passed\n";
}
EOF
