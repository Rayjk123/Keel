import java.util.Arrays;
import java.util.List;

class SeriesCheck {
    private static void checkSlices(String input, int length, String... expected) {
        List<String> actual = new Series(input).slices(length);
        if (!actual.equals(Arrays.asList(expected))) {
            throw new AssertionError(input + ", " + length + ": " + actual);
        }
    }

    private static void checkError(Runnable action, String message) {
        try {
            action.run();
            throw new AssertionError("Expected IllegalArgumentException: " + message);
        } catch (IllegalArgumentException e) {
            if (!message.equals(e.getMessage())) {
                throw new AssertionError("Unexpected message: " + e.getMessage());
            }
        }
    }

    public static void main(String[] args) {
        checkSlices("1", 1, "1");
        checkSlices("12", 1, "1", "2");
        checkSlices("35", 2, "35");
        checkSlices("9142", 2, "91", "14", "42");
        checkSlices("49142", 3, "491", "914", "142");
        checkSlices("49142", 4, "4914", "9142");
        checkSlices("777777", 3, "777", "777", "777", "777");
        checkSlices("918493904243", 5, "91849", "18493", "84939", "49390", "93904", "39042", "90424", "04243");
        checkSlices("0001", 2, "00", "00", "01");
        checkError(() -> new Series(""), "series cannot be empty");
        Series series = new Series("12345");
        checkError(() -> series.slices(0), "slice length cannot be negative or zero");
        checkError(() -> series.slices(-1), "slice length cannot be negative or zero");
        checkError(() -> series.slices(Integer.MIN_VALUE), "slice length cannot be negative or zero");
        checkError(() -> series.slices(6), "slice length cannot be greater than series length");
        checkError(() -> series.slices(42), "slice length cannot be greater than series length");
        checkError(() -> series.slices(Integer.MAX_VALUE), "slice length cannot be greater than series length");
        List<String> first = series.slices(2);
        first.clear();
        if (!series.slices(2).equals(Arrays.asList("12", "23", "34", "45"))) {
            throw new AssertionError("Results should be independent");
        }
        System.out.println("All series checks passed.");
    }
}
EOF
