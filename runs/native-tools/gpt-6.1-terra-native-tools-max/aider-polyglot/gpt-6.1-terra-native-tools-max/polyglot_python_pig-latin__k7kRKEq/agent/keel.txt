import unittest
from pig_latin import translate

class PigLatinTests(unittest.TestCase):
    def test_vowel_prefixes(self):
        for english, expected in {
            'apple': 'appleay', 'ear': 'earay', 'igloo': 'iglooay',
            'object': 'objectay', 'under': 'underay',
            'xray': 'xrayay', 'yttria': 'yttriaay',
        }.items():
            with self.subTest(word=english):
                self.assertEqual(translate(english), expected)

    def test_consonant_prefixes(self):
        for english, expected in {
            'pig': 'igpay', 'chair': 'airchay', 'thrush': 'ushthray',
            'school': 'oolschay', 'yellow': 'ellowyay', 'xenon': 'enonxay',
            'liquid': 'iquidlay', 'sky': 'yskay', 'my': 'ymay',
            'rhythm': 'ythmrhay', 'xylem': 'ylemxay', 'yt': 'ytay',
            'squash': 'ashsquay', 'quick': 'ickquay', 'square': 'aresquay',
            'queen': 'eenquay', 'qu': 'quay', 'q': 'qay',
        }.items():
            with self.subTest(word=english):
                self.assertEqual(translate(english), expected)

    def test_text_and_boundaries(self):
        for english, expected in {
            'quick fast run': 'ickquay astfay unray',
            '  apple\tchair\nsquare  ': '  appleay\tairchay\naresquay  ',
            '': '', ' \n\t': ' \n\t', 'Hello, world!': 'elloHay, orldway!',
            "can't stop": "an'tcay opstay",
        }.items():
            with self.subTest(text=english):
                self.assertEqual(translate(english), expected)

unittest.main(verbosity=2)

from pig_latin import translate

examples = {
    'apple': 'appleay', 'xray': 'xrayay', 'yttria': 'yttriaay',
    'pig': 'igpay', 'chair': 'airchay', 'thrush': 'ushthray',
    'quick': 'ickquay', 'square': 'aresquay', 'my': 'ymay',
    'rhythm': 'ythmrhay', 'yellow': 'ellowyay',
    'quick fast run': 'ickquay astfay unray',
    '  apple\tchair\nsquare  ': '  appleay\tairchay\naresquay  ',
    '': '',
}
for text, expected in examples.items():
    actual = translate(text)
    assert actual == expected, (text, actual, expected)
print(f'Passed {len(examples)} translation checks')
