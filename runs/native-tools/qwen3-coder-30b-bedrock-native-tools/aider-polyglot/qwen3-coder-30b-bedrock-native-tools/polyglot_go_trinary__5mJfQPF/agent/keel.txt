package main

import (
	"fmt"
)

func parseTrinary(arg string) (int64, error) {
	result := int64(0)
	
	for _, digit := range arg {
		switch digit {
		case '0', '1', '2':
			result = result*3 + int64(digit-'0')
		default:
			return 0, fmt.Errorf("invalid trinary digit: %c", digit)
		}
	}
	
	return result, nil
}

func main() {
	// Test the example from the problem
	result, err := parseTrinary("102012")
	if err != nil {
		fmt.Printf("Error: %v\n", err)
	} else {
		fmt.Printf("102012 in trinary = %d in decimal\n", result)
		// Verify: 1*3^5 + 0*3^4 + 2*3^3 + 0*3^2 + 1*3^1 + 2*3^0
		//        = 243 + 0 + 54 + 0 + 3 + 2 = 302
		expected := int64(243 + 0 + 54 + 0 + 3 + 2)
		fmt.Printf("Expected: %d\n", expected)
		if result == expected {
			fmt.Println("✓ Correct!")
		} else {
			fmt.Println("✗ Incorrect!")
		}
	}
	
	// Test simple cases
	testCases := []struct{
		input string
		expected int64
	}{
		{"0", 0},
		{"1", 1},
		{"2", 2},
		{"10", 3},
		{"201", 19},
	}
	
	fmt.Println("\nTesting simple cases:")
	for _, tc := range testCases {
		result, err := parseTrinary(tc.input)
		if err != nil {
			fmt.Printf("%s -> Error: %v\n", tc.input, err)
		} else {
			fmt.Printf("%s -> %d (expected %d) %v\n", 
				tc.input, result, tc.expected, 
				map[bool]string{true:"✓", false:"✗"}[result==tc.expected])
		}
	}
	
	// Test invalid cases
	fmt.Println("\nTesting invalid cases:")
	invalidCases := []string{"123", "abc", "12a"}
	for _, input := range invalidCases {
		result, err := parseTrinary(input)
		if err != nil {
			fmt.Printf("%s -> Error: %v ✓\n", input, err)
		} else {
			fmt.Printf("%s -> %d ✗\n", input, result)
		}
	}
}

	tests := []struct {
		input    string

		{"1", 1},
		{"2", 2},
		{"10", 3},
		{"201", 19},
	}

	for _, test := range tests {
		result, _ := ParseTrinary(test.input)
		if result != test.expected {
			t.Errorf("ParseTrinary(%q) = %d, want %d", test.input, result, test.expected)
		}
	}

	// Test invalid inputs
	invalidTests := []string{"123", "abc", "12a"}
	for _, input := range invalidTests {
		_, err := ParseTrinary(input)

		}
	}
}
