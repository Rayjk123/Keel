public class TwelveDays {
    public static void main(String[] args) {
        System.out.println(fullLyrics());
    }

    public static String fullLyrics() {
        StringBuilder lyrics = new StringBuilder();

        int day = 1;
        while (day <= 12) {
            lyrics.append(dayVerse(day)).append("\n");
            day++;
        }

        return lyrics.toString().trim();
    }

    public static String dayVerse(int day) {
        String[] gifts = {
            "a Partridge in a Pear Tree",
            "two Turtle Doves",
            "three French Hens",
            "four Calling Birds",
            "five Gold Rings",
            "six Geese-a-Laying",
            "seven Swans-a-Swimming",
            "eight Maids-a-Milking",
            "nine Ladies Dancing",
            "ten Lords-a-Leaping",
            "eleven Pipers Piping",
            "twelve Drummers Drumming"
        };

        String[] dayPrefixes = {
            "", "second", "third", "fourth", "fifth", "sixth", "seventh", "eighth",
            "ninth", "tenth", "eleventh", "twelfth"
        };

        StringBuilder verse = new StringBuilder();
        verse.append("On the ").append(dayPrefixes[day - 1]).append(" day of Christmas my true love gave to me: ");

        if (day == 1) {
            verse.append(gifts[0]).append(".");
        } else {
            // Add all gifts from current day backwards to day 2
            for (int i = day - 1; i >= 1; i--) {
                if (i > 1) {
                    verse.append(gifts[i]).append(", ");
                } else {
                    verse.append(gifts[i]).append(", ");
                }
            }
            // Add the Partridge in a Pear Tree
            verse.append(gifts[0]).append(".");
        }

        return verse.toString();
    }
}
