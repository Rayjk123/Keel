			return 0
		}
	}
	
	fmt.Printf("Validating: %s\n", arg)
	for i := len(arg) - 1; i >= 0; i-- {
		digit := int64(arg[i] - '0')
		position := len(arg) - 1 - i
		fmt.Printf("  digit '%c' at position %d = %d * 3^%d\n", arg[i], position, digit, position)
		power := int64(1)
		for j := int64(0); j < int64(position); j++ {
			power *= 3
		}
		fmt.Printf("    -> 3^%d = %d, term = %d * %d = %d\n", position, power, digit, power, digit*power)
		result += digit * power
	}
	fmt.Printf("Total: %d\n\n", result)
	return result
}
