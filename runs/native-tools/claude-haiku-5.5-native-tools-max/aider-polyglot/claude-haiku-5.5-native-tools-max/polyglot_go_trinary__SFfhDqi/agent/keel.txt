// expected is an independent reference built on math/big. Invalid input and
// values that do not fit in an int64 both map to 0.
func expected(s string) int64 {
	if s == "" {
		return 0
	}
	v := new(big.Int)
	three := big.NewInt(3)
	for _, r := range s {
		if r < '0' || r > '2' {

		v.Add(v, big.NewInt(int64(r-'0')))
	}
	if !v.IsInt64() {
		return 0
	}
	return v.Int64()
}

func TestInt64Boundary(t *testing.T) {
	maxStr := strconv.FormatInt(math.MaxInt64, 3)
	if got := ParseTrinary(maxStr); got != math.MaxInt64 {
		t.Errorf("max int64 (%s): got %d", maxStr, got)
	}
	// max ends in digit 1, so swapping that digit for 2 gives max+1, which does not fit.
	next := maxStr[:len(maxStr)-1] + "2"
	if got := ParseTrinary(next); got != 0 {
		t.Errorf("max+1 (%s): got %d, want 0", next, got)
	}
	cases := map[string]int64{
		strings.Repeat("2", 39): 4052555153018976266, // 3^39 - 1, fits
		strings.Repeat("2", 40): 0,                   // 3^40 - 1, overflows
		strings.Repeat("1", 40): 6078832729528464400, // (3^40 - 1) / 2, fits
	}
	for in, want := range cases {
		if got := ParseTrinary(in); got != want {
			t.Errorf("ParseTrinary(%q) = %d, want %d", in, got, want)
		}
	}
}

// check compares ParseTrinary against the reference for one input.
func check(t *testing.T, s string) {
	t.Helper()
	got, ok := ParseTrinary(s)
	want, wantOK := reference(s)
	if got != want || ok != wantOK {
		t.Fatalf("ParseTrinary(%q) = (%d, %v), want (%d, %v)", s, got, ok, want, wantOK)
	}
}

type testCase struct {
	in   string
	want int64
	ok   bool
}

func runCases(t *testing.T, cases []testCase) {
	t.Helper()
	for _, c := range cases {
		got, ok := ParseTrinary(c.in)
		if got != c.want || ok != c.ok {
			t.Errorf("ParseTrinary(%q) = (%d, %v), want (%d, %v)", c.in, got, ok, c.want, c.ok)
		}
		check(t, c.in)
	}
}
