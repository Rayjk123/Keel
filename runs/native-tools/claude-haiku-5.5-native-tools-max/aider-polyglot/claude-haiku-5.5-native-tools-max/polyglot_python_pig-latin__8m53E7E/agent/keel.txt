import unittest

from pig_latin import translate


class PigLatinTest(unittest.TestCase):
    def test_vowel_words_get_ay(self):
        for word, expected in [("apple", "appleay"), ("ear", "earay"),
                               ("igloo", "iglooay"), ("object", "objectay"),
                               ("under", "underay"), ("equal", "equalay")]:
            self.assertEqual(translate(word), expected)

    def test_xr_and_yt_are_vowel_like(self):
        self.assertEqual(translate("xray"), "xrayay")
        self.assertEqual(translate("yttria"), "yttriaay")

    def test_leading_consonants_move_to_end(self):
        for word, expected in [("pig", "igpay"), ("koala", "oalakay"),
                               ("dog", "ogday"), ("chair", "airchay"),
                               ("therapy", "erapythay"),
                               ("thrush", "ushthray")]:
            self.assertEqual(translate(word), expected)

    def test_y_is_consonant_at_start(self):
        self.assertEqual(translate("yellow"), "ellowyay")
        self.assertEqual(translate("you"), "ouyay")

    def test_y_after_consonants_acts_as_vowel(self):
        self.assertEqual(translate("my"), "ymay")
        self.assertEqual(translate("rhythm"), "ythmrhay")
        self.assertEqual(translate("myth"), "ythmay")
        self.assertEqual(translate("gym"), "ymgay")
        self.assertEqual(translate("try"), "ytray")

    def test_qu_moves_as_a_unit(self):
        self.assertEqual(translate("quick"), "ickquay")
        self.assertEqual(translate("square"), "aresquay")
        self.assertEqual(translate("queen"), "eenquay")
        self.assertEqual(translate("quiet"), "ietquay")
        self.assertEqual(translate("queue"), "eequay")

    def test_phrase(self):
        self.assertEqual(translate("quick fast run"), "ickquay astfay unray")
        self.assertEqual(translate("The quick brown fox"),
                         "Ethay ickquay ownbray oxfay")

    def test_capitalization_and_punctuation_kept(self):
        self.assertEqual(translate("Hello, world!"), "Ellohay, orldway!")
        self.assertEqual(translate("Quick"), "Ickquay")
        self.assertEqual(translate("OK"), "OKAY")

    def test_contractions_stay_one_word(self):
        self.assertEqual(translate("don't"), "on'tday")
        self.assertEqual(translate("Don't"), "On'tday")
        self.assertEqual(translate("DON'T"), "ON'TDAY")
        self.assertEqual(translate("can\u2019t"), "an\u2019tcay")

    def test_quotes_at_word_edges_are_not_part_of_word(self):
        self.assertEqual(translate("dogs' bowl"), "ogsday' owlbay")
        self.assertEqual(translate("'hello'"), "'ellohay'")

    def test_empty_and_no_letters(self):
        self.assertEqual(translate(""), "")
        self.assertEqual(translate("123 !"), "123 !")


if __name__ == "__main__":
    unittest.main()

import os
import random
import re
import sys

sys.path.insert(0, "/app")
from pig_latin import translate  # noqa: E402

CONS = "bcdfghjklmnpqrstvwxz"  # the 20 consonants other than y
R1 = re.compile(r"(?:[aeiou]|xr|yt)")
R3 = re.compile(r"([^aeiou]*)qu(.*)")
R4 = re.compile(r"([%s]+)y(.*)" % CONS)
R2 = re.compile(r"([^aeiou]*)(.*)")


def ref(w):
    if R1.match(w):
        return w + "ay"
    m = R3.fullmatch(w)
    if m:
        return m.group(2) + m.group(1) + "qu" + "ay"
    m = R4.fullmatch(w)
    if m:
        return "y" + m.group(2) + m.group(1) + "ay"
    m = R2.fullmatch(w)
    return m.group(2) + m.group(1) + "ay"


path = "/usr/share/dict/words"
if os.path.exists(path):
    with open(path) as fh:
        words = [w.strip().lower() for w in fh
                 if w.strip().isalpha() and w.strip().isascii()]
    source = path
else:
    rng = random.Random(1234)
    letters = "abcdefghijklmnopqrstuvwxyz"
    words = ["".join(rng.choice(letters) for _ in range(rng.randint(1, 8)))
             for _ in range(200000)]
    source = "random strings"

diffs = []
for w in words:
    got = translate(w)
    exp = ref(w)
    if got != exp:
        diffs.append((w, got, exp))

print("source:", source, "| words:", len(words), "| differences:", len(diffs))
for w, got, exp in diffs[:25]:
    print(f"  {w!r}: impl={got!r} ref={exp!r}")
