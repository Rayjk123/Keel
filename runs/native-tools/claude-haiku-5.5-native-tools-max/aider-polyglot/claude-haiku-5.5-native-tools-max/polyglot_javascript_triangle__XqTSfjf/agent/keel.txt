import { Triangle } from './triangle';

const EQ = { equilateral: true, isosceles: true, scalene: false };
const ISO = { equilateral: false, isosceles: true, scalene: false };
const SCA = { equilateral: false, isosceles: false, scalene: true };
const NONE = { equilateral: false, isosceles: false, scalene: false };

const kinds = (...sides) => {
  const t = new Triangle(...sides);
  return {
    equilateral: t.isEquilateral,
    isosceles: t.isIsosceles,
    scalene: t.isScalene,
  };
};

describe('getters', () => {
  test('are accessor properties on the class, not methods', () => {
    for (const name of ['isEquilateral', 'isIsosceles', 'isScalene']) {
      const desc = Object.getOwnPropertyDescriptor(Triangle.prototype, name);
      expect(typeof desc.get).toBe('function');
    }
  });
});

describe('equilateral', () => {
  test('all sides are equal', () => expect(kinds(2, 2, 2)).toEqual(EQ));
  test('sides may be floats', () => expect(kinds(0.5, 0.5, 0.5)).toEqual(EQ));
  test('any side is zero is not a triangle', () => expect(kinds(0, 0, 0)).toEqual(NONE));
});

describe('isosceles', () => {
  test('last two sides are equal', () => expect(kinds(3, 4, 4)).toEqual(ISO));
  test('first two sides are equal', () => expect(kinds(4, 4, 3)).toEqual(ISO));
  test('first and last sides are equal', () => expect(kinds(4, 3, 4)).toEqual(ISO));
  test('floats', () => expect(kinds(0.4, 0.6, 0.6)).toEqual(ISO));
  test('degenerate: a + b == c still counts (>= rule)', () => expect(kinds(1, 1, 2)).toEqual(ISO));
});

describe('scalene', () => {
  test('no sides are equal', () => expect(kinds(5, 4, 6)).toEqual(SCA));
  test('sides may be floats', () => expect(kinds(0.4, 0.6, 0.3)).toEqual(SCA));
});

describe('not a triangle', () => {
  test.each([
    ['triangle inequality, 1 1 3', 1, 1, 3],
    ['triangle inequality, 3 1 1', 3, 1, 1],
    ['triangle inequality, 1 3 1', 1, 3, 1],
    ['triangle inequality, 7 3 2', 7, 3, 2],
    ['negative side', -3, 4, 5],
    ['all negative', -2, -2, -2],
    ['one zero side', 0, 4, 4],
    ['zero in the middle', 4, 0, 4],
    ['zero last', 3, 4, 0],
    ['Infinity is not a finite length', Infinity, Infinity, Infinity],
    ['NaN side', NaN, 2, 2],
  ])('%s', (_label, a, b, c) => {
    expect(kinds(a, b, c)).toEqual(NONE);
  });

  test.each([
    ['too few sides', [2, 2]],
    ['no sides', []],
    ['too many sides', [2, 2, 2, 2]],
    ['numeric strings are not lengths', ['2', '2', '2']],
  ])('%s', (_label, sides) => {
    expect(kinds(...sides)).toEqual(NONE);
  });
});

describe('agrees with an independent oracle over all integer triples 0..8', () => {
  // Oracle: validity by sorting (smallest + middle >= largest), kind by counting equal pairs.
  const oracle = (a, b, c) => {
    const [x, y, z] = [a, b, c].sort((p, q) => p - q);
    const valid = x > 0 && x + y >= z;
    const equalPairs = (a === b) + (b === c) + (a === c);
    return {
      equilateral: valid && equalPairs === 3,
      isosceles: valid && equalPairs >= 1,
      scalene: valid && equalPairs === 0,
    };
  };

  test('every triple matches', () => {
    let checked = 0;
    for (let a = 0; a <= 8; a++) {
      for (let b = 0; b <= 8; b++) {
        for (let c = 0; c <= 8; c++) {
          expect(kinds(a, b, c)).toEqual(oracle(a, b, c));
          checked++;
        }
      }
    }
    expect(checked).toBe(729);
  });
});

// Ground truth: sides are integers N_i scaled by 10^D. Validity is decided with
// integer arithmetic only (exact while values < 2^53), so the oracle cannot share
// the float rounding it is meant to judge.
const exactValidInt = (P, Q, R) => {
  const [s, m, l] = [P, Q, R].sort((u, v) => u - v);
  return s > 0 && s + m >= l;
};
// Candidate: the float check with a tolerance of k ulps of the largest side.
const floatValid = (x, y, z, k) => {
  const tol = k * Number.EPSILON * Math.max(x, y, z);
  return x > 0 && y > 0 && z > 0 &&
    x + y >= z - tol && y + z >= x - tol && x + z >= y - tol;
};
// Deterministic PRNG so runs are reproducible.
let seed = 12345;
const rnd = () => ((seed = (seed * 1103515245 + 12345) >>> 0) / 4294967296);
const randInt = (lo, hi) => lo + Math.floor(rnd() * (hi - lo + 1));

// Bias samples toward degenerate triples (largest = sum of other two), since that
// is where rounding can flip the answer; also include random triples.
const sample = (D, maxN, count) => {
  const out = [];
  for (let i = 0; i < count; i++) {
    let P = randInt(1, maxN), Q = randInt(1, maxN), R;
    if (i % 2 === 0) R = P + Q;                 // exactly degenerate
    else R = randInt(1, maxN);                  // general
    const arr = [P, Q, R];
    // shuffle so the largest is not always last
    for (let j = arr.length - 1; j > 0; j--) { const k2 = Math.floor(rnd() * (j + 1)); [arr[j], arr[k2]] = [arr[k2], arr[j]]; }
    out.push(arr);
  }
  return out;
};

const evaluate = (label, triples, D, ks) => {
  const scale = 10 ** D;
  const expected = triples.map(([P, Q, R]) => exactValidInt(P, Q, R));
  // sanity: parsing the decimal string gives the same double as N / 10^D
  for (const [P, Q, R] of triples.slice(0, 2000)) for (const N of [P, Q, R]) {
    const s = (N / scale).toFixed(D);
    if (Number(s) !== N / scale) throw new Error('double mismatch for ' + s);
  }
  console.log(`--- ${label}  (D=${D}, n=${triples.length})`);
  for (const k of ks) {
    let fn = 0, fp = 0;
    triples.forEach(([P, Q, R], i) => {
      const got = floatValid(P / scale, Q / scale, R / scale, k);
      if (got !== expected[i]) { got ? fp++ : fn++; }
    });
    console.log(`   k=${String(k).padEnd(2)}  wrong=${String(fn + fp).padEnd(5)} (rejects valid=${fn}, accepts invalid=${fp})`);
  }
};

const ks = [0, 1, 2, 3, 4, 8];
evaluate('small decimals, grid-like', sample(1, 30, 20000), 1, ks);
evaluate('hundredths', sample(2, 300, 20000), 2, ks);
evaluate('thousandths', sample(3, 3000, 20000), 3, ks);
evaluate('mixed large-ish magnitudes', sample(2, 100000, 20000), 2, ks);

// numerators over a common scale 10^D, so comparisons are exact.
const exactKind = (P, Q, R) => {
  const [s, m, l] = [P, Q, R].sort((u, v) => u - v);
  const valid = s > 0 && s + m >= l;
  const eq = (P === Q) + (Q === R) + (P === R);
  return { valid, e: valid && eq === 3, i: valid && eq >= 1, sc: valid && eq === 0 };

const ri = (lo, hi) => lo + Math.floor(rnd() * (hi - lo + 1));

// cases mixed in so the boundary is exercised, not just random values.

  const mode = ri(0, 5);
  if (mode === 0) R = P + Q;                       // exactly degenerate
  else if (mode === 1) R = P;                      // isosceles-ish

  const arr = [P, Q, R];

};

//
// Classifies a triangle as equilateral, isosceles or scalene from its side lengths.
//
// A set of sides only describes a triangle if there are exactly three sides, each a
// finite number greater than zero, and the sum of any two sides is at least the
// length of the third. Sides that fail these checks are none of the three kinds, so
// every getter returns false for them.
//

// Decimal side lengths can carry rounding error after arithmetic: 0.1 + 0.7 is
// 0.7999999999999999, not 0.8. Without a tolerance, the valid degenerate triangle
// (0.1, 0.7, 0.8) would be rejected, and 0.1 + 0.2 would not equal 0.3. Rounding error
// is relative to magnitude, so the tolerance is 4 * Number.EPSILON times the size of
// the values being compared. That is between 4 and 8 units in the last place.
const RELATIVE_TOLERANCE = 4 * Number.EPSILON;

// Whole-number inputs get no tolerance, so their comparisons stay exact. For example,
// 2^52 and 2^52 + 1 are never treated as equal.
const isWhole = (value) => Number.isInteger(value);

// Slack allowed when checking the triangle inequality. Sums of sides accumulate error
// at the size of the largest side, so the largest side sets the scale.
const sumTolerance = (a, b, c) =>
  [a, b, c].every(isWhole) ? 0 : RELATIVE_TOLERANCE * Math.max(a, b, c);

// Whether two sides are equal within rounding error. Only the two sides being compared
// set the scale, so a large side cannot blur a smaller one.
const equalWithin = (x, y) => {
  if (isWhole(x) && isWhole(y)) {
    return x === y;
  }
  return Math.abs(x - y) <= RELATIVE_TOLERANCE * Math.max(Math.abs(x), Math.abs(y));
};

// Returns 'equilateral', 'isosceles' or 'scalene' for a triangle, or null if the sides
// do not form one.
const classify = (sides) => {
  if (sides.length !== 3) {
    return null;
  }

  if (!sides.every((side) => Number.isFinite(side) && side > 0)) {
    return null;
  }

  const [a, b, c] = sides;
  const slack = sumTolerance(a, b, c);
  const isTriangle =
    a + b >= c - slack && b + c >= a - slack && a + c >= b - slack;
  if (!isTriangle) {
    return null;
  }

  const equalPairs = [equalWithin(a, b), equalWithin(b, c), equalWithin(a, c)].filter(
    Boolean,
  ).length;

  if (equalPairs === 3) {
    return 'equilateral';
  }
  return equalPairs > 0 ? 'isosceles' : 'scalene';
};

export class Triangle {
  constructor(...sides) {
    this.sides = sides;
  }

  get isEquilateral() {
    return classify(this.sides) === 'equilateral';
  }

  get isIsosceles() {
    // At least two sides are equal. Equilateral triangles count as isosceles too.
    const kind = classify(this.sides);
    return kind === 'isosceles' || kind === 'equilateral';
  }

  get isScalene() {
    return classify(this.sides) === 'scalene';
  }
}
