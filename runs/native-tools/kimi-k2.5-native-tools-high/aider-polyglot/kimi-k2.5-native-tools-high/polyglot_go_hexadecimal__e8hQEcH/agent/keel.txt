package main

import (
	"fmt"
	"hexadecimal"
)

func testTry(in string, expected int64, shouldFail bool) {
	var out int64
	var errCase string
	hexadecimal.ParseHex(in, &out, &errCase)
	
	if shouldFail {
		if errCase == "" {
			fmt.Printf("FAIL: expected failure for %q, but got success with value %d\n", in, out)
		} else {
			fmt.Printf("PASS: expected failure for %q, got error: %s\n", in, errCase)
		}
	} else {
		if errCase != "" {
			fmt.Printf("FAIL: unexpected error for %q: %s\n", in, errCase)
		} else if out != expected {
			fmt.Printf("FAIL: expected %d for %q, but got %d\n", expected, in, out)
		} else {
			fmt.Printf("PASS: %q -> %d\n", in, out)
		}
	}
}

func main() {
	// Valid hex strings
	testTry("008000", 0x008000, false)   // green
	testTry("008080", 0x008080, false)   // teal
	testTry("000080", 0x000080, false)   // navy
	testTry("10af8c", 0x10af8c, false)   // example from instructions
	testTry("0", 0, false)
	testTry("1", 1, false)
	testTry("A", 10, false)
	testTry("a", 10, false)
	testTry("F", 15, false)
	testTry("f", 15, false)
	testTry("FF", 255, false)
	testTry("ff", 255, false)
	testTry("100", 256, false)
	testTry("DEADBEEF", 3735928559, false)
	testTry("deadbeef", 3735928559, false)
	
	// Invalid hex strings
	testTry("", 0, true)         // empty string
	testTry("g", 0, true)        // invalid character
	testTry("10g8", 0, true)     // invalid character in middle
	testTry("xyz", 0, true)      // all invalid characters
}
EOF
