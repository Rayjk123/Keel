import sys

from phone_number import PhoneNumber

# Comprehensive test suite
errors = []

# Valid number tests
valid_numbers = [
    ('+1 (613)-995-0253', '6139950253'),
    ('613-995-0253', '6139950253'),
    ('1 613 995 0253', '6139950253'),
    ('613.995.0253', '6139950253'),
    ('1(613)9950253', '6139950253'),
    ('(613) 995-0253', '6139950253'),
    ('16139950253', '6139950253'),
]

for input_num, expected in valid_numbers:
    try:
        p = PhoneNumber(input_num)
        if p.number != expected:
            errors.append(f'PhoneNumber({input_num!r}).number = {p.number!r}, expected {expected!r}')
    except Exception as e:
        errors.append(f'PhoneNumber({input_num!r}) raised {e}')

# Error tests
error_cases = [
    ('123456789', ValueError, 'must not be fewer than 10 digits'),
    ('12345678901234567', ValueError, 'must not be greater than 11 digits'),
    ('21234567890', ValueError, '11 digits must start with 1'),
    ('0234567890', ValueError, 'area code cannot start with zero'),
    ('1234567890', ValueError, 'area code cannot start with one'),
    ('2220567890', ValueError, 'exchange code cannot start with zero'),
    ('2221567890', ValueError, 'exchange code cannot start with one'),
    ('222@4567890', ValueError, 'punctuations not permitted'),
    ('222abc4567', ValueError, 'letters not permitted'),
]

for input_num, expected_exc, expected_msg in error_cases:
    try:
        p = PhoneNumber(input_num)
        errors.append(f'PhoneNumber({input_num!r}) should have raised {expected_exc.__name__}')
    except expected_exc as e:
        if str(e) != expected_msg:
            errors.append(f'PhoneNumber({input_num!r}) raised {expected_exc.__name__}({e!r}), expected {expected_msg!r}')
    except Exception as e:
        errors.append(f'PhoneNumber({input_num!r}) raised {type(e).__name__}({e!r}), expected {expected_exc.__name__}')

if errors:
    print('FAILED:')
    for e in errors:
        print(f'  {e}')
else:
    print('All tests passed!')
