from typing import *

def translate_word(w):
    v='aeiou'
    lw=w.lower()
    if lw.startswith(('xr','yt')) or lw[0] in v:
        return w+'ay'
    i=0
    while i < len(lw):
        if lw[i] in v:
            break
        if i+1 < len(lw) and lw[i]=='q' and lw[i+1]=='u':
            i+=2
            break
        i+=1
    if i==0:
        return w+'ay'
    if i < len(lw) and lw[i]=='y':
        return w[i:]+w[:i]+'ay'
    return w[i:]+w[:i]+'ay'

samples={'apple':'appleay','xray':'xrayay','yttria':'yttriaay','pig':'igpay','chair':'airchay','thrush':'ushthray','quick':'ickquay','square':'aresquay','my':'ymay','rhythm':'ythmrhay'}
for s,t in samples.items():
    print(s, translate_word(s), t)
