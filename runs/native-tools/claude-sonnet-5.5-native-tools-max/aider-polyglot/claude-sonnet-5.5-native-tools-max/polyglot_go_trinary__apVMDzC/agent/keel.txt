# Independent facts I want to rely on for the overflow boundary tests.
def to_base3(n):
    if n == 0:
        return "0"
    s = ""
    while n:
        s = str(n % 3) + s
        n //= 3
    return s

MAX = 2**63 - 1
print("MaxInt64        =", MAX)
print("MaxInt64 base 3 =", to_base3(MAX), "len", len(to_base3(MAX)))
print("MaxInt64 mod 3  =", MAX % 3, "(last trinary digit)")
print("MaxInt64+1 b3   =", to_base3(MAX + 1))
print("3^39 =", 3**39, "fits:", 3**39 <= MAX, "| 3^40 fits:", 3**40 <= MAX)
print("302 ->", to_base3(302), "| 32091 ->", to_base3(32091))
# check cutoffs used by an overflow guard value > (MAX-digit)//3
for d in (0, 1, 2):
    c = (MAX - d) // 3
    print("digit", d, "cutoff", c, "-> c*3+d =", c*3 + d, "<= MAX:", c*3+d <= MAX, "| (c+1)*3+d <= MAX:", (c+1)*3+d <= MAX)
EOF
