import { sum } from './sum-of-multiples';

// Independent reference: walk every number below the level and keep it if any
// positive base value divides it. Uses a different method than the solution.
const reference = (baseValues, level) => {
  let total = 0;
  for (let n = 1; n < level; n += 1) {
    if (baseValues.some((b) => b > 0 && n % b === 0)) total += n;
  }
  return total;
};

// Small deterministic PRNG so any failure is reproducible.
const makeRng = (seed) => {
  let state = seed >>> 0;
  return () => {
    state = (Math.imul(state, 1664525) + 1013904223) >>> 0;
    return state / 2 ** 32;
  };
};

describe('Sum Of Multiples', () => {
  test('example from the task: level 20 with items 3 and 5', () => {
    expect(sum([3, 5], 20)).toEqual(78);
  });

  test('no multiples within limit', () => {
    expect(sum([3, 5], 1)).toEqual(0);
  });
  test('one factor has multiples within limit', () => {
    expect(sum([3, 5], 4)).toEqual(3);
  });
  test('more than one multiple within limit', () => {
    expect(sum([3], 7)).toEqual(9);
  });
  test('more than one factor with multiples within limit', () => {
    expect(sum([3, 5], 10)).toEqual(23);
  });
  test('each multiple is only counted once', () => {
    expect(sum([3, 5], 100)).toEqual(2318);
  });
  test('a much larger amount of numbers', () => {
    expect(sum([43, 47], 10000)).toEqual(2203160);
  });
  test('no factors means an empty sum', () => {
    expect(sum([], 10000)).toEqual(0);
  });
  test('the only multiple of 0 is 0', () => {
    expect(sum([0], 1)).toEqual(0);
  });
  test('the factor 0 does not affect the sum of multiples of other factors', () => {
    expect(sum([3, 0], 4)).toEqual(3);
  });
  test('solutions using include-exclude must extend to cardinality greater than 3', () => {
    expect(sum([2, 3, 5, 7, 11], 10000)).toEqual(39614537);
  });

  // ---- extra edge cases -------------------------------------------------
  test('the level itself is excluded (multiples must be strictly less)', () => {
    expect(sum([3], 9)).toEqual(3 + 6);
    expect(sum([5], 5)).toEqual(0);
  });
  test('level 0 gives no points', () => {
    expect(sum([3, 5], 0)).toEqual(0);
  });
  test('duplicate base values are only counted once', () => {
    expect(sum([3, 3], 10)).toEqual(18);
  });
  test('order of base values does not matter', () => {
    expect(sum([5, 3], 20)).toEqual(78);
  });
  test('a base value of 1 counts every number below the level', () => {
    expect(sum([1], 1000000)).toEqual(499999500000);
  });
  test('negative base values are ignored and do not hang', () => {
    expect(sum([-3, 5], 20)).toEqual(30);
  });
  test('does not modify its input', () => {
    const baseValues = [5, 3, 3, 0];
    sum(baseValues, 50);
    expect(baseValues).toEqual([5, 3, 3, 0]);
  });
  test('matches a brute-force divisibility check on random inputs', () => {
    const random = makeRng(12345);
    for (let i = 0; i < 1000; i += 1) {
      const count = Math.floor(random() * 6);
      const baseValues = Array.from({ length: count }, () => Math.floor(random() * 25));
      const level = Math.floor(random() * 400);
      expect(sum(baseValues, level)).toEqual(reference(baseValues, level));
    }
  });
});
EOF

I implemented `sum(baseValues, level)` in `/app/sum-of-multiples.js`, and it passes my own tests. The project has no test file, so there was nothing to run against. For each positive base value it collects the multiples below the level into a `Set`, which removes duplicates, then adds them up. Your example (level 20, items 3 and 5) returns 78.

- **Parameter order is my assumption:** the skeleton took no parameters, so I used base values first and level second, e.g. `sum([3, 5], 20)`. That is the usual Exercism convention for this exercise. A caller that passes the level first would break.
- **Edge cases:** the level itself is excluded, and an empty list gives 0. Zero and negative base values are skipped, which also stops a base value of 0 from looping forever.
- **Tests:** I ran a throwaway Jest suite from a temporary directory and deleted it afterwards. It had 19 tests: your example, edge cases I wrote from memory of the standard Exercism ones, and 1,000 random inputs checked against a brute-force count. All passed.
- **Lint:** ESLint's `eslint:recommended` rules found nothing. I couldn't run the project's own `@exercism` ESLint config because it isn't installed here.
