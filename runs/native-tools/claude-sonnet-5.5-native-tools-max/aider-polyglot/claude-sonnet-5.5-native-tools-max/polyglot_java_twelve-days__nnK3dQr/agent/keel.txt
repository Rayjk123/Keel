import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;

public class Check {
    static int failures = 0;

    static void expect(String label, boolean ok) {
        System.out.println((ok ? "PASS  " : "FAIL  ") + label);
        if (!ok) failures++;
    }

    static void expectThrows(String label, Runnable r) {
        try {
            r.run();
            expect(label + " (no exception thrown)", false);
        } catch (IllegalArgumentException e) {
            expect(label + " -> IllegalArgumentException: " + e.getMessage(), true);
        } catch (RuntimeException e) {
            expect(label + " (wrong exception " + e + ")", false);
        }
    }

    public static void main(String[] args) throws Exception {
        String expectedSong = Files.readString(Path.of("expected.txt"), StandardCharsets.UTF_8);
        TwelveDays td = new TwelveDays();

        // 1. Whole song must match the text from the task statement exactly.
        String sung = td.sing();
        expect("sing() equals the lyrics from the task, byte for byte", sung.equals(expectedSong));

        // 2. Every individual verse must equal the matching paragraph of the expected text.
        String[] paragraphs = expectedSong.split("\n\n");
        expect("expected text has 12 paragraphs", paragraphs.length == 12);
        for (int i = 1; i <= 12; i++) {
            String want = paragraphs[i - 1] + (i == 12 ? "" : "") ;
            String got = td.verse(i);
            // paragraphs were split on the blank line, so the trailing "\n" of a verse is
            // only kept on the last one; verse() always ends with exactly one "\n".
            String wantVerse = want.endsWith("\n") ? want : want + "\n";
            expect("verse(" + i + ")", got.equals(wantVerse));
        }

        // 3. Ranges must be the matching slice of the song.
        expect("verses(1, 3)", td.verses(1, 3).equals(
                String.join("\n\n", java.util.Arrays.copyOfRange(paragraphs, 0, 3)) + "\n"));
        expect("verses(4, 6)", td.verses(4, 6).equals(

        expect("verses(5, 5) == verse(5)", td.verses(5, 5).equals(td.verse(5)));
        expect("verses(1, 12) == sing()", td.verses(1, 12).equals(td.sing()));
        expect("verses(12, 12) == verse(12)", td.verses(12, 12).equals(td.verse(12)));

        // 4. Formatting details.
        expect("sing() ends with a single newline", sung.endsWith(".\n") && !sung.endsWith("\n\n"));
        expect("sing() contains no carriage returns", sung.indexOf('\r') < 0);
        expect("sing() has no trailing spaces", !sung.contains(" \n"));

        expectThrows("verse(-1)", () -> td.verse(-1));
