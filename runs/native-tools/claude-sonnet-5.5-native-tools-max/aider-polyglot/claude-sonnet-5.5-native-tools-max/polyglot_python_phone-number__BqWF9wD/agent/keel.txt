"""Differential test: PhoneNumber vs. an independently written, regex-based oracle."""
import collections
import random
import re

from phone_number import PhoneNumber


def oracle(text):
    if re.search(r"[A-Za-z]", text):
        return ("err", "letters not permitted")
    if re.search(r"[^0-9\s+\-.()]", text):
        return ("err", "punctuations not permitted")
    digits = re.sub(r"[^0-9]", "", text)
    if len(digits) < 10:
        return ("err", "must not be fewer than 10 digits")
    if len(digits) > 11:
        return ("err", "must not be greater than 11 digits")
    if len(digits) == 11:
        if not digits.startswith("1"):
            return ("err", "11 digits must start with 1")
        digits = digits[1:]
    m = re.fullmatch(r"(\d)\d\d(\d)\d{6}", digits)
    area, exch = m.group(1), m.group(2)
    if area == "0":
        return ("err", "area code cannot start with zero")
    if area == "1":
        return ("err", "area code cannot start with one")
    if exch == "0":
        return ("err", "exchange code cannot start with zero")
    if exch == "1":
        return ("err", "exchange code cannot start with one")
    return ("ok", digits)


def actual(text):
    try:
        return ("ok", PhoneNumber(text).number)
    except ValueError as exc:
        return ("err", exc.args[0])


rng = random.Random(20240601)
ALPHABET = list("0123456789" * 4) + list(" \t-.()+") + list("aZq") + list("@:!#_/,;'\"")


def gen_random():
    return "".join(rng.choice(ALPHABET) for _ in range(rng.randint(0, 16)))


def gen_structured():
    first = rng.choice("0123456789" if rng.random() < .4 else "23456789")
    exch = rng.choice("0123456789" if rng.random() < .4 else "23456789")
    body = first + rng.choice("0123456789") + rng.choice("0123456789") + exch + \
        "".join(rng.choice("0123456789") for _ in range(6))
    prefix = rng.choice(["", "", "1", "+1", "1 ", "+1 ", "2", "0", "11", "12"])
    text = list(prefix + body)
    for _ in range(rng.randint(0, 4)):          # sprinkle formatting characters
        text.insert(rng.randint(0, len(text)), rng.choice(" -.()+"))
    if rng.random() < .1:                        # occasionally add junk
        text.insert(rng.randint(0, len(text)), rng.choice("aZ@:!#_/"))
    if rng.random() < .1:                        # occasionally drop / add a digit
        if rng.random() < .5 and text:
            text.pop(rng.randrange(len(text)))
        else:
            text.insert(rng.randint(0, len(text)), rng.choice("0123456789"))
    return "".join(text)


outcomes = collections.Counter()
mismatches = []
N = 200_000
for gen in (gen_random, gen_structured):
    for _ in range(N):
        s = gen()
        want, got = oracle(s), actual(s)
        outcomes[got[1] if got[0] == "err" else "OK (valid number)"] += 1
        if want != got:
            mismatches.append((s, want, got))

print(f"checked {2 * N} inputs, mismatches: {len(mismatches)}")
for s, want, got in mismatches[:10]:
    print(f"  {s!r}: oracle={want} actual={got}")
print("outcome coverage:")
for key, count in sorted(outcomes.items(), key=lambda kv: -kv[1]):
    print(f"  {count:>7}  {key}")

- **Cleaning:** `PhoneNumber(text).number` returns the 10-digit number. It removes spaces, dots, dashes, parentheses and `+`, and drops a leading country code `1`. All four formats from your examples give `6139950253`.

The checks run in this order: letters, other punctuation, digit count, the 11-digit-must-start-with-1 rule, then the area code and exchange code. Letters and punctuation come first so that `523-abc-7890` reports the letters, not a digit count.
