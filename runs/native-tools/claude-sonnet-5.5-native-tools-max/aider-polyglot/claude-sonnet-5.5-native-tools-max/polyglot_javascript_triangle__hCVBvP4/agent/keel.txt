import { Triangle } from './triangle';

const kinds = (...sides) => {
  const t = new Triangle(...sides);
  return { eq: t.isEquilateral, iso: t.isIsosceles, sc: t.isScalene };
};
const NONE = { eq: false, iso: false, sc: false };

describe('degenerate triangles (a + b == c) are valid because the rule is >=', () => {
  test('1,2,3 is scalene', () => expect(kinds(1, 2, 3)).toEqual({ eq: false, iso: false, sc: true }));
  test('3,2,1 is scalene (order does not matter)', () => expect(kinds(3, 2, 1)).toEqual({ eq: false, iso: false, sc: true }));
  test('2,2,4 is isosceles', () => expect(kinds(2, 2, 4)).toEqual({ eq: false, iso: true, sc: false }));
  test('4,2,2 is isosceles', () => expect(kinds(4, 2, 2)).toEqual({ eq: false, iso: true, sc: false }));
});

describe('just barely invalid', () => {
  test('1,2,4 is not a triangle', () => expect(kinds(1, 2, 4)).toEqual(NONE));
  test('2,2,5 is not a triangle', () => expect(kinds(2, 2, 5)).toEqual(NONE));
  test('5,2,2 is not a triangle', () => expect(kinds(5, 2, 2)).toEqual(NONE));
});

describe('invalid sides', () => {
  test('one zero side', () => expect(kinds(0, 4, 4)).toEqual(NONE));
  test('zero + equal others is not isosceles', () => expect(kinds(4, 0, 4)).toEqual(NONE));
  test('negative side', () => expect(kinds(-2, 2, 2)).toEqual(NONE));
  test('all negative', () => expect(kinds(-3, -3, -3)).toEqual(NONE));
  test('negatives that would "pass" the inequality', () => expect(kinds(-1, -1, -1)).toEqual(NONE));
  test('NaN', () => expect(kinds(NaN, 2, 2)).toEqual(NONE));
  test('all NaN (NaN !== NaN must not look scalene)', () => expect(kinds(NaN, NaN, NaN)).toEqual(NONE));
  test('Infinity', () => expect(kinds(Infinity, Infinity, Infinity)).toEqual(NONE));
  test('one Infinity', () => expect(kinds(Infinity, 2, 2)).toEqual(NONE));
  test('numeric strings are not accepted', () => expect(kinds('2', '2', '2')).toEqual(NONE));
  test('mixed string would concatenate if coerced', () => expect(kinds('1', '1', 1)).toEqual(NONE));
  test('null / undefined', () => {
    expect(kinds(null, 2, 2)).toEqual(NONE);
    expect(kinds(undefined, 2, 2)).toEqual(NONE);
  });
});

describe('wrong number of sides', () => {
  test('no sides', () => expect(kinds()).toEqual(NONE));
  test('one side', () => expect(kinds(3)).toEqual(NONE));
  test('two sides', () => expect(kinds(3, 3)).toEqual(NONE));
  test('four equal sides', () => expect(kinds(3, 3, 3, 3)).toEqual(NONE));
  test('four sides, three of which would be a triangle', () => expect(kinds(3, 4, 5, 6)).toEqual(NONE));
});

describe('valid shapes', () => {
  test('very large equal sides', () => expect(kinds(1e150, 1e150, 1e150)).toEqual({ eq: true, iso: true, sc: false }));
  test('very small equal sides', () => expect(kinds(1e-9, 1e-9, 1e-9)).toEqual({ eq: true, iso: true, sc: false }));
  test('Number.MAX_VALUE sides overflow the sum to Infinity but remain a valid triangle',
    () => expect(kinds(Number.MAX_VALUE, Number.MAX_VALUE, Number.MAX_VALUE)).toEqual({ eq: true, iso: true, sc: false }));
  test('3-4-5 right triangle is scalene', () => expect(kinds(3, 4, 5)).toEqual({ eq: false, iso: false, sc: true }));
});

describe('structural properties', () => {
  const samples = [
    [2, 2, 2], [3, 4, 4], [4, 4, 3], [4, 3, 4], [2, 3, 4], [5, 4, 6], [1, 1, 3], [7, 3, 2],
    [0, 0, 0], [1, 2, 3], [2, 2, 4], [0.5, 0.5, 0.5], [-1, 2, 2], [3, 4], [3, 3, 3, 3],
  ];
  test.each(samples)('%p: equilateral implies isosceles; scalene excludes isosceles; getters are repeatable', (...s) => {
    const t = new Triangle(...s);
    if (t.isEquilateral) expect(t.isIsosceles).toBe(true);
    expect(t.isScalene && t.isIsosceles).toBe(false);
    expect(t.isEquilateral && t.isScalene).toBe(false);
    // Reading getters twice must give the same answer (no hidden state is consumed).
    expect([t.isEquilateral, t.isIsosceles, t.isScalene]).toEqual([t.isEquilateral, t.isIsosceles, t.isScalene]);
    // Every getter returns a real boolean, never undefined/0/1.
    for (const v of [t.isEquilateral, t.isIsosceles, t.isScalene]) expect(typeof v).toBe('boolean');
  });

  test('side order never changes the classification', () => {
    const perms = [[3, 4, 4], [4, 3, 4], [4, 4, 3]];
    const results = perms.map((p) => kinds(...p));
    results.forEach((r) => expect(r).toEqual(results[0]));
  });

  test('the original array passed via spread is not aliased/mutated', () => {
    const input = [3, 4, 5];
    const t = new Triangle(...input);
    input[0] = 100;
    expect(t.isScalene).toBe(true);
  });
});

`Triangle` is implemented in `/app/triangle.js`, and it passes my tests. I couldn't run the project's own `npm test`, because `@exercism/babel-preset-javascript` isn't installed here and `babel.config.js` needs it.

- **Validity:** a shape is a triangle only if it has exactly three sides, each a finite number greater than 0, and it satisfies `a+b ≥ c`, `b+c ≥ a` and `a+c ≥ b`. Degenerate shapes like `1,2,3` count as valid.
- **Invalid shapes:** the constructor doesn't throw. An invalid shape returns `false` from `isEquilateral`, `isIsosceles` and `isScalene`.
- **Classification:** I count the distinct side lengths. One distinct length is equilateral, at most two is isosceles (so equilateral triangles are also isosceles), and three is scalene.
- **Number check:** I use `Number.isFinite`, which doesn't coerce. Strings, `NaN` and `Infinity` are rejected, and `'1'+'1'` can't concatenate into a valid-looking sum.

**Testing:**
- I ran the code in a scratch directory with a temporary Babel config built on the installed `@babel/preset-env`.
- There were 21 tests I wrote in the style of the usual Exercism suite, plus 45 edge-case tests (degenerate and invalid shapes, wrong side counts, `NaN`, `Infinity`, strings). All 66 passed.
