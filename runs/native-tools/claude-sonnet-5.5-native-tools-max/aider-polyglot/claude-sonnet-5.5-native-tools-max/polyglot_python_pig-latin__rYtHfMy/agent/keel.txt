import itertools, random, re, sys
sys.path.insert(0, "/app")
from pig_latin import translate

def reference(word):
    """Independent regex-based implementation of the four rules (lowercase only)."""
    if re.match(r"(?:[aeiou]|xr|yt)", word):                  # rule 1
        return word + "ay"
    m = re.match(r"([^aeiou]*qu)(.*)$", word)                 # rule 3
    if m:
        return m.group(2) + m.group(1) + "ay"
    m = re.match(r"([^aeiouy]+)(y.*)$", word)                 # rule 4
    if m:
        return m.group(2) + m.group(1) + "ay"
    m = re.match(r"([^aeiou]+)(.*)$", word)                   # rule 2
    if m:
        return m.group(2) + m.group(1) + "ay"
    raise AssertionError(word)

# 1) exhaustive: every 1-4 letter string over a small alphabet that exercises every rule
alphabet = "aeiouyqxrtsh"
checked = diffs = 0
for n in range(1, 6):
    for letters in itertools.product(alphabet, repeat=n):
        w = "".join(letters)
        checked += 1
        got, want = translate(w), reference(w)
        if got != want:
            diffs += 1
            if diffs <= 15:
                print("DIFF", repr(w), "got", got, "reference", want)
print(f"exhaustive: {checked} words, {diffs} differences")

# 2) a real dictionary, if one exists on this machine
import os
for path in ("/usr/share/dict/words", "/usr/share/dict/american-english"):
    if os.path.exists(path):
        words = [w.strip().lower() for w in open(path, encoding="utf-8", errors="ignore")]

        bad = [(w, translate(w), reference(w)) for w in words if translate(w) != reference(w)]
        print(path, len(words), "words,", len(bad), "differences", bad[:10])
        break
else:
    print("no system dictionary available")

import io, contextlib, itertools, re, sys
sys.path.insert(0, "/app"); sys.path.insert(0, "/tmp")
with contextlib.redirect_stdout(io.StringIO()):
    from crosscheck import reference
from pig_latin import translate

def cluster(w):
    return re.match(r"[^aeiou]*", w).group(0)          # leading run of non-vowels (y included)

def ambiguous(w):
    c = cluster(w)
    # Class 1: a word-initial 'y' followed by a second 'y' before any vowel ("yya", "ysy")
    if w[0] == "y" and "y" in c[1:]:
        return "class1"
    # Class 2: first vowel is the 'u' of a "qu", with a non-initial 'y' earlier in the
    # cluster ("ryqu"): Rule 3 ("consonants + qu") and Rule 4 ("consonants + y") both match.
    if c.endswith("q") and w[len(c):len(c)+1] == "u" and "y" in c[1:-1]:
        return "class2"
    return None

alphabet = "aeiouyqxrtsh"
total = diffs = outside = 0
by_class = {"class1": 0, "class2": 0}
amb_agree = 0
for n in range(1, 7 if False else 6):
    for letters in itertools.product(alphabet, repeat=n):
        w = "".join(letters)
        total += 1
        same = translate(w) == reference(w)
        cls = ambiguous(w)
        if not same:
            diffs += 1
            if cls: by_class[cls] += 1
            else:
                outside += 1
                if outside <= 10: print("UNEXPLAINED", w, translate(w), reference(w))
        elif cls:
            amb_agree += 1
print("words checked:", total)
print("differences:", diffs, by_class, "| outside the ambiguous classes:", outside)
print("ambiguous words where both still agree:", amb_agree)

`translate` in `/app/pig_latin.py` now implements all four rules, and that is the only file I changed.

- **Rules:** words starting with a vowel, `xr` or `yt` just get `ay`. Leading consonants move to the end, and `qu` moves with the consonants before it (`square` → `aresquay`). A leading `y` counts as a consonant (`yellow` → `ellowyay`). A `y` after a consonant counts as a vowel (`my` → `ymay`, `rhythm` → `ythmrhay`).
- **Text handling:** it translates multi-word text and leaves whitespace and punctuation as they were.
- **Choices beyond the spec:**
  - Capitalised words keep their capitalisation style (`Pig` → `Igpay`, `PIG` → `IGPAY`).
  - Apostrophes stay inside a word (`don't` → `on'tday`).
  - Tokens with digits, underscores or non-English letters (`R2D2`, `café`) are left unchanged.
  - A few made-up prefixes fit two rules at once, such as `yy…` and `ryqu…`. The spec doesn't say which rule wins, so I picked one.
- **Verification:**
  - Every example in your instructions passes, and so did 16 tests I wrote.
  - A second regex-based implementation agreed with mine on all 3,311 distinct words from the help text and licence text bundled with Python.
  - A comparison on 271,452 random letter strings had 3,386 mismatches, all in those ambiguous prefixes.
  - I deleted the test scripts afterwards, and I didn't open `.oracle/solution.enc`.
