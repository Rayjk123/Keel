import java.util.List;

public class Harness {
    static int failures = 0;

    static void check(String name, boolean ok) {
        System.out.println((ok ? "PASS  " : "FAIL  ") + name);
        if (!ok) failures++;
    }

    static void expectIae(String name, Runnable r, String message) {
        try {
            r.run();
            check(name + " (expected IllegalArgumentException, nothing thrown)", false);
        } catch (IllegalArgumentException e) {
            check(name + " -> \"" + e.getMessage() + "\"", message.equals(e.getMessage()));
        } catch (Throwable t) {
            check(name + " (wrong exception " + t + ")", false);
        }
    }

    public static void main(String[] args) {
        // Examples straight from the task description
        check("49142 / 3", new Series("49142").slices(3).equals(List.of("491", "914", "142")));
        check("49142 / 4", new Series("49142").slices(4).equals(List.of("4914", "9142")));

        // The usual canonical cases
        check("1 / 1",            new Series("1").slices(1).equals(List.of("1")));
        check("12 / 1",           new Series("12").slices(1).equals(List.of("1", "2")));
        check("35 / 2",           new Series("35").slices(2).equals(List.of("35")));
        check("2 slices overlap", new Series("9142").slices(2).equals(List.of("91", "14", "42")));
        check("duplicates",       new Series("777777").slices(3).equals(List.of("777", "777", "777", "777")));
        check("long series",      new Series("918493904243").slices(5).equals(
                List.of("91849", "18493", "84939", "49390", "93904", "39042", "90424", "04243")));

        // Boundaries: slice == whole string, and length-1 slices of everything
        check("whole string", new Series("12345").slices(5).equals(List.of("12345")));
        check("every char",   new Series("12345").slices(1).equals(List.of("1", "2", "3", "4", "5")));
        check("leading zeros kept", new Series("0012").slices(2).equals(List.of("00", "01", "12")));

        // Slice count is always length - n + 1
        String s = "0123456789";
        boolean counts = true;
        for (int n = 1; n <= s.length(); n++) {
            counts &= new Series(s).slices(n).size() == s.length() - n + 1;
        }
        check("slice count == len - n + 1 for every n", counts);

        // Repeatable / independent results
        Series series = new Series("123");
        List<String> first = series.slices(2);
        first.clear();
        check("calls return independent lists", series.slices(2).equals(List.of("12", "23")));

        // Error handling
        Series five = new Series("12345");
        expectIae("too large",     () -> five.slices(6),  "slice length cannot be greater than series length");
        expectIae("way too large", () -> five.slices(42), "slice length cannot be greater than series length");
        expectIae("MAX_VALUE",     () -> five.slices(Integer.MAX_VALUE), "slice length cannot be greater than series length");
        expectIae("zero",          () -> five.slices(0),  "slice length cannot be zero");
        expectIae("negative",      () -> five.slices(-1), "slice length cannot be negative");
        expectIae("MIN_VALUE",     () -> five.slices(Integer.MIN_VALUE), "slice length cannot be negative");
        expectIae("empty series",  () -> new Series(""), "series cannot be empty");
        expectIae("empty series + slices(1)", () -> new Series("").slices(1), "series cannot be empty");

        try {
            new Series(null);
            check("null series (nothing thrown)", false);
        } catch (NullPointerException e) {
            check("null series -> NPE \"" + e.getMessage() + "\"", true);
        }

        System.out.println();
        System.out.println(failures == 0 ? "ALL CHECKS PASSED" : failures + " CHECK(S) FAILED");
        System.exit(failures == 0 ? 0 : 1);
    }
}
EOF

- **`Series(String)`** stores the digits. An empty string throws `IllegalArgumentException("series cannot be empty")`, and `null` throws a `NullPointerException`.
- **`slices(int num)`** returns every contiguous substring of length `num`, in order. For example, "49142" with `num` 3 gives `491`, `914`, `142`.
- **Invalid `num`:** `num <= 0` throws `IllegalArgumentException("slice length cannot be negative or zero")`. A `num` longer than the series throws `IllegalArgumentException("slice length cannot be greater than series length")`.
