#pragma once
#include <exception>
#include <functional>
#include <iostream>
#include <sstream>
#include <string>
#include <vector>

namespace mini_catch {
struct TestCase { std::string name; std::function<void()> fn; };
inline std::vector<TestCase>& registry() { static std::vector<TestCase> r; return r; }
struct Registrar {
    Registrar(const char* name, void (*fn)()) { registry().push_back({name, fn}); }
};
struct RequireFailure { std::string msg; };
}  // namespace mini_catch

#define MC_CAT2(a, b) a##b
#define MC_CAT(a, b) MC_CAT2(a, b)
#define TEST_CASE(name, tags)                                                       \
    static void MC_CAT(mc_test_, __LINE__)();                                       \
    static mini_catch::Registrar MC_CAT(mc_reg_, __LINE__)(name, MC_CAT(mc_test_, __LINE__)); \
    static void MC_CAT(mc_test_, __LINE__)()
#define REQUIRE(expr)                                                               \
    do {                                                                            \
        if (!(expr)) {                                                              \
            std::ostringstream mc_os;                                               \
            mc_os << #expr << " at line " << __LINE__;                              \
            throw mini_catch::RequireFailure{mc_os.str()};                          \
        }                                                                           \
    } while (0)

#ifdef CATCH_CONFIG_MAIN
int main() {
    std::size_t failed = 0;
    for (auto& t : mini_catch::registry()) {
        try { t.fn(); std::cout << "PASS  " << t.name << "\n"; }
        catch (const mini_catch::RequireFailure& f) { ++failed; std::cout << "FAIL  " << t.name << ": " << f.msg << "\n"; }
        catch (const std::exception& e) { ++failed; std::cout << "ERROR " << t.name << ": " << e.what() << "\n"; }
    }
    std::cout << (mini_catch::registry().size() - failed) << "/" << mini_catch::registry().size() << " passed\n";
    return failed ? 1 : 0;
}
#endif
EOF

#include "gigasecond.h"
#include "test/catch.hpp"

using namespace boost::posix_time;
using namespace boost::gregorian;

// ---- canonical cases -------------------------------------------------------
TEST_CASE("date only specification of time", "[canon]") {
    const auto actual = gigasecond::advance(ptime(date(2011, 4, 25)));
    const auto expected = ptime(date(2043, 1, 1), time_duration(1, 46, 40));
    REQUIRE(expected == actual);
}
TEST_CASE("second test for date only specification of time", "[canon]") {
    const auto actual = gigasecond::advance(ptime(date(1977, 6, 13)));
    const auto expected = ptime(date(2009, 2, 19), time_duration(1, 46, 40));
    REQUIRE(expected == actual);
}
TEST_CASE("third test for date only specification of time", "[canon]") {
    const auto actual = gigasecond::advance(ptime(date(1959, 7, 19)));
    const auto expected = ptime(date(1991, 3, 27), time_duration(1, 46, 40));
    REQUIRE(expected == actual);
}
TEST_CASE("full time specified", "[canon]") {
    const auto actual = gigasecond::advance(ptime(date(2015, 1, 24), time_duration(22, 0, 0)));
    const auto expected = ptime(date(2046, 10, 2), time_duration(23, 46, 40));
    REQUIRE(expected == actual);
}
TEST_CASE("full time with day roll-over", "[canon]") {
    const auto actual = gigasecond::advance(ptime(date(2015, 1, 24), time_duration(23, 59, 59)));
    const auto expected = ptime(date(2046, 10, 3), time_duration(1, 46, 39));
    REQUIRE(expected == actual);
}

// ---- extra edge cases ------------------------------------------------------
TEST_CASE("lvalue argument is accepted and not modified", "[extra]") {
    const ptime start(date(2015, 1, 24), hours(22));
    ptime mutable_start = start;
    const ptime r = gigasecond::advance(mutable_start);
    REQUIRE(mutable_start == start);
    REQUIRE(r - start == seconds(1000000000));
}
TEST_CASE("difference is exactly 10^9 seconds", "[extra]") {
    const ptime start(date(2000, 2, 29), time_duration(12, 34, 56));   // leap day
    const ptime r = gigasecond::advance(start);
    REQUIRE((r - start).total_seconds() == 1000000000LL);
}
TEST_CASE("start before the unix epoch", "[extra]") {
    const auto actual = gigasecond::advance(ptime(date(1901, 1, 1)));
    REQUIRE(actual == ptime(date(1932, 9, 9), time_duration(1, 46, 40)));
}
TEST_CASE("fractional seconds are preserved", "[extra]") {
    const ptime start = time_from_string("2015-01-24 22:00:00.123456");
    const ptime r = gigasecond::advance(start);
    REQUIRE(r == time_from_string("2046-10-02 23:46:40.123456"));
}
TEST_CASE("special values pass through", "[extra]") {
    REQUIRE(gigasecond::advance(ptime(pos_infin)).is_pos_infinity());
    REQUIRE(gigasecond::advance(ptime(neg_infin)).is_neg_infinity());
    REQUIRE(gigasecond::advance(ptime(not_a_date_time)).is_not_a_date_time());
}
EOF

`gigasecond::advance` is implemented in `gigasecond.h` and `gigasecond.cpp`. It takes a `boost::posix_time::ptime` and returns the moment 1,000,000,000 seconds later. The example from the instructions works: 2015-01-24 22:00:00 gives 2046-10-02 23:46:40.

- **Signature:** The supplied files were empty namespaces, so I chose `advance(const ptime&)`. I recall that being the name Exercism's C++ tests call, but I couldn't confirm it because the test file isn't in `/app`.
- **Implementation:** `ptime` and `seconds` come from Boost.DateTime, which `CMakeLists.txt` already links. The constant is `1000000000LL` so it can't overflow.
- **Checks:**
  - I rebuilt the canonical test cases plus some edge cases and ran them through the project's own `CMakeLists.txt` with a small stand-in for Catch. All 10 passed.
  - I compared 200,010 random and edge-case start times against Python's `datetime + timedelta`. There were no mismatches, and it ran clean under AddressSanitizer and UBSan.
