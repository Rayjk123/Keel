package hexadecimal

import "strconv"

// ParseHex converts a hexadecimal string to its decimal value.
// It returns a *strconv.NumError wrapping strconv.ErrSyntax for invalid
// input and strconv.ErrRange if the value overflows int64.
func ParseHex(in string) (int64, error) {
	if in == "" {
		return 0, &strconv.NumError{Func: "ParseHex", Num: in, Err: strconv.ErrSyntax}
	}
	var out int64
	for _, c := range in {
		var d int64
		switch {
		case c >= '0' && c <= '9':
			d = int64(c - '0')
		case c >= 'a' && c <= 'f':
			d = int64(c-'a') + 10
		case c >= 'A' && c <= 'F':
			d = int64(c-'A') + 10
		default:
			return 0, &strconv.NumError{Func: "ParseHex", Num: in, Err: strconv.ErrSyntax}
		}
		if out > (1<<63-1-d)/16 {
			return 0, &strconv.NumError{Func: "ParseHex", Num: in, Err: strconv.ErrRange}
		}
		out = out*16 + d
	}
	return out, nil
}
