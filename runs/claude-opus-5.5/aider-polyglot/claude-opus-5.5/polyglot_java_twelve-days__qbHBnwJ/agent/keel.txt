class TwelveDays {
    private static final String[] ORDINALS = {
        "first", "second", "third", "fourth", "fifth", "sixth",
        "seventh", "eighth", "ninth", "tenth", "eleventh", "twelfth"
    };

    private static final String[] GIFTS = {
        "a Partridge in a Pear Tree",
        "two Turtle Doves",
        "three French Hens",
        "four Calling Birds",
        "five Gold Rings",
        "six Geese-a-Laying",
        "seven Swans-a-Swimming",
        "eight Maids-a-Milking",
        "nine Ladies Dancing",
        "ten Lords-a-Leaping",
        "eleven Pipers Piping",
        "twelve Drummers Drumming"
    };

    String verse(int verseNumber) {
        StringBuilder sb = new StringBuilder();
        sb.append("On the ").append(ORDINALS[verseNumber - 1])
          .append(" day of Christmas my true love gave to me: ");
        for (int i = verseNumber - 1; i >= 1; i--) {
            sb.append(GIFTS[i]).append(", ");
        }
        if (verseNumber > 1) {
            sb.append("and ");
        }
        sb.append(GIFTS[0]).append(".\n");
        return sb.toString();
    }

    String verses(int startVerse, int endVerse) {
        StringBuilder sb = new StringBuilder();
        for (int i = startVerse; i <= endVerse; i++) {
            if (i > startVerse) {
                sb.append("\n");
            }
            sb.append(verse(i));
        }
        return sb.toString();
    }
    
    String sing() {
        return verses(1, 12);
    }
}
EOF

- **`verse(n)`**: builds the line from a list of ordinals ("first", "second", …) and a list of gifts, counting the gifts down from day *n*. It adds "and" before the Partridge on every day after the first. Each verse ends with a newline.
- **`verses(start, end)`**: joins a range of verses with a blank line between them.
- **`sing()`**: returns `verses(1, 12)`.
