// Package trinary converts trinary strings to decimal values.
package trinary

import (
	"errors"
	"math"
)

// ParseTrinary converts a trinary string to its int64 value.
// Invalid input or overflow yields 0 and an error.
func ParseTrinary(s string) (int64, error) {
	if s == "" {
		return 0, errors.New("empty input")
	}
	var n int64
	for _, r := range s {
		if r < '0' || r > '2' {
			return 0, errors.New("invalid trinary digit")
		}
		d := int64(r - '0')
		if n > (math.MaxInt64-d)/3 {
			return 0, errors.New("overflow")
		}
		n = n*3 + d
	}
	return n, nil
}
EOF
